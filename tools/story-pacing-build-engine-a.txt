    Object.entries(KNIGHTMARES).map(([id, k]) => {
      const c = CLASSES[k.cls];
      return [
        id,
        {
          ...c,
          ...k,
          id,
          short: k.name,
          code: k.name
            .replace(/[^A-Za-z ]/g, '')
            .split(' ')
            .map(w => w[0])
            .join('')
            .slice(0, 2)

          desc: k.desc || `${c.rule}${k.float ? ' Integrated Float System: ignores terrain movement costs.' : ''}`,
        },

    ...Object.fromEntries(
      MAJORS.map(side => [
        side,
        Object.fromEntries(
          Object.entries(KNIGHTMARES)
            .filter(([, k]) => k.side === side && !k.elite && !k.campaign && !k.naval && !k.worldOnly)
            .map(([id, k]) => [k.cls, id]),
        ),

    ),

    cf: { ...LINEUPS.cf2017 },

  // The 2018 U.F.N. deliberately retains the 'cf' engine side for save and
  // diplomacy compatibility, but its equipment belongs to the Black Knights.
  function ufnBlackKnights(g, side) {
    return g?.mode === 'conquest' && g.era === 'ufn2018' && side === 'cf';
  }
  function federationLineup(g, side) {
    if (side !== 'cf') return null;
    return ufnBlackKnights(g, side) ? LINEUPS.bk : LINEUPS.cf2017;
  }
  // Euro Britannia has its own regular heavy and super-heavy in every mode.
  // Scenario-specific mixed rosters may change its other classes, but not these two.
  function typeFor(side, cls, g = null) {
    const federation = federationLineup(g, side);
    if (federation) return federation[cls];
    if (side === 'eb' && (cls === 'heavy' || cls === 'super')) return LINEUPS.eb[cls];
    return g?.lineup?.[side]?.[cls] || ROSTER[side]?.[cls] || ROSTER.britannia[cls];
  }
  function lineupOf(g, side) {
    const federation = federationLineup(g, side);
    if (federation) return { ...federation };
    const lineup = { ...(ROSTER[side] || {}), ...(g?.lineup?.[side] || {}) };
    if (side === 'eb') {
      lineup.heavy = LINEUPS.eb.heavy;
      lineup.super = LINEUPS.eb.super;
    }
    return lineup;
  }
  // A custom scenario's buildable union must not reintroduce Britannian
  // regular heavies into Euro-Britannian factories and automation.
  function buildableTypes(g, side) {
    const available = g?.buildable?.[side] || Object.values(lineupOf(g, side));
    if (side !== 'eb') return available;
    const types = available.map(id => {
      const t = TYPES[id];
      return t && !t.elite && !t.naval && !t.air &&
        (t.cls === 'heavy' || t.cls === 'super') ? LINEUPS.eb[t.cls] : id;
    });
    return [...new Set([...types, LINEUPS.eb.heavy, LINEUPS.eb.super])];
  }
  // Keep explicit scripted scenario units and imported saves era-accurate.
  // Preserve the unit's class and allegiance; only change the equipment ID.
  function eraUnitType(g, type, side) {
    const t = TYPES[type];
    if(side==='eb'&&t?.air)return 'logres_airship';
    // Normalize ordinary armor at creation. Leave Elite Forces, naval units
    // and explicitly neutral special/story models such as Siegfried intact.
    if (side === 'eb' && t && !t.elite && !t.naval && t.side !== 'neutral' &&
        (t.cls === 'heavy' || t.cls === 'super'))
      return LINEUPS.eb[t.cls];
    if (side !== 'cf' || !t) return type;
    // The 2017 Federation flies the Shenlong; the 2018 U.F.N. flies the Ikaruga.
    if (t.air) return airshipType(g, side) || type;
    if (t.naval) {
      // Ordinary ships and marines use the era's navy; Elite marines use
      // their own era-checked deployment path below.
      if (t.elite) return eliteEligible(g, side, ELITE_TYPE_TO_ID[type]) ? type :
        (navalRoster(g, side)[t.tier >= 2 ? 'amphibious2' : 'amphibious'] || type);
      const navy = navalRoster(g, side);
      if (Object.values(navy).includes(type)) return type;
      return navy[t.naval === 'ship' ? 'carrier' : t.tier >= 2 ? 'amphibious2' : 'amphibious'] || type;
    }
    if (t.elite) {
      const id = ELITE_TYPE_TO_ID[type];
      if (eliteEligible(g, side, id)) return type;
      // No cross-era Elite equipment: use a native Elite of the SAME class
      // when one exists, otherwise retain that class's ordinary frame.
      const native = Object.keys(ELITE_FORCES).find(key =>
        eliteEligible(g, side, key) && TYPES[ELITE_FORCES[key].type].cls === t.cls &&
        !TYPES[ELITE_FORCES[key].type].naval);
      return native ? ELITE_FORCES[native].type : typeFor(side, t.cls, g);
    }
    const roster = federationLineup(g, side);
    return Object.values(roster).includes(type) ? type : roster[t.cls] || type;
  }


  // ======== Elite Forces: persistent unique units ========
  const ELITE_TYPE_TO_ID = Object.fromEntries(Object.entries(ELITE_FORCES).map(([id, e]) => [e.type, id]));
  function eliteProfile(profile = {}) {
    profile.elites ||= {};
    for (const [id, e] of Object.entries(ELITE_FORCES)) {
      if (e.scenarioOnly) continue; // Mission ships never enter persistent HQ data.
      const rec = profile.elites[id] || {};
      profile.elites[id] = {
        level: Math.max(0, Math.min(ELITE_MAX_LEVEL, Number.isInteger(rec.level) ? rec.level : 0)),
        fragments: Math.max(0, Number.isFinite(rec.fragments) ? Math.floor(rec.fragments) : 0),
      };
    }
    return profile.elites;
  }
  function eliteRecord(profile, id) {
    return ELITE_FORCES[id] && !ELITE_FORCES[id].scenarioOnly ? eliteProfile(profile)[id] : null;
  }
  function eliteScale(u) {
    if (!u?.elite) return ELITE_LEVEL_SCALE[1];
    return ELITE_LEVEL_SCALE[Math.max(1, Math.min(ELITE_MAX_LEVEL, u.eliteLevel || 1))];
  }
  function eliteFx(u) {
    const e = u?.elite && ELITE_FORCES[u.elite];
    if (!e || (u.eliteLevel || 1) < 3) return {};
    return (u.eliteLevel || 1) >= 5 ? { ...e.lv3, ...e.lv5 } : e.lv3;
  }
  function eliteStats(id, level = 1) {
    const e = ELITE_FORCES[id], t = e && TYPES[e.type];
    if (!e || !t) return null;
    level = Math.max(1, Math.min(ELITE_MAX_LEVEL, level | 0));
    const s = ELITE_LEVEL_SCALE[level],
      ef = level >= 5 ? { ...e.lv3, ...e.lv5 } : level >= 3 ? e.lv3 : {};
    return {
      hp: Math.round(t.hp * s.hp),
      attack: Math.round(t.attack * s.attack),
      armor: t.armor + s.armor,
      move: t.move + (ef.move || 0),
      min: t.min,
      max: t.max + (ef.range || 0),
    };
  }
  function eliteUpgradeReason(profile, id) {
    const e = ELITE_FORCES[id], rec = e && eliteRecord(profile, id);
    if (!rec) return 'Unknown Elite Force';
    if (rec.level >= ELITE_MAX_LEVEL) return 'Maximum Elite level reached';
    // All permanent HQ Elite Forces can be unlocked with fragments, without Story Mode prerequisites.
    const next = rec.level + 1,
      cost = ELITE_UPGRADE_FRAGMENTS[next];
    return rec.fragments < cost
      ? `Need ${cost - rec.fragments} more ${TYPES[e.type].name} fragments`
      : null;
  }
  function upgradeElite(profile, id) {
    const why = eliteUpgradeReason(profile, id);
    if (why) return { ok: false, reason: why };
    const rec = eliteRecord(profile, id),
      next = rec.level + 1,
      cost = ELITE_UPGRADE_FRAGMENTS[next];
    rec.fragments -= cost;
    rec.level = next;
    return { ok: true, level: rec.level, fragments: rec.fragments, cost };
  }
  function grantEliteFragments(profile, awards = {}) {
    const records = eliteProfile(profile);
    for (const [id, amount] of Object.entries(awards))
      if (records[id] && amount > 0) records[id].fragments += Math.floor(amount);
    return records;
  }
  // Old broad faction-wide fragment drops are intentionally disabled.
  // Victory handlers now queue exactly one player-selected Elite reward.
  function eliteVictoryReward(_g, _profile = {}) { return {}; }

  // ======== Commanders (recruitable officers): signature abilities are data, read by the combat rules ========
  const NOFX = {};
  // Calculated once: permanent personal stats and situational skills are displayed separately,
  // but the existing combat pipeline can read both without applying either twice.
  const COMMANDER_EFFECTS = Object.fromEntries(Object.entries(COMMANDERS).map(([id, c]) =>
    [id, { ...(c.stats || {}), ...c.fx }]));
  const COMMANDER_VERSION = 3;
  function commanderStatsText(k) {
    const st = COMMANDERS[k]?.stats || {}, out = [], pct = n => Math.round(n * 100);
    if (st.dmg) out.push(`Damage +${pct(st.dmg)}%${st.attackOnly ? ' on attacks' : ''}`);
    for (const [branch, n] of Object.entries(st.dmgBranch || {})) out.push(`${branch} damage +${pct(n)}%${st.attackOnly ? ' on attacks' : ''}`);
    if (st.crit) out.push(`Critical chance +${pct(st.crit)} percentage points`);
    if (st.critBonus) out.push(`Critical multiplier +${st.critBonus.toFixed(2)}`);
    if (st.pen) out.push(`Armor penetration +${pct(st.pen)} percentage points`);
    if (st.taken) out.push(`Damage taken −${pct(1 - st.taken)}%`);
    if (st.counter) out.push(`Counter-fire +${pct(st.counter)}%`);
    return out.join(' · ');
  }
  // Flat index: 'armor.guns' → node, with its branch and id.
  const TECH_NODES = Object.fromEntries(
    Object.entries(TECH_TREE).flatMap(([b, tree]) =>
      Object.entries(tree.nodes).map(([k, n]) => [
        `${b}.${k}`,
        { ...n, id: `${b}.${k}`, branch: b, max: n.values.length },
      ]),
    ),
  );
  function branchOf(type) {
    const t = TYPES[type];
    if (t.naval === 'ship') return 'naval';
    if (t.naval === 'amphibious') return 'amphibious';
    return BRANCHES[t.branch];
  }
  // No old research purchases migrate; unrelated HQ progress remains intact.
  function normalizeResearch(research = {}) {
    const out = {};
    for (const [id, level] of Object.entries(research || {}))
      if (TECH_NODES[id] && Number.isInteger(level) && level > 0)
        out[id] = Math.min(level, TECH_NODES[id].max);
    return out;
  }
  function techLevel(g, side, id) {
    return g?.tech?.[side]?.[id] || 0;
  }
  function techValue(g, side, id) {
    const l = techLevel(g, side, id);
    return l ? TECH_NODES[id].values[l - 1] : 0;
  }
  // Branch-wide tech for a unit, e.g. unitTech(g, u, 'guns') reads 'armor.guns' for a Sutherland.
  function unitTech(g, u, k) {
    const branch = branchOf(u.type);
    // Infantry Body Armor is a distinct node, so it does not conflict
    // with the Armor branch's own armor-research key.
    if (branch === 'infantry' && k === 'armor')
      return techValue(g, u.side, 'infantry.bodyarmor');
    return techValue(g, u.side, `${branch}.${k}`);
  }
  // Naval units use the Naval HP research instead of inheriting Armor/Artillery frames.
  function hullTech(g, u) {
    return unitTech(g, u, 'hull') + (TYPES[u.type].naval === 'ship' ? techValue(g, u.side, 'naval.hulls') : 0);
  }


  // ======== Commander development ========
  // Two kinds of commander. Scenario commanders come with the operation, sit on their units with fixed stats
  // (g.officers) and are never upgraded. Your commanders (profile.roster) are bought once, upgraded in HQ, kept
  // between operations and assignable in any operation, even beside the scenario's own version (u.personal).
  const HQ_RANK_CAP = PROMOTE_COST.length - 1; // Supreme Commander, the highest rank for every side.
  // Chloe has returned as her own commander; do not migrate her into Toru.
  const EU_ROSTER_REPLACEMENTS = Object.freeze({margaret:'romero',kate:'miguel',hilda:'masaki'});
  const STARTERS = { britannia: ['suzaku'], bk: ['zero'], eu: ['akito'], cf: ['xingke'] };
  function fx(u) {
    return u?.expSilenced ? NOFX : (u?.cmd && COMMANDER_EFFECTS[u.cmd]) || NOFX;
  }
  // Commission determines generic slots; rank determines combat-star caps.
  const COMMISSION_NAMES = { standard: 'Standard', distinguished: 'Distinguished', legendary: 'Legendary' };
  const COMMISSION_ORDER = ['standard', 'distinguished', 'legendary'];
  const HIGH_COMMAND_BASE_CAPACITY = 2;
  const HIGH_COMMAND_MAX_PURCHASED_CAPACITY = 10;
  // Each High Command expansion is permanent; the Challenge Pass bonus is separate.
  const COMMISSION_UPGRADE_PRICES = { distinguished: 200, legendary: 350 };
  const HIGH_COMMAND_EXPANSION_PRICES = { 3: 1000, 4: 2000, 5: 3000, 6: 4000, 7: 5000, 8: 6000, 9: 7000, 10: 8000 };
  function commissionName(c) { return COMMISSION_NAMES[c] || 'Standard'; }
  function commissionOf(k, officer = null) {
    const tier = officer?.commission || COMMANDERS[k]?.commission || 'standard';
    return COMMISSION_NAMES[tier] ? tier : 'standard';
  }
  function genericSlots(k, officer = null) { return COMMISSION_ORDER.indexOf(commissionOf(k, officer)) + 1; }
  function commandStage(rank) { return rank >= 8 ? 2 : rank >= 4 ? 1 : 0; }
  function commandStageName(rank) { return ['Tactical Command', 'Operational Command', 'High Command'][commandStage(rank)]; }
  // Battle credits paid each time a personal commander is assigned. Commission affects generic slots;
  // military rank affects command strength. Neither the frame nor its Elite status affects this cost.
  function assignmentCost(g, k) {
    const officer = g?.roster?.[k];
    return 100 + 40 * COMMISSION_ORDER.indexOf(commissionOf(k, officer))
      + 30 * commandStage(officer?.rank ?? 0);
  }
  function ratingCap(rank, branch = 'armor') {
    // New stars unlock at Captain, Lieutenant Colonel, Brigadier General and Lieutenant General.
    return rank >= 8 ? 6 : rank >= 6 ? 5 : rank >= 4 ? 4 : rank >= 2 ? 3 : 2;
  }
  // Purchased positions run from two to ten; Challenge Pass adds one separate bonus seat.
  function purchasedHighCommandCapacity(profile) {
    return Math.max(HIGH_COMMAND_BASE_CAPACITY,
      Math.min(HIGH_COMMAND_MAX_PURCHASED_CAPACITY, profile?.highCommandCapacity || HIGH_COMMAND_BASE_CAPACITY));
  }
  function highCommandCapacity(profile) {
    return purchasedHighCommandCapacity(profile) +
      (profile?.challengePass?.milestones?.[1600] === true ? 1 : 0);
  }
  function recruitPrice(k) {
    const a = COMMANDERS[k];
    return a?.recruit ?? ({ standard: 200, distinguished: 300, legendary: 400 })[commissionOf(k)];
  }
  function defaultOfficer(k, personal = false) {
    const ratings = { ...RATINGS[k] };
    if (personal) {
      for (const branch of ['infantry', 'armor', 'artillery']) ratings[branch] = Math.min(2, ratings[branch]);
      ratings.mobility = Math.min(ratingCap(0, 'mobility'), ratings.mobility);
    }
    return {
      rank: personal ? 0 : (commissionOf(k) === 'legendary' ? 1 : 0),
      ratings, commission: commissionOf(k), medals: [], honors: [], generics: {},
      spending: { promotions: {}, stars: {} }, genericSpending: {}, commanderVersion: COMMANDER_VERSION,
    };
  }
  function cleanOfficer(k, rec, rankCap = HQ_RANK_CAP) {
    const base = defaultOfficer(k);
    if (!rec) return base;
    const ratings = { ...rec.ratings };
    // Cornelia's old +2 skill movement becomes two Mobility stars, once per record.
    if (k === 'cornelia' && !rec.commanderVersion && ratings.mobility != null)
      ratings.mobility = Math.min(MAX_RATING, ratings.mobility + 2);
    return {
      commanderVersion: COMMANDER_VERSION,
      rank: clamp(Number.isInteger(rec.rank) ? rec.rank : base.rank, 0, rankCap),
      commission: commissionOf(k, rec),
      spending: rec.spending && typeof rec.spending === 'object'
        ? { promotions: { ...(rec.spending.promotions || {}) }, stars: { ...(rec.spending.stars || {}) } }
        : { promotions: {}, stars: {} },
      ratings: Object.fromEntries(
        Object.keys(base.ratings).map(b => [b, clamp((ratings[b] ?? base.ratings[b]) | 0, 1, MAX_RATING)]),
      ),
      medals: (rec.medals || []).filter(m => MEDALS[m]),
      honors: (Array.isArray(rec.honors) ? rec.honors : []).filter(h => h && MEDALS[h.id] && Array.isArray(h.citations)),
      generics: Object.fromEntries(Object.entries(rec.generics || {})
        .filter(([id, level]) => GENERIC_SKILLS[id] && Number.isInteger(level) && level >= 1 && level <= 5)
        .slice(0, genericSlots(k, rec))),
      // Older saves never recorded per-skill spending; use the old price schedule
      // conservatively unless a newer explicit spending record is available.
      genericSpending: Object.fromEntries(Object.entries(rec.generics || {})
        .filter(([id, level]) => GENERIC_SKILLS[id] && Number.isInteger(level) && level >= 1 && level <= 5)
        .map(([id, level]) => [id, Number.isSafeInteger(rec.genericSpending?.[id]) &&
          rec.genericSpending[id] >= 0 ? rec.genericSpending[id] : legacyGenericInvestment(level)])),
    };
  }
  // One-time baseline rebalance. Retain purchased higher commissions and grandfather
  // saved commission slots when demotion would hide invested generic skills.
  const COMMISSION_REBALANCE_PREVIOUS = Object.freeze({
    schneizel: 'distinguished',
    shin: 'distinguished',
    guilford: 'standard',
    ryo: 'distinguished',
    yukiya: 'standard',
    anna: 'standard',
    smilas: 'distinguished',
    fernando: 'standard',
    marirrosa: 'standard',
    honggu: 'standard',
    cao: 'standard',
    leifeng: 'distinguished',
    meiling: 'standard',
    rakshata: 'distinguished',
    cc: 'distinguished',
    senba: 'standard',
    urabe: 'standard',
    dorothea: 'distinguished',
    nonette: 'distinguished',
    marianne: 'legendary',
    oldrin: 'legendary',
    marrybell: 'legendary',
    norland: 'legendary',
    kaguya: 'distinguished',
    vritra: 'standard',
    euphemia: 'distinguished',
    cecile: 'distinguished',
  });
  function rebalanceSavedCommissions(records) {
    if (!records || typeof records !== 'object') return;
    for (const [id, previous] of Object.entries(COMMISSION_REBALANCE_PREVIOUS)) {
      const officer = records[id];
      if (!officer || !COMMANDERS[id]) continue;
      const next = COMMANDERS[id].commission;
      const oldIndex = COMMISSION_ORDER.indexOf(previous);
      const savedIndex = COMMISSION_ORDER.indexOf(officer.commission || previous);
      const newIndex = COMMISSION_ORDER.indexOf(next);
      if (oldIndex < 0 || savedIndex < 0 || newIndex < 0) continue;
      if (newIndex > oldIndex && savedIndex < newIndex) {
        officer.commission = next; // Free new baseline, preserving higher paid upgrades.
      } else if (newIndex < oldIndex && savedIndex === oldIndex) {
        const usedSlots = Object.values(officer.generics || {})
          .filter(level => Number(level) > 0).length;
        if (usedSlots <= newIndex + 1) officer.commission = next;
      }
    }
  }
  // Every faction has a free starting HQ commander; existing roster upgrades are preserved.
  function roster(profile) {
    profile.roster ||= {};
    // Carry retired E.U. commanders' HQ purchases and decorations into their
    // named successors once, retaining all progress without extra slots.
    for(const [oldId,newId] of Object.entries(EU_ROSTER_REPLACEMENTS))if(profile.roster[oldId]){
      profile.roster[newId] ||= profile.roster[oldId];delete profile.roster[oldId];
    }
    for (const k of Object.values(STARTERS).flat())
      if (!profile.roster[k]) profile.roster[k] = defaultOfficer(k, true);
    for (const [k, rec] of Object.entries(profile.roster))
      if (COMMANDERS[k] && rec?.commanderVersion !== COMMANDER_VERSION) {
        // Mobility now starts at 6: return tokens paid for Cornelia's old fifth/sixth star. Only records from before
        // versioning (no commanderVersion) paid for them; later version bumps must not refund again.
        if (k === 'cornelia' && !rec?.commanderVersion) for (let star = 5; star <= Math.min(6, rec?.ratings?.mobility || 4); star++)
          profile.tokens = (profile.tokens || 0) + STAR_COST[star];
        profile.roster[k] = cleanOfficer(k, rec);
      }
    // Apply the adjusted baseline once to existing recruited commanders.
    if (profile.commissionRebalanceVersion !== 1) {
      rebalanceSavedCommissions(profile.roster);
      profile.commissionRebalanceVersion = 1;
    }
    // Migrate shared and equipped legacy medals into an account archive. Original
    // transferable medals have no trustworthy individual recipient; never invent one.
    if (profile.honorsVersion !== 1) {
      const legacy = (profile.legacyHonors ||= []);
      const moveLegacy = (id, formerOwner = null) => {
        if (!MEDALS[id]) return;
        const row = legacy.find(m => m.id === id && m.formerOwner === formerOwner);
        if (row) row.count += 1;
        else legacy.push({ id, formerOwner, count: 1 });
      };
      for (const id of profile.medals || []) moveLegacy(id);
      for (const [k, officer] of Object.entries(profile.roster))
        for (const id of officer.medals || []) moveLegacy(id, k);
      profile.medals = [];
      for (const officer of Object.values(profile.roster)) officer.medals = [];
      profile.honorsVersion = 1;
    }
    // One-time migration: preserve saved ratings as inactive beyond new rank caps.
    if (profile.commanderProgressionVersion !== 1) {
      profile.commanderProgressionVersion = 1;
      profile.highCommandCapacity = purchasedHighCommandCapacity(profile);
      const high = Object.keys(profile.roster).filter(k => COMMANDERS[k] && profile.roster[k].rank >= 8)
        .sort((a, b) => profile.roster[b].rank - profile.roster[a].rank || a.localeCompare(b));
      for (const k of high.slice(highCommandCapacity(profile))) relinquishHighCommand(profile, k, 1);
    }
    return profile.roster;
  }
  function highCommandUsed(profile) {
    return Object.entries(roster(profile)).filter(([k,o]) => COMMANDERS[k] && o.rank >= 8).length;
  }
  function highCommandExpansionPrice(profile) {
    return HIGH_COMMAND_EXPANSION_PRICES[purchasedHighCommandCapacity(profile) + 1] ?? null;
  }
  function expandHighCommandReason(profile) {
    if (purchasedHighCommandCapacity(profile) >= HIGH_COMMAND_MAX_PURCHASED_CAPACITY) return 'All ten purchasable High Command positions unlocked';
    const price = highCommandExpansionPrice(profile);
    return price == null ? 'High Command expansion unavailable' : tokenShort(profile, price);
  }
  function expandHighCommand(profile) {
    const why = expandHighCommandReason(profile);
    if (why) return { ok: false, reason: why };
    profile.tokens -= highCommandExpansionPrice(profile);

  function moraleFloor(g, v) {
    if (leilaCommand(g, v) || starOfMadridNear(g, v)) return 0;

  // Retiring releases one global High Command seat, retaining Commission and generic skills.
  function relinquishHighCommand(profile, k, proportion = 0.9) {
    const o = profile.roster?.[k];
    if (!o || o.rank < 8) return { ok: false, reason: 'Not currently in High Command' };
    const spent = o.spending || { promotions: {}, stars: {} };
    let reimbursable = 0;
    for (let i = 8; i <= o.rank; i++) {
      reimbursable += spent.promotions?.[i] ?? PROMOTE_COST[i];
      if (spent.promotions) delete spent.promotions[i];
    }
    for (const branch of Object.keys(o.ratings)) {
      const cap = ratingCap(7, branch);
      for (let level = cap + 1; level <= o.ratings[branch]; level++) {
        reimbursable += spent.stars?.[branch]?.[level] ??
          (level > (RATINGS[k]?.[branch] || 0) ? (STAR_COST[level] || 0) : 0);
        if (spent.stars?.[branch]) delete spent.stars[branch][level];
      }
      o.ratings[branch] = Math.min(cap, o.ratings[branch]);
    }
    o.rank = 7;
    o.spending = spent;
    const refund = Math.round(reimbursable * proportion);
    profile.tokens = (profile.tokens || 0) + refund;
    return { ok: true, rank: o.rank, refund, eligible: reimbursable };
  }

    for (const branch of Object.keys(o.ratings))
      for (let level = ratingCap(7, branch) + 1; level <= o.ratings[branch]; level++)
        value += o.spending?.stars?.[branch]?.[level] ?? (level > (RATINGS[k]?.[branch] || 0) ? STAR_COST[level] : 0);

  // Full-development planner. Future stars require Supreme Commander rank; all three
  // generic skills require Legendary Commission. These are accounted for in
  // the grand total, without purchasing anything or mutating the roster.
  function commanderPlan(profile, k) {
    const o = roster(profile)[k];
    if (!o) return null;
    const promotions = PROMOTE_COST.slice(o.rank + 1).reduce((a, b) => a + b, 0);
    const stars = Object.fromEntries(Object.keys(BRANCH_NAMES).map(branch => [
      branch, STAR_COST.slice(Math.min(MAX_RATING, o.ratings[branch] || 0) + 1)
        .reduce((a, b) => a + b, 0),
    ]));
    const commissionIndex = COMMISSION_ORDER.indexOf(commissionOf(k, o));
    const commission = COMMISSION_ORDER.slice(commissionIndex + 1)
      .reduce((a, tier) => a + (COMMISSION_UPGRADE_PRICES[tier] || 0), 0);
    const skillLevels = Object.values(o.generics || {}).slice(0, 3);
    const skills = Array.from({ length: 3 }, (_, i) =>
      GENERIC_COSTS.slice((skillLevels[i] || 0) + 1).reduce((a, b) => a + b, 0));
    const starTotal = Object.values(stars).reduce((a, b) => a + b, 0);
    const skillsTotal = skills.reduce((a, b) => a + b, 0);
    return {
      promotions, stars, starTotal, commission, skills, skillsTotal,
      total: promotions + starTotal + commission + skillsTotal,
      highCommandSeat: o.rank < 8 && highCommandUsed(profile) >= highCommandCapacity(profile),
    };
  }
  function genericReason(profile, k, id) {

  function award(g, side, id, reason, unit, key = reason) {
    if (!canEarnHonors(g)) return;
    if (side !== g.player || !MEDALS[id] || !unit?.personal || !unit?.cmd || !COMMANDERS[unit.cmd]) return;
    const events = (g.honorEvents ||= []);
    // Repeat achievements become citations across operations, not spammable
    // identical decorations in the same battle.
    if (events.some(e => e.id === id && e.commander === unit.cmd &&
        e.personal === !!unit.personal && (id !== 'star' || e.key === key))) return;
    events.push({ id, commander: unit.cmd, personal: !!unit.personal,
      reason, key, turn: g.turn });
    log(g, `${MEDALS[id].name}: ${COMMANDERS[unit.cmd].short} — ${reason}.`, side);
  }
  function finalizeHonors(g) {
    if (!canEarnHonors(g) || !g.over || g.honorsFinalized) return;
    g.honorsFinalized = true;
    if (g.over.winner !== g.player) return;
    const services = Object.values(g.honorService || {})
      .filter(s => s.personal && (s.combats || s.kills || s.cities || s.capitals || s.commanderDefeats));
    const operation = g.mode === 'campaign' ? (g.campaign?.id || 'Story Mode') :
      g.mode === 'scenario' ? (g.scenario?.title || g.scenario?.id || 'Scenario') :
        (g.era || 'Conquest');
    for (const s of services) {
      const actions = (s.combats || 0) + (s.cities || 0);
      if (actions >= 20)
        award(g, g.player, 'campaign', `Completed ${actions} attacks and/or captures in the victorious ${operation} operation`,
          { cmd: s.commander, personal: s.personal });
    }
    const laurel = { hard: 'laurel_hard', challenge: 'laurel_challenge', elite: 'laurel_elite' }[g.difficulty];
    if (laurel && g.mode === 'conquest') {
      const best = services.filter(s => s.personal).sort((a, b) => {
        const score = s => 10*s.capitals + 5*s.commanderDefeats + 3*s.cities + Math.min(5,s.kills);
        return score(b)-score(a) || b.capitals-a.capitals ||
          b.commanderDefeats-a.commanderDefeats || b.kills-a.kills ||
          a.commander.localeCompare(b.commander);
      })[0];
      if (best) award(g, g.player, laurel, `Highest contribution to the ${g.difficulty} conquest victory`,
        { cmd: best.commander, personal: true });
    }
  }

  function claimHonors(profile, g, claimId) {
    if (!canEarnHonors(g)) return { awarded: [] };
    roster(profile);
    if (!claimId || (profile.honorClaims ||= {})[claimId]) return { awarded: [] };
    const awarded = [];
    const operation = g.mode === 'campaign' ? (g.campaign?.id || 'Story Mode') :
      g.mode === 'scenario' ? (g.scenario?.id || 'Scenario') : (g.era || 'Conquest');
    const difficulty = g.difficulty || 'normal';
    for (const event of g.honorEvents || []) {
      if (!event.personal || !profile.roster[event.commander] || !MEDALS[event.id]) continue;
      const officer = profile.roster[event.commander];
      const honors = (officer.honors ||= []);
      let medal = honors.find(m => m.id === event.id);
      if (!medal) honors.push(medal = { id: event.id, citations: [] });
      medal.citations.push({ operation, difficulty, reason: event.reason,
        turn: event.turn, claimId });
      awarded.push(event);
    }
    // An old operation save can contain awards from the previous shared-inventory
    // system. Preserve them in the legacy cabinet without assigning recipients.
    for (const old of g.medalsEarned || []) {
      if (!MEDALS[old.id]) continue;
      const row = (profile.legacyHonors ||= []).find(m => m.id === old.id && m.formerOwner === null);
      if (row) row.count += 1;
      else profile.legacyHonors.push({ id: old.id, formerOwner: null, count: 1 });
    }
    profile.honorClaims[claimId] = true;
    return { awarded };
  }

  function shortfall(e, cost) {
    const need = [
      ['credits', 'credits'],
      ['industry', 'industry'],
      ['science', 'research'],
      ['sakuradite', 'Sakuradite'],

  function turnReason(g, side) {

  // Authoritative preflight for tactical actions. This does not consume resources,
  // random rolls, action flags or cooldowns. Specialized executors retain their
  // effects but enter through this same legality check.
  function validateAction(g, action) {
    if (!action || !action.kind) return { ok: false, reason: 'Unknown action.' };
    const { kind, id, c, r } = action;
    if (kind === 'fortress') {
      const station = g.stations.find(s => s.id === id);
      if (!fortressReady(g, station)) return { ok: false, reason: 'The battery is not ready.' };
      const defender = unitAt(g, { c, r });
      if (!defender || !isFoe(g, defender.side, station.owner) ||
          rangedImmuneTo(g,defender,station) ||
          dist(g, station, defender) > fortressRange(g, station))
        return { ok: false, reason: `No enemy unit within ${fortressRange(g, station)} hexes of the city.` };
      return { ok: true, station, defender };
    }
    const actor = g.units.find(u => u.id === id && (kind !== 'deploy' || u.hp > 0));
    if (kind === 'attack') {
      if (!actor) return { ok: false, reason: 'Unit not found.' };
      const why = turnReason(g, actor.side) ||
        (actor.hp <= 0 ? 'Unit destroyed' : actor.expRoloFrozen ? 'Frozen by Absolute Suspension' : actor.expFarneseSlaughterTurn === g.turn ? 'End the Slaughter: no attacks remaining' : (actor.expCharlesMemoryRewrite || actor.morale <= -3) ? 'Unit is confused' : actor.attacked ? 'Already fired' : null);
      if (why) return { ok: false, reason: why };
      const shot = preview(g, id, c, r);
      if (!shot) return { ok: false, reason: atSea(g, actor) ? 'Embarked units cannot fire.' : 'No hostile target in range.' };
      return { ok: true, actor, preview: shot };
    }
    if (kind === 'move') {
      if (!actor) return { ok: false, reason: 'Unit not found.' };
      const destination = tile(g, c, r);
      const cost = destination && reachable(g, actor).get(key(destination));
      if (!destination || cost == null) return { ok: false, reason: 'That hex is not reachable this turn.' };
      return { ok: true, actor, destination, cost };
    }
    if (kind === 'deploy') {
      const why = actor ? deployReason(g, actor, action.index) : 'Carrier not found';
      if (why) return { ok: false, reason: why };
      const destination = deployTargets(g, actor, action.index).find(p => p.c === c && p.r === r);
      if (!destination) return { ok: false, reason: 'Choose an empty land hex next to the carrier.' };
      return { ok: true, actor, destination };
    }
    if (kind === 'repair' || kind === 'reinforce' || kind === 'commander') {
      const why = kind === 'repair' ? repairReason(g, actor) :
        kind === 'reinforce' ? reinforceReason(g, actor) : feintReason(g, actor);
      return why ? { ok: false, reason: why } : { ok: true, actor };
    }
    return { ok: false, reason: 'Unknown action.' };
  }
  // Optional common dispatch. Existing UI/AI direct calls remain fully compatible,
  // and every executor below independently preflights through validateAction.
  function resolveAction(g, action) {
    if (!action) return { ok: false, reason: 'Unknown action.' };
    switch (action.kind) {
      case 'attack': return attack(g, action.id, action.c, action.r);
      case 'move': return move(g, action.id, action.c, action.r);
      case 'deploy': return deploy(g, action.id, action.index, action.c, action.r);
      case 'repair': return repair(g, action.id);
      case 'reinforce': return reinforce(g, action.id);
      case 'commander': return feint(g, action.id, action.targetId);

      l = research[id] || 0;

  function applyElites(g, profile = {}) {
    // World Campaign regional battles inherit Elite upgrades, not the rest of HQ progression.
    if (g.mode === 'scenario' && !g.warCampaign) return g;
    const records = eliteProfile(profile);
    g.eliteDeployed ||= {};
    // Kept on the game so Elite Forces that join mid-operation (campaign spawns and upgrades) use the same levels.
    g.eliteLevels = Object.fromEntries(Object.entries(records).filter(([, r]) => r?.level).map(([id, r]) => [id, r.level]));
    for (const u of allUnits(g)) {
      // Only your own Elite Forces use your HQ levels; mission aces on other sides keep the level they were given.
      // World Campaign starting conversions remain Lv.1, independent of HQ upgrades.
      // Newly recruited elites receive their level directly from the HQ profile.
      if (!u.elite || u.side !== g.player || g.mode === 'campaign' ||
          (g.mode === 'scenario' && u.eliteStart)) continue;
      const old = maxHP(u),
        rec = records[u.elite];
      if (rec?.level) u.eliteLevel = rec.level;
      if (u.hp > 0) u.hp = Math.max(1, maxHP(u) - (old - u.hp));
    }
    return g;
  }
  // Refresh your commanders inside an operation, e.g. after an HQ promotion; personal units keep their damage.
  function applyRoster(g, profile = {}) {
    // Regional scenarios also need personal HQ commanders. Their scripted officers
    // remain untouched, including their authored ranks and special Elite bonuses.
    g.roster = Object.fromEntries(Object.entries(roster(profile)).map(([k, rec]) => [k, cleanOfficer(k, rec)]));
    // Newly created or HQ-refreshed battles already use the Suzaku starter
    // policy. Do not re-migrate later Cornelia recruitments on next load.
    g.britanniaStarterMigrationVersion = 1;
    for (const u of allUnits(g)) {
      if (!u.cmd || (g.mode === 'scenario' && !u.personal)) continue;
      const old = maxHP(u);
      u.cmdRank = officerOf(g, u).rank;
      if (u.hp > 0) u.hp = Math.max(1, maxHP(u) - (old - u.hp));
    }
    return g;
  }
  // Load HQ research into the player's side: frame bonuses keep each unit's damage, city defenses follow.
  function applyTech(g, research = {}) {
    if (g.mode === 'scenario') return g;
    g.tech ||= {};
    g.tech[g.player] = Object.fromEntries(
      Object.entries(normalizeResearch(research))
        .filter(([id, l]) => TECH_NODES[id] && Number.isInteger(l) && l > 0)
        .map(([id, l]) => [id, Math.min(l, TECH_NODES[id].max)]),
    );
    for (const u of allUnits(g)) {
      if (u.side !== g.player) continue;
      const old = maxHP(u);
      u.hpTech = hullTech(g, u);
      if (u.hp > 0) u.hp = Math.max(1, maxHP(u) - (old - u.hp));
    }
    g.stations.forEach(s => fortify(g, s));
    return g;
  }
  // Fortification research raises the defenses of cities its owner holds.
  function fortify(g, s) {
    const bonus = techValue(g, s.owner, 'cities.fort'),
      old = s.fortBonus || 0;
    if (bonus === old) return;
    s.maxShield += bonus - old;
    s.shield = clamp(s.shield + Math.max(0, bonus - old), 0, s.maxShield);
    s.fortBonus = bonus;
  }

  // ======== Hex geometry: odd-r offset rows; world maps wrap east to west ========
  const DIRS = [
    [1, 0],
    [-1, 0],
    [0, 1],
    [0, -1],
    [1, -1],
    [-1, 1],
  ];
  const clamp = (n, a, b) => Math.max(a, Math.min(b, n));
  const axial = p => ({ q: p.c - Math.floor(p.r / 2), r: p.r });
  function hexDistance(a, b) {
    a = axial(a);
    b = axial(b);
    return (Math.abs(a.q - b.q) + Math.abs(a.r - b.r) + Math.abs(a.q + a.r - b.q - b.r)) / 2;
  }
  // Pass g for wrapped maps: the shortest way round the world counts.
  function distance(a, b, g = null) {
    const d = hexDistance(a, b);
    if (!g?.wrap) return d;
    const W = g.cols;
    return Math.min(d, hexDistance(a, { c: b.c + W, r: b.r }), hexDistance(a, { c: b.c - W, r: b.r }));
  }
  const dist = (g, a, b) => distance(a, b, g);
  const key = p => p.c + ',' + p.r;
  function tile(g, c, r) {
    if (r < 0 || r >= g.rows) return null;
    if (g.wrap) c = ((c % g.cols) + g.cols) % g.cols;
    else if (c < 0 || c >= g.cols) return null;
    return g.tiles[r * g.cols + c] || null;
  }
  function adjacent(g, p) {
    const a = axial(p);
    return DIRS.map(d => {
      const r = a.r + d[1],
        c = a.q + d[0] + Math.floor(r / 2);
      return tile(g, c, r);
    }).filter(Boolean);
  }
  // Every tile within n hexes of p (including p).
  function within(g, p, n) {
    const out = [],
      a = axial(p);
    for (let dq = -n; dq <= n; dq++)
      for (let dr = Math.max(-n, -dq - n); dr <= Math.min(n, -dq + n); dr++) {
        const r = a.r + dr,
          c = a.q + dq + Math.floor(r / 2),
          t = tile(g, c, r);
        if (t) out.push(t);
      }
    return out;
  }
  // Occupancy indexes: rebuilt when the unit list grows or a stale entry is found; move() keeps them current.
  const unitIndex = new WeakMap(),
    cityIndex = new WeakMap();
  function indexUnits(g) {
    const m = new Map();
    for (const u of g.units) if (u.hp > 0) m.set(key(u), u);
    const idx = { len: g.units.length, map: m };
    unitIndex.set(g.units, idx);
    return idx;
  }
  function unitAt(g, p) {
    let idx = unitIndex.get(g.units);
    if (!idx || idx.len !== g.units.length) idx = indexUnits(g);
    const k = key(p);
    let u = idx.map.get(k);
    if (u && (u.hp <= 0 || u.c !== p.c || u.r !== p.r)) u = indexUnits(g).map.get(k);
    return u || null;
  }
  function reindex(g, u, from) {
    const idx = unitIndex.get(g.units);
    if (!idx) return;
    if (from && idx.map.get(key(from)) === u) idx.map.delete(key(from));
    if (u.hp > 0) idx.map.set(key(u), u);
  }
  function stationAt(g, p) {
    let idx = cityIndex.get(g.stations);
    if (!idx || idx.len !== g.stations.length) {
      idx = { len: g.stations.length, map: new Map(g.stations.map(s => [key(s), s])) };
      cityIndex.set(g.stations, idx);
    }
    return idx.map.get(key(p)) || null;
  }

  function migrateCoastalTerrain(g) {

    for (const [c, r] of CASPIAN_LAND_FIXES) {
      if ('.w'.includes(WORLD_ROWS[r]?.[c])) continue;
      const t = g.tiles[r * g.cols + c];

  function persianGulfPoliticalBank(t) {
    const lon = WORLD.lon0 + WORLD.dlon * (t.c + 0.5 * (t.r & 1));
    const lat = WORLD.lat0 - WORLD.dlat * t.r;
    const axis = [[47,30.4],[49,29.1],[51,27.8],[53.5,26.7],[56,25.7],[58,25.2]];
    if (lon < 47 || lon > 58 || lat < 23 || lat > 32) return 0;
    for (let i = 1; i < axis.length; i++) {

      const a = axis[i - 1], b = axis[i];
      const middle = a[1] + (lon - a[0]) * (b[1] - a[1]) / (b[0] - a[0]);
      return Math.abs(lat - middle) <= 3.25 ? (lat < middle ? -1 : 1) : 0;
    }
    return 0;
  }
  function crossesWorldPoliticalStrait(g, a, b) {
    if (!a || !b || g.cols !== WORLD.cols || g.rows !== WORLD.rows) return false;
    // The Soya/La Perouse Strait separates Russian Sakhalin from Area 11.
    if (isSakhalinWorldHex(a) !== isSakhalinWorldHex(b) &&
        (isSakhalinWorldHex(a) || isSakhalinWorldHex(b))) return true;
    // Gibraltar (87,22) is on Iberia; coastal tiles (86,23) and (87,23)
    // are on Morocco. The coarse hex grid makes them adjacent across water.
    // Treat that contact as a strait for province/starting ownership only:
    // military movement and naval passage remain governed by terrain.
    const gibraltar = t => t.c === 87 && t.r === 22;
    const morocco = t => t.r === 23 && (t.c === 86 || t.c === 87);
    if ((gibraltar(a) && morocco(b)) || (gibraltar(b) && morocco(a))) return true;
    const aa = persianGulfPoliticalBank(a), bb = persianGulfPoliticalBank(b);
    return aa !== 0 && bb !== 0 && aa !== bb;
  }
  function assignCityProvinces(g) {
    // City influence follows same-faction playable LAND, not a direct line
    // through oceans. This also repairs detached provinces in older saves.
    const components = new Int32Array(g.tiles.length);
    components.fill(-1);
    let nextComponent = 0;
    for (let i = 0; i < g.tiles.length; i++) {
      const source = g.tiles[i];
      if (isSea(source) || !source.owner || components[i] >= 0) continue;
      const queue = [source], owner = source.owner;
      components[i] = nextComponent;
      for (let head = 0; head < queue.length; head++)
        for (const n of adjacent(g, queue[head])) {
          const k = n.r * g.cols + n.c;
          if (isSea(n) || n.owner !== owner || components[k] >= 0 ||
              crossesWorldPoliticalStrait(g, queue[head], n)) continue;
          components[k] = nextComponent;
          queue.push(n);
        }
      nextComponent++;
    }
    const byOwner = new Map(), byComponent = new Map();
    const cityById = new Map(g.stations.map(s => [s.id, s]));
    for (const s of g.stations) {
      if (!byOwner.has(s.owner)) byOwner.set(s.owner, []);
      byOwner.get(s.owner).push(s);
      const component = components[s.r * g.cols + s.c];
      if (component < 0) continue;
      if (!byComponent.has(component)) byComponent.set(component, []);
      byComponent.get(component).push(s);
    }
    for (const t of g.tiles) {
      if (isSea(t) || !t.owner) {
        delete t.provinceCity;
        continue;
      }
      const component = components[t.r * g.cols + t.c];
      const local = byComponent.get(component);
      const cities = local?.length ? local : (byOwner.get(t.owner) || []);
      const previous = cityById.get(t.provinceCity);
      // Capture retains the fixed province unless it points across water to
      // another landmass which has its own city of the same owner.
      if (previous && previous.owner === t.owner &&
          (!local?.length || components[previous.r * g.cols + previous.c] === component))
        continue;
      let nearestCity = null, nearestDistance = Infinity;
      for (const s of cities) {
        const d = dist(g, s, t);
        if (d < nearestDistance || (d === nearestDistance &&
            (nearestCity === null || s.id < nearestCity.id))) {
          nearestCity = s;
          nearestDistance = d;
        }
      }
      if (nearestCity) t.provinceCity = nearestCity.id;
      else delete t.provinceCity;
    }
  }

  const baseTerrain = (c, r) => TERRAIN_CODES[WORLD_ROWS[r]?.[c]] || 'sea';
  function packSave(g) {
    // Regional scenarios have their own map, so their terrain is packed with the other tile fields
    // instead of as differences from the world map.
    const regional = g.mode === 'scenario';
    if (g.mode === 'campaign' || (!regional && (g.cols !== WORLD.cols || g.rows !== WORLD.rows)) ||
        g.tiles?.length !== g.cols * g.rows)
      return g;
    const { tiles, units, ...rest } = g;
    const fields = [...new Set(tiles.flatMap(t => Object.keys(t)))]
      .filter(k => k !== 'c' && k !== 'r' && (regional || k !== 'terrain'));
    const runs = {};
    for (const f of fields) {
      const out = (runs[f] = []);
      let value, has, n = 0;
      for (const t of tiles) {
        const h = f in t;
        if (n && h === has && t[f] === value) n++;
        else {
          if (n) out.push(has ? [value, n] : [n]);
          value = t[f];
          has = h;
          n = 1;
        }
      }
      if (n) out.push(has ? [value, n] : [n]);
    }
    return {
      ...rest,
      units: units.filter(u => u.hp > 0),
      packedTiles: {
        v: SAVE_PACK,
        terrain: regional ? [] : tiles.flatMap((t, i) => (t.terrain !== baseTerrain(t.c, t.r) ? [[i, t.terrain]] : [])),
        fields: runs,
      },
    };
  }
  function unpackSave(p) {
    if (!p?.packedTiles) return p;
    const { packedTiles: pk, ...g } = p;
    if (pk.v !== SAVE_PACK || !(g.cols > 0 && g.rows > 0)) return null;
    const n = g.cols * g.rows,
      tiles = new Array(n);
    for (let i = 0; i < n; i++) {
      const c = i % g.cols,
        r = (i / g.cols) | 0;
      tiles[i] = { c, r, terrain: g.mode === 'scenario' ? 'sea' : baseTerrain(c, r) };
    }
    for (const [i, terrain] of pk.terrain || []) if (tiles[i]) tiles[i].terrain = terrain;
    for (const [f, list] of Object.entries(pk.fields || {})) {
      let i = 0;
      for (const run of list) {
        const count = run.length === 1 ? run[0] : run[1];
        if (run.length === 2) for (let k = i; k < Math.min(n, i + count); k++) tiles[k][f] = run[0];
        i += count;
      }
    }
    g.tiles = tiles;
    return g;
  }
  function migrateSave(g) {
    g = unpackSave(g);
    if (!g || g.game !== 'knightmare' || !Array.isArray(g.units)) return null;
    // A restored regional battle keeps its existing city layout, even when
    // loaded on the first turn after starting-income city pruning was added.
    if (g.mode === 'scenario' && g.scenario && !g.scenario.cityPruningVersion)
      g.scenario.cityPruningVersion = 'legacy-save';
    // Existing operations migrate commanders, cargo, scenario allegiances
    // and purchased officer progression without losing saved battle state.
    for(const records of [g.officers,g.roster])if(records)
      for(const [oldId,newId] of Object.entries(EU_ROSTER_REPLACEMENTS))if(records[oldId]){
        records[newId] ||= records[oldId];delete records[oldId];
      }
    // Existing battle snapshots migrate independently of the persistent profile.
    if (g.commissionRebalanceVersion !== 1) {
      rebalanceSavedCommissions(g.officers);
      rebalanceSavedCommissions(g.roster);
      g.commissionRebalanceVersion = 1;
    }
    for(const u of g.units.flatMap(v=>[v,...(v.cargo||[])]))
      if(u.cmd&&EU_ROSTER_REPLACEMENTS[u.cmd])u.cmd=EU_ROSTER_REPLACEMENTS[u.cmd];
    for(const pool of Object.values(g.scenario?.commanderSides||{}))if(Array.isArray(pool))
      for(let i=0;i<pool.length;i++)pool[i]=EU_ROSTER_REPLACEMENTS[pool[i]]||pool[i];
    // Saved battles keep their own snapshot of the HQ roster and personal unit
    // assignments. Apply the Britannian starter swap once, independently of the
    // profile migration; leave scripted (non-personal) Cornelia units untouched.
    // The version flag allows players to recruit Cornelia normally afterwards.
    if (g.britanniaStarterMigrationVersion !== 1) {
      const personal = g.units.flatMap(u => [u, ...(u.cargo || [])])
        .filter(u => u.personal);
      const legacyAssignments = personal.filter(u => u.cmd === 'cornelia');
      if (g.roster?.cornelia || legacyAssignments.length) {
        g.roster ||= {};
        g.roster.suzaku ||= defaultOfficer('suzaku', true);
        delete g.roster.cornelia; // Do not transfer investment or grant a refund.
        let suzakuDeployed = personal.some(u => u.cmd === 'suzaku' && u.hp > 0);
        for (const u of legacyAssignments) {
          if (!suzakuDeployed && u.hp > 0) {
            u.cmd = 'suzaku';
            u.cmdRank = g.roster.suzaku.rank;
            suzakuDeployed = true;
          } else {
            // Suzaku cannot be personally deployed twice in the same battle.
            u.cmd = null;
            u.cmdRank = 0;
            u.personal = false;
            if (u.hp > 0) u.hp = Math.min(u.hp, maxHP(u));
          }
        }
      }
      g.britanniaStarterMigrationVersion = 1;
    }
    const regional = g.mode === 'scenario';
    // Regional scenarios carry their own map; the world-map checks and repairs below do not apply to them.
    if (regional && (!g.scenario?.id || !(g.cols > 0) || g.tiles?.length !== g.cols * g.rows)) return null;
    // An existing battle keeps its actual city tiers and fortress positions.
    // Apply the new bidirectional balancing only when starting a fresh scenario.
    if (regional) g.cityStandardMigrated = true;
    // Existing saves are in progress, not new starts. Do not retroactively
    // double their already-spent Research on the next income calculation.
    if (regional) g.scenario.startingResearchDoubled = true;
    // Old Stonehenge saves had a different map and an extra playable power.
    if (regional && scenarioMeta(g).saveVersion && g.scenario.version !== scenarioMeta(g).saveVersion) return null;
    // One-time migration: preserve city HP fraction, city tiers and all active wars.
    // Do this before legacy regional funding logic can recalculate city subsidies.
    if(g.cityStandardVersion!==CITY_STANDARD_VERSION){
      g.cityStandardMigrated=true;
      standardizeCities(g,true);
    }
    // Retire legacy Division of Japan recurring city subsidies. Starting
    // reserves and captured infrastructure are already in the saved state.
    if(regional&&g.scenario.id==='division-japan'){
      g.scenario.funding={};
      g.scenario.economyLocalized=false;
    }
    // Existing saved regional battles also migrate their original funding
    // into cities. Do this once; already-localized saves are unchanged.
    if (regional && g.scenario.id !== 'division-japan')
      localizeScenarioIncome(g);
    // The high-resolution conquest rebuild cannot safely load saves from the old 100 × 42 world.
    if (g.mode !== 'campaign' && !regional && (g.cols !== WORLD.cols || g.rows !== WORLD.rows || g.tiles?.length !== WORLD.cols * WORLD.rows)) return null;
    if (g.rulesVersion !== RULES_VERSION) return null;
    // Optional scenario-owned, one-time terrain migration. The scenario asset
    // has already loaded before resumeSavedGame calls migrateSave, so this
    // updates the actual saved board rather than starting a replacement war.
    // Other scenarios and their save versions remain entirely unchanged.
    if (regional) root.Knightmare?.scenarios?.byId?.[g.scenario.id]?.migrateSavedTerrain?.(g);
    // Legacy saves tied HP to factory tiers. Pin a distinct city tier and
    // retain existing battle damage while dropping old factory HP bonuses.
    // Revisit regional saves with an existing cityTier as well: earlier loads
    // may already have inflated an HQ to the true-capital defense tier.
    for (const s of g.stations || []) {
      if (s.cityTier == null) {
        const startingTier = g.mode === 'conquest'
          ? CITY_DATA.find(row => row[0] === s.name)?.[4] : null;
        s.cityTier = regional ? cityTierOf(s) : s.fort ? 4 :
          Math.max(1, Math.min(3, Math.trunc(Number(startingTier ?? s.tier) || 1)));
      } else if (!regional) continue;
      const damage = Math.max(0, (s.maxShield || 0) - (s.shield || 0));
      s.maxShield = cityDefenseFor(g, s) + (s.fortBonus || 0);
      s.shield = Math.max(0, s.maxShield - damage);
    }
    // Preserve saves containing retired scenario-only Elite carrier-battleships.
    const legacyCarriers = {
      elite_europe_eu_carrier: 'eu_carrier',
      elite_europe_eb_carrier: 'carrier_battleship',
      elite_japan_britannia_carrier: 'carrier_battleship',
      elite_japan_eu_carrier: 'eu_carrier',
      elite_japan_cf_carrier: 'cf_carrier',
      elite_japan_bk_carrier: 'bk_carrier',
    };
    for (const u of g.units.flatMap(v => [v, ...(v.cargo || [])])) {
      if (!legacyCarriers[u.type]) continue;
      u.type = legacyCarriers[u.type];
      delete u.elite;
      delete u.eliteLevel;
      delete u.eliteStart;
      u.stack = 1;
      u.hp = Math.min(u.hp, maxHP(u));
    }
    if (g.scenario?.eliteRosters) for (const side of Object.keys(g.scenario.eliteRosters))
      g.scenario.eliteRosters[side] = (g.scenario.eliteRosters[side] || [])
        .filter(id => !!ELITE_FORCES[id]);
    if (g.eliteDeployed) for (const id of Object.keys(g.eliteDeployed))
      if (/(?:japan|europe)_.*_carrier/.test(id)) delete g.eliteDeployed[id];
    if (!g.units.every(u => TYPES[u.type])) return null;
    // Existing saves keep their positions, commanders, action state and damage.
    // Retire accidental cross-era Chinese/UFN equipment and Euro-Britannian
    // borrowed heavy armor (including embarked cargo), preserving damage and orders.
    for (const u of g.units.flatMap(v => [v, ...(v.cargo || [])])) {
      if (u.side !== 'cf' && u.side !== 'eb') continue;
      const correct = eraUnitType(g, u.type, u.side);
      if (correct === u.type) continue;
      const oldMax = Math.max(1, maxHP(u)), fraction = u.hp / oldMax;
      u.type = correct;
      u.elite = ELITE_TYPE_TO_ID[correct] || null;
      if (u.elite) {
        u.eliteLevel = Math.max(1, u.eliteLevel || 1);
        u.stack = 1;
      } else {
        u.eliteLevel = 0;
        delete u.eliteStart;
      }
      u.hpTech = hullTech(g, u);
      u.hp = fraction > 0 ? Math.max(1, Math.min(maxHP(u), Math.round(maxHP(u) * fraction))) : 0;
    }
    if (g.scenario?.eliteRosters) for (const side of Object.keys(g.scenario.eliteRosters))
      g.scenario.eliteRosters[side] = g.scenario.eliteRosters[side]
        .filter(id => eliteEligible(g, side, id));
    if (g.mode === 'conquest') {
      const cfg = CONQUESTS[g.era || 'world'];
      if (!cfg) return null;
      g.era = cfg.id;
      g.majors = [...cfg.majors];
      g.teams = { ...cfg.teams };
      g.factions = cfg.factions ? JSON.parse(JSON.stringify(cfg.factions)) : null;
      if (!cfg.selectable.includes(g.player)) return null;
      // Reconcile old saves to one warhead per pool. In 2018 the Britannian
      // partners count together, with a human-owned warhead/project kept first.
      // Cancelled projects refund their ORIGINAL owner's treasury, not the ally's.
      g.arsenal ||= {};
      const visitedPools = new Set();
      for (const side of g.majors) {
        const members = fleijaPoolSides(g, side);
        const poolKey = members.join('/');
        if (visitedPools.has(poolKey)) continue;
        visitedPools.add(poolKey);
        const entries = [];
        for (const member of members) {
          if ((g.arsenal[member] || 0) > 0)
            entries.push({ member, kind: 'stored' });
          for (const s of g.stations.filter(s => s.project?.side === member))
            entries.push({ member, kind: 'project', station: s });
        }
        entries.sort((a, b) =>
          Number(b.member === g.player) - Number(a.member === g.player) ||
          Number(b.kind === 'stored') - Number(a.kind === 'stored') ||
          (a.station?.project.ready || 0) - (b.station?.project.ready || 0) ||
          g.majors.indexOf(a.member) - g.majors.indexOf(b.member));
        for (const member of members) g.arsenal[member] = 0;
        for (const [i, entry] of entries.entries()) {
          if (i < FLEIJA.max) {
            if (entry.kind === 'stored') g.arsenal[entry.member] = 1;
          } else if (entry.kind === 'project') {
            entry.station.project = null;
            const treasury = funds(g, entry.member);
            if (treasury) for (const [resource, amount] of Object.entries(FLEIJA.cost))
              treasury[resource] = (treasury[resource] || 0) + amount;
          }
        }
      }
    }
    if (g.mode !== 'campaign' && !regional) {
      migrateCoastalTerrain(g);
      attachUnclaimedNewZealand(g);
      repairCapeYorkOwnership(g);
      repairSakhalinTerritory(g);
      repairLegacyRegionalZones(g);
      // Older saves have no province IDs. Bind their existing painted land to
      // its closest still-controlled city, without resetting conquest progress.
      assignCityProvinces(g);
    }
    g.eliteDeployed ||= {};
    for (const [which, records] of [['officers', g.officers], ['roster', g.roster]])
      if (records) for (const [k, rec] of Object.entries(records))
        if (COMMANDERS[k]) {
          const scenarioOfficers = which === 'officers' && eliteScenario(g)
            ? allUnits(g).filter(u => u.cmd === k && !u.personal && foe(g, u.side, g.player)) : [];
          // Older loads could cap the officer record but leave its unit's rank intact.
          // Recover that rank, including officers currently aboard carriers.
          const record = scenarioOfficers.length ? { ...rec,
            rank: Math.max(rec?.rank || 0, ...scenarioOfficers.map(u => u.cmdRank || 0)) } : rec;
          // Saves from before the rank merge may hold the two retired ranks
          // above Supreme Commander; cleanOfficer caps them at HQ_RANK_CAP.
          records[k] = cleanOfficer(k, record);
        }
    if (g.mode !== 'campaign') automationState(g);
    for (const u of g.units.flatMap(v => [v, ...(v.cargo || [])])) {
      if (u.cmd && COMMANDERS[u.cmd] && (eliteScenario(g) || u.cmdRank > HQ_RANK_CAP)) {
        // The retired ranks above Supreme Commander scaled frames to 208% and 216%.
        const retired = { 11: 2.08, 12: 2.16 }[u.cmdRank],
          old = retired ? maxHP({ ...u, cmdRank: HQ_RANK_CAP }) * retired / RANK_HP[HQ_RANK_CAP] : maxHP(u);
        u.cmdRank = Math.min(HQ_RANK_CAP, officerOf(g, u)?.rank ?? u.cmdRank);
        if (u.hp > 0) u.hp = Math.max(1, Math.round(maxHP(u) - (old - u.hp)));
      }
      const elite = u.elite || ELITE_TYPE_TO_ID[u.type];
      if (elite) {
        u.elite = elite;
        u.eliteLevel ||= 1;
        // Elite Forces are single frames; older saves let rival elites carry up to three.
        if (u.stack > 1 && u.hp > 0) {
          const share = u.hp / maxHP(u);
          u.stack = 1;
          u.hp = Math.max(1, Math.round(share * maxHP(u)));
        }
        // Scripted campaign elites and preplaced Elite Conquest/scenario starts never spend the player's
        // recruit allowance; in scenarios only the player's own deployments count.
        const preplaced = (eliteConquest(g) || eliteScenario(g)) && u.eliteStart;
        if (!eliteCampaign(g) && !preplaced && (!regional || u.side === g.player))
          g.eliteDeployed[elite] = true;
      }
      // Older saves left carried units at the hex where they boarded.
      for (const c of u.cargo || []) {
        c.c = u.c;
        c.r = u.r;
      }
    }
    return g;
  }

  function newUnit(g, type, side, c, r, stack = 1, cmd = null, ready = true) {
    cmd=EU_ROSTER_REPLACEMENTS[cmd]||cmd;
    // Normalize ALL scenario-scripted and recruited Chinese/UFN frames at birth;
    // faction allegiance and the frame's combat class remain unchanged.
    type = eraUnitType(g, type, side);
    const u = {
      id: g.nextId++,
      type,
      side,
      c,
      r,
      stack: ELITE_TYPE_TO_ID[type] ? 1 : stack, // Elite Forces are always single frames, for every side.
      hp: 0,
      morale: 0,
      moved: !ready,
      attacked: !ready,
      cmd,
      kills: 0,
      chain: 0,
      xp: 0,
      feintCD: 0,
      held: false,
      guardReady: false,
      entrenched: false,
      movedDistance: 0,
      moveActionTurn: null,
      moveActionsThisTurn: 0,
      skillReposition: 0,
      withdrawMove: false,
      elite: ELITE_TYPE_TO_ID[type] || null,
      eliteLevel: ELITE_TYPE_TO_ID[type]
        ? (g.warCampaign && side === g.player ? (g.eliteLevels?.[ELITE_TYPE_TO_ID[type]] || 1) : 1)
        : 0,
      eliteMoveAfterKill: false,
    };
    u.hpTech = hullTech(g, u);
    u.hp = maxHP(u);
    g.units.push(u);
    // Scenario-created Leila immediately protects nearby formations from low morale.
    if (cmd === 'leila') applyLeilaMoraleFloor(g);
    // Challenge Pass: count newly created frames, including ships, amphibious and Elite Forces.
    if (g.mode === 'conquest' && side === g.player && g.challengePassRun?.version === 1) {
      g.challengePassRun.produced += u.stack;
      if (u.elite) g.challengePassRun.eliteProduced = true;
    }
    if(g.warReport)recordWarEvent(g,'unitProduced',side,{unitId:u.id,unitType:type,stack:u.stack});
    return u;
  }
  function movement(g, u) {
    const t = TYPES[u.type],
      f = fx(u),
      mobilityStars = u.cmd ? effectiveRating(g, u, 'mobility') : 0;
    let n = t.move + (eliteFx(u).move || 0);
    // Conquest and Scenarios grant +2 movement to ground and amphibious units.
    // Story Mode uses base movement. Warships never receive this bonus.
    if (g && (g.mode === 'conquest' || g.mode === 'scenario') && t.naval !== 'ship')
      n += UNIVERSAL_MOVE_BONUS;
    // HQ mobility: Infantry +4, land Armor +2, Artillery +2.
    // Float's two levels still give amphibious Knightmares up to +4 on land.
    // Clamp old in-progress games that saved Float levels III/IV.
    // Warship and amphibious sea movement remain unchanged.
    if (g && t.naval !== 'ship') {
      const floatLevel = Math.min(2, techLevel(g, u.side, 'armor.float'));
      if (t.naval === 'amphibious') n += floatLevel * 2;
      else if (t.branch === 'Armor') n += floatLevel;
      else if (t.branch === 'Infantry') n += techLevel(g, u.side, 'infantry.nav');
      else if (t.branch === 'Artillery') n += techLevel(g, u.side, 'artillery.traction');
    }
    // Every commander Mobility star grants +1 movement (1★ = +1 through 6★ = +6).
    n += mobilityStars;
    if(u.expJammedActive)n=Math.max(1,n-2);
    if (u.expYukiyaElectronicActive) n = Math.max(1, n - 2);
    n += f.move || 0;
    if (u.expImperialCommand) n += 2;
    // Banked first-strike movement adds to a later full movement action.
    // An immediate postattack reposition instead spends from the bank itself.
    if (f.firstStrike && !u.expKallenReposition && !u.skillReposition &&
        t.naval !== 'ship') n += u.expKallenMoveBank || 0;
    if (u.expKallenSlowActive) n = Math.max(1, n - 2);
    if (u.expArkFleetSlow) n = Math.max(1, n - (u.expArkFleetSlow.penalty || 2));
    if(u.expMiguelSupply)n=Math.max(0,n-2);
    // Mao's mind-reading pulse lasts through the commander's next activation.
    if(u.expMaoMoveActive)n=Math.max(1,n-2);
    return n;
  }
  function terrainStepCost(g, t) {
    const regional = g?.mode === 'scenario' ? scenarioMeta(g).terrainCosts?.[t.terrain] : null;
    return regional || TERRAIN[t.terrain]?.cost || 1;
  }
  function terrainText(g, t) {
    const info = TERRAIN[t.terrain], cost = terrainStepCost(g, t);
    return info.desc.replace(/Movement cost \d+\./, 'Movement cost ' + cost + '.');
  }
  function terrainCost(g, u, t) {
    if (isSea(t)) return 1;
    const type = TYPES[u.type];
    if (fx(u).ignoreTerrain || u.expRetrofit || type.float || eliteFx(u).float) return 1;
    return terrainStepCost(g, t);
  }
  // Iraq–Iran only: crossing a Tigris, Euphrates or Karun river edge consumes
  // one extra movement point. Marked bridge-city approaches have no surcharge.
  // All other scenarios and the global Conquest rules remain unchanged.
  function riverCrossingCost(g, a, b, u = null) {
    if (g?.mode !== 'scenario' || !g.scenario.riverBarriers ||
        isSea(a) || isSea(b)) return 0;
    if (u && (fx(u).ignoreTerrain || u.expRetrofit || TYPES[u.type]?.float ||
        eliteFx(u).float || TYPES[u.type]?.naval)) return 0;
    const i = a.r * g.cols + a.c, j = b.r * g.cols + b.c;
    return g.scenario.riverBarriers?.[i < j ? i + ':' + j : j + ':' + i] ? 1 : 0;
  }
  function canCapture(u) {
    return TYPES[u.type].naval !== 'ship';
  }
  function isReady(g, u) {
    return !g.over && g.phase === u.side && u.hp > 0 && !u.expRoloFrozen && !u.expCharlesMemoryRewrite && u.morale > -3;
  }
  // Count actual movement actions, not hex distance. This also covers abilities
  // that reset 'moved', and works on old saves where these fields are missing.
  function isHeavyChainMover(u) {
    const cls = TYPES[u.type]?.cls;
    return cls === 'heavy' || cls === 'super';
  }
  function heavyMoveLimitReached(g, u) {
    return isHeavyChainMover(u) && u.moveActionTurn === g.turn && (u.moveActionsThisTurn || 0) >= 2;
  }
  function markMoveAction(g, u) {
    u.moveActionsThisTurn = (u.moveActionTurn === g.turn ? (u.moveActionsThisTurn || 0) : 0) + 1;
    u.moveActionTurn = g.turn;
  }
  // Movement: a Dijkstra search. Land units may embark onto an adjacent sea hex (which ends the move); embarked
  // units sail up to their sea movement and may land on a coast hex (which ends the move).
  function reachable(g, u) {
    const found = new Map();
    const ef = eliteFx(u);
    // Brain Raid restores the full normal movement budget after a shot,
    // but only while Akito remains isolated; this never restores an attack.
    const akitoFollowUp = !!u.expAkitoMoveReady && akitoIsolated(g, u);
    if (!isReady(g, u) || u.expFarneseSlaughterTurn === g.turn || u.expFernandoOverload || u.expDisabledActive || u.expPinnedActive || (u.expJammedActive && u.attacked) || (u.moved && !u.expOscarReposition) || heavyMoveLimitReached(g, u) || (u.attacked && !ef.moveAfterAttack && !u.eliteMoveAfterKill && !u.skillReposition && !u.withdrawMove && !akitoFollowUp && !u.expOscarReposition)) return found;
    const t = TYPES[u.type],
      start = tile(g, u.c, u.r),
      fromSea = isSea(start),
      ship = t.naval === 'ship',
      air = !!t.air,
      amphibious = t.naval === 'amphibious',
      landMove = movement(g, u),
      seaMv = amphibious ? amphibiousSea(g, u) : 0,
      budget = u.expOscarReposition ? 3 : u.expKallenReposition ? (u.expKallenMoveBank || 0)
        : u.skillReposition ? Math.min(landMove, u.skillReposition)
        : ship ? landMove : amphibious ? landMove * seaMv : fromSea ? seaMove(g, u) : landMove,
      boards = !u.expOscarReposition && !u.skillReposition && !u.expKallenReposition && !t.naval && u.deployedTurn !== g.turn,
      stateKey = p => key(p),
      costs = new Map([[stateKey(start), 0]]),
      queue = [{ p: start, cost: 0, order: 0 }];
    let nextOrder = 1;
    const before = (a, b) => a.cost < b.cost || (a.cost === b.cost && a.order < b.order);
    const push = entry => {
      entry.order = nextOrder++;
      let i = queue.length;
      queue.push(entry);
      while (i > 0) {
        const parent = (i - 1) >> 1;
        if (!before(entry, queue[parent])) break;
        queue[i] = queue[parent];
        i = parent;
      }
      queue[i] = entry;
    };
    const pop = () => {
      const first = queue[0], last = queue.pop();
      if (queue.length) {
        let i = 0;
        for (;;) {
          const left = i * 2 + 1, right = left + 1;
          if (left >= queue.length) break;
          const child = right < queue.length && before(queue[right], queue[left]) ? right : left;
          if (!before(queue[child], last)) break;
          queue[i] = queue[child];
          i = child;
        }
        queue[i] = last;
      }
      return first;
    };
    const rivers = g.mode === 'scenario' && !!g.scenario?.riverBarriers; // decided once, not per step
    while (queue.length) {
      const { p, cost } = pop();
      if (cost > costs.get(stateKey(p))) continue;
      for (const n of adjacent(g, p)) {
        // Airships fly over impassable peaks and land alike; other warships keep to the water.
        if ((!air && TERRAIN[n.terrain]?.blocked) || (ship && !air && !navigable(n)) ||
            (u.skillReposition && !t.naval && isSea(n) && !fromSea) ||
            ((u.expMargaretRedeploy||u.expShadowMove||u.expOscarReposition)&&isSea(n)!==fromSea)) continue;
        const occ = unitAt(g, n),
          st = stationAt(g, n);
        if (ship && st) continue;
        if (occ && occ.side !== u.side) continue;
        if (occ && boards && canBoard(g, occ)) {
          if (cost < budget) found.set(key(n), budget);
          continue;
        }
        if (st && foe(g, st.owner, u.side) && !(canCapture(u) && st.shield <= 0)) continue;
        const cross = !t.naval && isSea(n) !== fromSea;
        // Ordinary sea transports must begin their movement beside the landing hex.
        // They may sail OR land this turn, never sail across water and land in one action.
        if (cross && fromSea && cost > 0) continue;
        const step = isSea(n) ? 1 : terrainCost(g, u, n) + (rivers ? riverCrossingCost(g, p, n, u) : 0);
        const nc = (u.skillReposition || u.expOscarReposition)
          ? cost + step
          : cross
          ? budget
          : ship
            ? cost + 1
            : amphibious
              ? cost + (isSea(n) ? landMove : (terrainCost(g, u, n) + (rivers ? riverCrossingCost(g, p, n, u) : 0)) * seaMv)
              : cost + step;
        const sk = stateKey(n);
        if (nc > budget || (cross && cost >= budget) || nc >= (costs.get(sk) ?? Infinity)) continue;
        costs.set(sk, nc);
        if (!cross) push({ p: n, cost: nc });
        if (!occ && key(n) !== key(start)) found.set(key(n), Math.min(found.get(key(n)) ?? Infinity, nc));
      }
    }
    return found;
  }
  // Fire Control II adds one hex of range to all Artillery.
  function rangeOf(g, u) {
    const t = TYPES[u.type];
    return {
      min: t.min,
      max:
        t.max +
        (eliteFx(u).range || 0) +
        (g && t.branch === 'Artillery' && !t.naval && techLevel(g, u.side, 'artillery.fire') >= 2 && (u.movedDistance || 0) === 0 ? 1 : 0) +
        (g && t.branch === 'Artillery' && spotted(g, u) ? 1 : 0) + (u.expNavalRelay ? 1 : 0) +
        (u.cmd === 'norland' && u.expNorlandTitan ? 1 : 0), // Foulbout Titan Mode extends all eligible weapons by one hex.
    };
  }
  function inRange(a, p, g) {
    const { min, max } = rangeOf(g, a),
      d = dist(g, a, p);
    return d >= min && d <= max;
  }
  // Only these can attack an airship or return its fire: Heavy and Super-heavy armor, Rocket and
  // Siege artillery (regular or Elite of those classes), warships, other airships and Float-equipped
  // units. Fortress batteries, which are not units, can also fire on them.
  const AIR_DEFENSE_CLASSES = new Set(['heavy', 'super', 'rocket', 'siege']);
  function canHitAir(u) {
    const t = TYPES[u.type];
    return t.naval === 'ship' || AIR_DEFENSE_CLASSES.has(t.cls) || !!t.float || !!eliteFx(u).float;
  }
  // The Wang Hu's Radiant Wave Barrier blocks ranged attacks from farther
  // than one hex, regardless of attacker's unit class or the save version.
  function rangedImmuneTo(g, target, source) {
    return !!(target && target.hp > 0 && source &&
      fx(target).rangedImmunity && dist(g, target, source) > 1);
  }
  function mindReadBlocksAttack(g,target,attacker) {
    return !!(target && target.hp>0 && attacker && attacker.hp>0 &&
      fx(target).mindReadAdjacent && foe(g,target.side,attacker.side) &&
      dist(g,target,attacker)===1);
  }
  function hostileTarget(g, u, p) {
    const target = unitAt(g, p),
      st = stationAt(g, p);
    if (target) return foe(g, target.side, u.side) &&
      !mindReadBlocksAttack(g,target,u) &&
      !rangedImmuneTo(g, target, u) &&
      (!isAir(target) || canHitAir(u));
    return !!st && foe(g, st.owner, u.side) && st.shield > 0;
  }
  // Whether a unit still has any order besides holding position: firing, moving, repairing, reinforcing or an action.
  function hasOrders(g, u) {
    if (!u || !isReady(g, u)) return false;
    if (!u.attacked && targets(g, u).length) return true;
    if (!u.moved && !u.goto && reachable(g, u).size) return true;
    if (!repairReason(g, u) || !reinforceReason(g, u)) return true;
    if (u.cargo?.some((c, i) => !deployReason(g, u, i))) return true;
    return !!COMMANDERS[u.cmd]?.action && !feintReason(g, u);
  }
  // Carrier cargo may land on an unoccupied enemy city only after its defenses reach zero.
  function deployTargetsAt(g, p, side, carriesTroops = false) {
    return adjacent(g, p).filter(t =>
      !isSea(t) && !TERRAIN[t.terrain]?.blocked && !unitAt(g, t) &&
      (!stationAt(g, t) || !foe(g, stationAt(g, t).owner, side) ||
        (carriesTroops && stationAt(g, t).shield <= 0)),
    );
  }
  const deployTargets = (g, ship, i = 0) => deployTargetsAt(g, ship, ship.side, !!ship.cargo?.[i]);
  function deployReason(g, ship, i) {
    const u = ship?.cargo?.[i];
    if (!u) return 'No unit aboard';
    return (
      turnReason(g, ship.side) ||
      (u.boardedTurn === g.turn ? 'Troops cannot deploy until the turn after boarding' : null) ||
      (!deployTargets(g, ship, i).length ? 'No empty land hex next to the carrier' : null)
    );
  }
  function deploy(g, shipId, i, c, r) {
    const check = validateAction(g, { kind: 'deploy', id: shipId, index: i, c, r });
    if (!check.ok) return check;
    const ship = check.actor, t = check.destination;
    const [u] = ship.cargo.splice(i, 1);
    u.c = t.c;
    u.r = t.r;
    u.moved = u.attacked = false;
    u.deployedTurn = u.launched = g.turn;
    u.expLandedTurn=g.turn;
    if(u.cmd==='sawasaki'){u.expBeachheadStart=g.turn;u.expBeachheadUntil=g.turn+2;}
    g.units.push(u);
    // Only city captures transfer territory; ordinary landings cannot repaint it.
    const { captured, annexed } = captureCity(g, u, true);
    const seized = seizeDeposit(g, u, t);
    log(g, `${COMMANDERS[u.cmd]?.short || TYPES[u.type].short} launches from the ${TYPES[ship.type].air ? TYPES[ship.type].name : 'Carrier-Battleship'}.`, u.side);
    checkVictory(g);
    return { ok: true, unit: u, to: { c: t.c, r: t.r }, captured, annexed, seized };
  }
  // Embarked units cannot fire.
  function targets(g, u) {
    if (atSea(g, u)) return [];
    const { min, max } = rangeOf(g, u); // once, not again for every hex in reach
    return within(g, u, max).filter(p => {
      const d = dist(g, u, p);
      return d >= min && d <= max && hostileTarget(g, u, p);
    });
  }
  // targets() restricted to `pool` (hexes to test), in the same order. For a unit standing on a hex with nothing
  // in its range disc this skips scanning the disc; not for wrapped maps, whose disc order differs.
  function targetsAmong(g, u, pool) {
    if (atSea(g, u)) return [];
    const { min, max } = rangeOf(g, u);
    return pool
      .filter(p => {
        const d = dist(g, u, p);
        return d >= min && d <= max && hostileTarget(g, u, p);
      })
      .sort((a, b) => a.c - (a.r >> 1) - (b.c - (b.r >> 1)) || a.r - b.r);
  }
  function claim(g, p, owner) {
    if (g.mode !== 'campaign') {
      // Capturing a city transfers its entire fixed province, not the nearby
      // provinces and not just the seven hexes immediately around the city.
      let changed = 0;
      for (const t of g.tiles)
        if (!isSea(t) && t.provinceCity === p.id) {
          setTileOwner(g, t, owner);
          changed++;
        }
      // Custom maps and old standalone tests may not have city provinces.
      if (!changed) setTileOwner(g, tile(g, p.c, p.r), owner);
      return;
    }
    // Campaign missions use small tactical maps, not the conquest province map.
    for (const t of [tile(g, p.c, p.r), ...adjacent(g, p)])
      if (!isSea(t) && !TERRAIN[t.terrain]?.blocked) setTileOwner(g, t, owner);
  }
  // A port is seized by entering its water hex, independently of its city and territory.
  function capturePort(g, u) {
    const s = portAtHex(g, u);
    // Airships cannot seize ports, just as they cannot seize cities or mines.
    if (!s || isAir(u) || !foe(g, s.portOwner || s.owner, u.side)) return null;
    const previousOwner=s.portOwner||s.owner;
    s.portOwner=u.side;
    recordWarEvent(g,'portCaptured',u.side,{cityId:s.id,city:s.name,previousOwner,unitId:u.id});
    log(g, `${COMMANDERS[u.cmd]?.short || TYPES[u.type].short} captures the port of ${s.name}.`, u.side);
    return s.name;
  }
  // City capture changes provinces, but does not automatically transfer ports.
  function captureCity(g, u, navalLanding = false) {
    const s = stationAt(g, u);
    if (u.hp <= 0 || !s || !foe(g, s.owner, u.side) ||
      !canCapture(u) || s.shield > 0)
      return { captured: null, annexed: null };
    const loser = s.owner;
    if(fx(u).taxSeizure){funds(g,u.side).credits+=Math.min(150,3*cityOutput(g,s).credits);s.expTaxUntil=g.turn+2;}
    delete s.expSupplyBlocked;delete s.expDecree;delete s.expShieldRegenBlocked;delete s.expYoshidaBreak;
    s.owner = u.side;
    recordWarEvent(g,'cityCaptured',u.side,{cityId:s.id,city:s.name,previousOwner:loser,
      unitId:u.id,commander:u.cmd||null,naval:!!navalLanding,group:warGroup(g,u)});
    s.shield = 0;
    s.capturedTurn = g.turn;
    dropProject(g, s, 'captured');
    dropEliminator(g, s, 'captured');
    u.morale = 1;
    funds(g, u.side).credits += 40;
    fortify(g, s);
    // Update the new owner's maximum fortification, but do not grant instant defenses.
    // Shields recover normally on the new owner's subsequent turn.
    s.shield = 0;
    claim(g, s, u.side);
    if (fx(u).captureHeal) u.hp = Math.min(maxHP(u), u.hp + Math.round(maxHP(u) * fx(u).captureHeal));
    // Fernando must personally capture the city. This passive triggers once
    // per faction turn, regardless of how many captures he chains.
    if (u.cmd === 'fernando' && fx(u).popularUprising && u.expPopularUprisingTurn !== g.turn) {
      u.expPopularUprisingTurn = g.turn;
      for (const ally of g.units) {
        if (ally.hp <= 0 || ally.side !== u.side || atSea(g, ally) || dist(g, u, ally) > 2) continue;
        ally.hp = Math.min(maxHP(ally), ally.hp + Math.round(maxHP(ally) * 0.20));
        ally.moved = false;
        ally.skillReposition = 0;
        // Keep attack usage and heavy movement limits intact, but allow
        // a full legal movement action even if the ally already attacked.
        ally.withdrawMove = !!ally.attacked;
      }
      log(g, 'Popular Uprising restores nearby allied formations.', u.side);
    }
    if(adjacent(g,s).some(t=>isSea(t)))
      for(const sw of g.units)if(sw.hp>0&&sw.side===u.side&&sw.cmd==='sawasaki'&&sw.expBeachheadStart!=null&&
        g.turn<=sw.expBeachheadUntil&&(sw.id===u.id||(TYPES[u.type].naval==='amphibious'&&dist(g,sw,s)<=2)))
        sw.expBeachheadUntil=Math.min(sw.expBeachheadStart+3,sw.expBeachheadUntil+1);

    if(u.cmd==='nonette' && u.expCaptureAttackTurn!==g.turn){u.attacked=false;u.expCaptureAttackTurn=g.turn;}
    if(u.cmd==='sugiyama'&&u.expCaptureMoveTurn!==g.turn){grantReposition(u,3);u.expCaptureMoveTurn=g.turn;}
    const kaguyas=g.units.filter(v=>v.hp>0&&v.side===u.side&&v.cmd==='kaguya'&&dist(g,v,s)<=3);
    if(kaguyas.length)for(const v of g.units)if(v.hp>0&&v.side===u.side&&!isAir(v)&&kaguyas.some(k=>dist(g,k,v)<=3))v.hp=Math.min(maxHP(v),v.hp+Math.round(maxHP(v)*.1));
    if (fx(u).specialOps && (s.gunReady || 0) > g.turn) s.gunReady--;
    log(g, `${COMMANDERS[u.cmd]?.short || TYPES[u.type].short} captures ${s.name}.`, u.side);
    hooks.capture?.(g, s, u, loser);
    honorAction(g, u, 'cities');
    if (s.capitalOf && s.capitalOf === loser) {
      honorAction(g, u, 'capitals');
      award(g, u.side, 'star', `${s.name} captured`, u, String(s.id));
    }
    // Preserve expeditionary forces, including carrier cargo, after their last city is lost.
    const expeditionSide = scenarioMeta(g).expeditionSide;
    const expeditionAfloat = loser === expeditionSide &&
      (g.units.some(v => v.side === loser && v.hp > 0) ||
       g.units.some(v => (v.cargo||[]).some(u => u.side === loser && u.hp > 0)));
    const annexed = g.mode !== 'campaign' && alive(g, loser) &&
      !g.stations.some(c => c.owner === loser) && !expeditionAfloat
      ? surrender(g, loser, u.side, s) : null;
    return { captured: s.name, annexed };
  }
  function move(g, id, c, r) {
    return collectAbilityVisuals(g, () => resolveMove(g,id,c,r));
  }
  function resolveMove(g, id, c, r) {
    const check = validateAction(g, { kind: 'move', id, c, r });
    if (!check.ok) return check;
    const u = check.actor, dest = check.destination, moveCost = check.cost;
    const from = { c: u.c, r: u.r },
      fromSea = isSea(tile(g, u.c, u.r)),
      t = TYPES[u.type],
      carrier = unitAt(g, dest);
    // Consume actual movement cost, not the whole bonus pool. Earned points
    // that cannot be used this turn remain available on Kallen's next turn.
    const oscarFreeMove = !!u.expOscarReposition;
    const originalActions = oscarFreeMove ? {
      moved:u.moved,attacked:u.attacked,skillReposition:u.skillReposition,
      withdrawMove:u.withdrawMove,eliteMoveAfterKill:u.eliteMoveAfterKill,
      akitoMoveReady:u.expAkitoMoveReady
    } : null;
    const kallenReposition = !oscarFreeMove && !!fx(u).firstStrike && !!u.expKallenReposition;
    const kallenNormalBonus = !oscarFreeMove && !!fx(u).firstStrike && !kallenReposition &&
      !u.skillReposition && !u.withdrawMove && !t.naval && !fromSea;
    const kallenBaseMove = kallenNormalBonus ? movement(g, u) - (u.expKallenMoveBank || 0) : 0;
    const kallenSpent = kallenReposition ? moveCost :
      kallenNormalBonus ? Math.max(0, moveCost - kallenBaseMove) : 0;
    // A legal move consumes the single post-attack Brain Raid movement grant.
    if (!oscarFreeMove) delete u.expAkitoMoveReady;
    // Boarding: the Knightmare goes aboard (off the map) and its action ends.
    if (carrier && carrier !== u && carrier.side === u.side && canBoard(g, carrier)) {
      markMoveAction(g, u);
      g.units.splice(g.units.indexOf(u), 1);
      u.c = dest.c;
      u.r = dest.r;
      u.moved = u.attacked = true;
      u.held = false;
      u.guardReady = false;
      u.skillReposition = 0;
      u.withdrawMove = false;
      if (kallenSpent) u.expKallenMoveBank = Math.max(0, (u.expKallenMoveBank || 0) - kallenSpent);
      delete u.expKallenReposition;
      u.movedDistance = (u.movedDistance || 0) + dist(g, from, dest);
      u.lastTurnMoved = true;
      u.boardedTurn = g.turn;
      (carrier.cargo ||= []).push(u);
      log(g, `${COMMANDERS[u.cmd]?.short || TYPES[u.type].short} boards the ${TYPES[carrier.type].air ? TYPES[carrier.type].name : 'Carrier-Battleship'}.`, u.side);
      return { ok: true, from, to: { c: dest.c, r: dest.r }, loaded: carrier.id };
    }
    markMoveAction(g, u);
    u.c = dest.c;
    u.r = dest.r;
    // Units aboard a Carrier-Battleship travel with it (rallies and other distance rules read their position).
    for (const c of u.cargo || []) {
      c.c = dest.c;
      c.r = dest.r;
    }
    u.moved = true;
    u.expRomeroMoved=true; // Once physically moved, Stand Fast stays off this turn.
    // Amphibious units gain one full extra movement and attack after landing each turn.
    // A per-turn stamp prevents unlimited actions by hopping across the coastline.
    const freshLanding = t.naval === 'amphibious' && fromSea && !isSea(dest) && u.landingRefreshTurn !== g.turn;
    if (freshLanding) {
      u.landingRefreshTurn = g.turn;
      u.moved = false;
      u.attacked = false;
      u.chain = 0;
    }
    u.held = false;
    u.guardReady = false;
    u.movedDistance = freshLanding ? 0 : (u.movedDistance || 0) + dist(g, from, dest);
    if(fromSea&&!isSea(dest)){u.expLandedTurn=g.turn;if(u.cmd==='sawasaki'){u.expBeachheadStart=g.turn;u.expBeachheadUntil=g.turn+2;}}
    u.lastTurnMoved = true;
    u.skillReposition = 0;
    u.withdrawMove = false;
    u.eliteMoveAfterKill = false;
    if (kallenSpent) u.expKallenMoveBank = Math.max(0, (u.expKallenMoveBank || 0) - kallenSpent);
    // Multiple earned movement points may be spent in successive legal
    // repositions, subject to the existing heavy-frame action limit.
    if (kallenReposition && u.expKallenMoveBank > 0 &&
        !heavyMoveLimitReached(g, u) && !freshLanding) {
      u.moved = false;
      u.skillReposition = u.expKallenMoveBank;
    } else if (!oscarFreeMove) {
      delete u.expKallenReposition;
    }
    // Oscar's 3-point reposition is free and independent of normal
    // movement/attack. Restore whatever action availability existed at
    // the moment of the reposition, including another earned reposition.
    if (oscarFreeMove) {
      u.moved=originalActions.moved;
      u.attacked=originalActions.attacked;
      u.skillReposition=originalActions.skillReposition;
      u.withdrawMove=originalActions.withdrawMove;
      u.eliteMoveAfterKill=originalActions.eliteMoveAfterKill;
      if(originalActions.akitoMoveReady)u.expAkitoMoveReady=originalActions.akitoMoveReady;
      delete u.expOscarReposition;
    }
    // The Last Promise banks one full extra move without bypassing heavy caps.
    if(u.expToruExtraMove>0&&u.hp>0&&!heavyMoveLimitReached(g,u)){
      u.expToruExtraMove--;u.moved=false;
    }
    // An unspent original move followed by Orderly Withdrawal earns one more
    // full movement action, respecting the existing heavy-frame two-move cap.
    if(u.expFarneseExtraMove > 0 && u.hp > 0 && !heavyMoveLimitReached(g,u) &&
       u.expFarneseSlaughterTurn !== g.turn && !oscarFreeMove) {
      u.expFarneseExtraMove--;
      u.moved=false;
      u.withdrawMove=true;
    }
    // Tactical redeployment preserves the original normal movement action.
    if(u.expMargaretRedeploy){u.moved=!!u.expMargaretRedeploy.wasMoved;delete u.expMargaretRedeploy;}
    if(u.expShadowMove){u.moved=true;delete u.expShadowMove;}
    reindex(g,u,from);
    // The Steady floor takes effect as soon as Leila or an ally enters her radius.
    if (u.cmd === 'leila' || u.cmd === 'fernando') applyLeilaMoraleFloor(g);
    else if (u.morale < 0 && moraleFloor(g, u) >= 0) u.morale = 0;
    const hostileReaction = newcomerOverwatch(g,u);
    if(u.hp<=0){checkVictory(g);return {ok:true,from,to:{c:dest.c,r:dest.r},destroyed:true,hostileReaction};}
    // Moving through enemy or neutral territory never changes its map color.
    const { captured, annexed } = captureCity(g, u, t.naval === 'amphibious' || (!t.naval && fromSea));
    const portCaptured = capturePort(g, u);
    const seized = seizeDeposit(g, u, dest);
    checkVictory(g);
    return { ok: true, from, to: { c: dest.c, r: dest.r }, captured, annexed, portCaptured, seized, hostileReaction };
  }
  // `last` is the city whose capture left the loser with none; anything it still held passes to the conqueror.
  function surrender(g, loser, winner, last) {
    (g.fallen ||= {})[loser] = { by: winner, turn: g.turn, city: last.name };
    recordWarEvent(g,'surrender',winner,{loser,city:last.name});
    // Clear strategic defenses while any remaining cities still identify their original owner.
    annexStrategic(g, loser);
    let cities = 0,
      units = 0;
    for (const s of g.stations) {
      // Defeated powers cannot keep ports, but third-party harbour control survives.
      if (s.portAt && s.portOwner === loser) s.portOwner = winner;
      if (s.owner === loser) {
        s.owner = winner;
        s.shield = Math.round(s.maxShield * 0.5);
        s.producedTurn = s.portProducedTurn = g.turn;
        fortify(g, s);
        cities++;
      }
    }
    // Any land formerly attached to an already destroyed city now belongs to
    // a surviving city of the conqueror, so later captures can still transfer it.
    const winnerCities = new Set(g.stations.filter(s => s.owner === winner).map(s => s.id));
    for (const t of g.tiles) if (t.owner === loser) {
      setTileOwner(g, t, winner);
      if (!winnerCities.has(t.provinceCity)) delete t.provinceCity;
    }
    if (g.mode !== 'campaign') assignCityProvinces(g);
    for (const v of g.units)
      if (v.hp > 0 && v.side === loser) {
        v.hp = 0;
        kill(g, v, null, true); // also sinks the Knightmares aboard a Carrier-Battleship
        units++;
      }
    const e = funds(g, loser), w = funds(g, winner);
    if (g.mode === 'conquest') {
      // Conquest capitulation transfers territory and mines, never stockpiles.
      for (const resource of ['credits', 'industry', 'science', 'sakuradite'])
        e[resource] = 0;
    } else {
      // Preserve the existing independent-scenario surrender rules.
      w.credits += Math.round(e.credits / 2);
      w.industry += Math.round(e.industry / 2);
      e.credits = e.industry = 0;
    }
    annexDeposits(g, loser, winner);
    log(
      g,
      `${last.name}, the last city of the ${FACTIONS[loser].name}, has fallen. It surrenders to the ${FACTIONS[winner].name}: ${units} units disbanded.`,
      winner,
    );
    return { loser, winner, cities, units, city: last.name };
  }
  // Command auras: every commander lifts adjacent friends by 8%; some reach farther or further.

  function auraBonus(g, u) {
    let bonus = 0;
    for (const a of g.units) {
      if (a.hp <= 0 || a.side !== u.side || a.id === u.id) continue;
      // Leila's new skill replaces both her old +12% damage aura and the
      // default commander +8% damage aura; she provides defense instead.
      if (a.cmd && a.cmd !== 'leila' && a.cmd !== 'zero' && a.cmd !== 'lelouch' && a.cmd !== 'schneizel' && a.cmd !== 'bismarck' && a.cmd !== 'ayano' && a.cmd !== 'yukiya' && a.cmd !== 'rakshata' && a.cmd !== 'xianglin' && a.cmd !== 'ohgi' && a.cmd !== 'smilas' && !a.auraDisrupted && (a.cmd !== 'ashley' || !atSea(g, a)) && dist(g, a, u) <= auraRange(a)) bonus = Math.max(bonus, fx(a).aura?.value || 0.08);
      const ea = eliteFx(a).aura;
      if (ea && dist(g, a, u) <= ea.range && (!ea.branches || ea.branches.includes(TYPES[u.type].branch)))
        bonus = Math.max(bonus, ea.value);
    }
    return bonus;
  }
  // Permanent frame/commander strength shared by the dock and combat. Situational
  // modifiers (morale, health, terrain, targets and auras) remain in power().
  function intrinsicAttack(g, u, counter = false) {
    const t = TYPES[u.type], f = fx(u), strike = !counter || !f.attackOnly,
      attackGeneric = { Infantry: 'raider', Armor: 'armored_assault', Artillery: 'accuracy' }[t.branch];
    let attack = (t.attack + officerBaseAttack(g, u)) * eliteScale(u).attack * (1 + 0.45 * (u.stack - 1));
    if (f.dmg && strike) attack *= 1 + f.dmg;
    if (f.dmgBranch?.[t.branch] && strike) attack *= 1 + f.dmgBranch[t.branch];
    return attack * officerAttack(g, u) * (1 + 0.06 * genericLevel(g, u, attackGeneric));
  }
  // Shared geography test: coastal land, not open water or inland terrain.
  function coastalLand(g, u) {
    const here = tile(g, u.c, u.r);
    return !!here && !isSea(here) &&
      (isCoast(here) || adjacent(g, u).some(n => isSea(n)));
  }
  // Suppressive Fire reduces retaliation to the initiating Infantry/Armor unit,
  // but never removes the defender's ability to counter.
  function suppressiveCounterFactor(g, u) {
    const t = TYPES[u.type];
    return !t.naval && (t.branch === 'Infantry' || t.branch === 'Armor')
      ? 1 - techValue(g, u.side, `${branchOf(u.type)}.suppressive`) : 1;
  }
  // Manfredi must be immediately adjacent to the enemy. Other friendly
  // formations must also be adjacent; the attacking unit may count as one.
  function manfrediSuperiority(g,attacker,target){
    if(!attacker||!target||attacker.side===target.side)return 0;
    const masters=g.units.filter(v=>v.hp>0&&v.cmd==='manfredi'&&v.side===attacker.side&&!atSea(g,v)&&dist(g,v,target)===1);
    for(const master of masters){
      const others=g.units.filter(v=>v.hp>0&&v.side===master.side&&v.id!==master.id&&!atSea(g,v)&&dist(g,v,target)===1);
      if(others.length>=2)return 2;
      if(others.length>=1)return 1;
    }
    return 0;
  }
  function power(g, u, target, st, counter = false, direct = true) {
    const t = TYPES[u.type],
      victim = target ? TYPES[target.type] : null,
      f = fx(u),
      ef = eliteFx(u),
      strike = !counter || !f.attackOnly;
    let attack = intrinsicAttack(g, u, counter) * (1 + 0.07 * Math.min(5, u.xp));
    // Preview Cornelia's opening morale step before the volley without mutating game state.
    // attack() applies the actual increase only after a legal attack is confirmed.
    const morale = !counter && f.firstAttackMorale && u.firstAttackMoraleTurn !== g.turn && u.morale > -3
      ? Math.min(1, u.morale + f.firstAttackMorale)
      : u.morale;
    attack *= morale >= 1 ? 1.25 : morale === -1 ? 0.75 : morale === -2 ? 0.5 : morale <= -3 ? 0 : 1;
    // Foulbout Titan Mode empowers both initiating attacks and defensive counterfire.
    if (u.cmd === 'norland' && u.expNorlandTitan) attack *= 1.30;
    if (u.despairWeakened) attack *= 1 - u.despairWeakened.value;
    if(u.expPinnedWeakenActive)attack*=1-u.expPinnedWeakenActive;
    if(u.expChengWeaken?.turns>0)attack*=1-u.expChengWeaken.value;
    if (!counter && u.expSmilasVanguardCharge) attack *= 1.50;
    if (!counter && u.expToruPromiseDamage) attack *= 1.30;
    if(!counter&&u.expSmilasSuppression)attack*=1-u.expSmilasSuppression;
    // Yukiya's electronic disruption weakens all attacks (including counterfire)
    // only during the victim's next activation.
    if (u.expYukiyaElectronicActive) attack *= 0.75;
    if(!counter&&u.forcedOffensive)attack*=1.35;
    const gritSkill = { Infantry: 'bayonet_charge', Armor: 'tide_of_iron', Artillery: 'artillery_barrage' }[t.branch];
    const missingHpPenalty = 0.6 * (1 - clamp(u.hp / maxHP(u), 0, 1));
    attack *= 1 - missingHpPenalty * (1 - 0.2 * genericLevel(g, u, gritSkill)) * (u.hp/maxHP(u)<.7&&g.units.some(v=>v.cmd&&v.hp>0&&v.side===u.side&&fx(v).stabilize&&dist(g,u,v)<=2)?.5:1);
    attack *= 1 + (u.expInformationWarfare?0:auraBonus(g,u));
    if(u.expInformationWarfare)attack*=.8;
    attack *= newcomerAttack(g,u,target,counter);
    // Ohgi's half-health leadership supports outgoing attacks and counterfire.
    if (ohgiSelflessLeadership(g, u)) attack *= 1.35;
    // Star of Madrid rewards wounded allies' initiating attacks and counterfire.
    if (u.hp < maxHP(u) * 0.50 && starOfMadridNear(g, u)) attack *= 1.25;
    // Xianglin's coordination only strengthens direct initiating attacks.
    // Previews and resolved volleys share this calculation; splash damage
    // and defensive counterattacks never receive the +30% bonus.
    if (!counter && direct && target && xianglinExploitWeakness(g, u, target))
      attack *= 1.30;
    // Checkmate boosts direct attacks and counterattacks against its marked
    // formation, without granting Schneizel a passive aura.
    if (target?.expCheckmate?.side === u.side) attack *= 1.25;
    // Portmans hunt transports and warships; the Portman II also fights harder in the water (Aquatic Combat).
    // Nothing deals bonus damage to airships: anti-ship and coastal bonuses skip them.
    const seaShip = target && isShip(target) && !isAir(target);
    if (t.naval === 'amphibious' && target && (atSea(g, target) || seaShip)) attack *= 1.25;
    if (t.naval === 'amphibious' && seaShip) attack *= 1 + techValue(g, u.side, 'amphibious.torpedoes');
    if (t.naval === 'amphibious' && target && !isAir(target) && coastalLand(g, target))
      attack *= 1 + techValue(g, u.side, 'amphibious.clearance');
    if (t.naval === 'amphibious' && counter && coastalLand(g, u))
      attack *= 1 + techValue(g, u.side, 'amphibious.counterfire');
    if (t.aquatic && isSea(tile(g, u.c, u.r))) attack *= 1 + t.aquatic;
    if (t.naval === 'ship') {
      attack *= 1 + techValue(g, u.side, 'naval.gunnery');
      if (counter) attack *= 1 + techValue(g, u.side, 'naval.firecontrol');
      if (target && victim.naval === 'amphibious')
        attack *= 1 + techValue(g, u.side, 'naval.antiamphib');
      if (target && atSea(g, target))
        attack *= 1 + techValue(g, u.side, 'naval.interdiction');
    }
    // Rapid Launch Systems: the first attack on the turn a Knightmare launched from a carrier (needs a level-3 port).
    if (!counter && u.launched === g.turn && hasPort3(g, u.side)) attack *= 1 + techValue(g, u.side, 'naval.launch');
    // Commander signature abilities (attacker side).
    if (f.opening && !counter && !u.moved) attack *= 1 + f.opening;
    // Only the first DIRECT attack against each individual enemy formation.
    // Splash damage, counterattacks and city-only bombardment do not trigger it.
    if (!counter && direct && target && f.firstStrike &&
        !(u.expKallenFirstTargets || []).includes(target.id))
      attack *= 1 + f.firstStrike;
    if (!counter && f.liveOnAttack && u.hp / maxHP(u) < 0.5) attack *= 1 + f.liveOnAttack;
    if (counter && f.counter) attack *= 1 + f.counter;
    // Ayano's swordplay counters only adjacent formations.
    if (counter && f.swordmaster && target && dist(g, u, target) === 1) attack *= 1.5;
    if (counter) attack *= 1 + 0.05 * genericLevel(g, u, 'crossfire');
    attack *= skillAttack(g, u, counter) * commanderAttack(g, u, target, counter);
    if(!counter){
      if(u.expMaskFalls?.attackReady)attack*=1.50;
      else if(u.expDeathEmperor)attack*=1.50;
    }
    if (ef.dmg && strike) attack *= 1 + ef.dmg;
    if (counter && ef.counter) attack *= 1 + ef.counter;
    if (ef.vsArmor && victim?.branch === 'Armor' && strike) attack *= 1 + ef.vsArmor;
    // Cornelia's Royal Guard Raid applies near any city, whether held or hostile.
    if (ef.cityAttack && g.stations.some(s => dist(g, s, u) <= 1) && strike)
      attack *= 1 + ef.cityAttack;
    // Amphibious Elite abilities: the Lv.5 values replace, rather than stack with, Lv.3.
    const amphibiousAtSea = isSea(tile(g, u.c, u.r));
    const coastalTarget = target ? (isAir(target) ? null : target) : st;
    if (ef.vsNaval && target &&
        (seaShip || (victim?.naval === 'amphibious' && isSea(tile(g, target.c, target.r)))))
      attack *= 1 + ef.vsNaval;
    if (counter && amphibiousAtSea && ef.seaCounter) attack *= 1 + ef.seaCounter;
    if (!counter && ef.coastalAttack && coastalTarget && coastalLand(g, coastalTarget))
      attack *= 1 + ef.coastalAttack;
    if (!counter && ef.landingStrike && amphibiousAtSea && coastalTarget &&
        !isSea(tile(g, coastalTarget.c, coastalTarget.r)))
      attack *= 1 + ef.landingStrike;
    // Shui Gun-Ru's coastal aura is positional and applies only on attacks against coastal land.
    // Multiple Shui supporters do not stack.
    if (!counter && coastalTarget && coastalLand(g, coastalTarget) &&
        g.units.some(v => v.hp > 0 && v.id !== u.id && v.side === u.side &&
          dist(g, v, u) === 1 && eliteFx(v).coastalSupport))
      attack *= 1.10;
    if (f.artist && target)
      attack *=
        1 +
        f.artist *
          Math.min(3, g.units.filter(v => v.hp > 0 && v.side === u.side && v.id !== u.id && dist(g, v, target) === 1).length);
    if (t.boarding && (target ? victim.branch === 'Armor' : !!st)) attack *= 1.50;
    // HQ research: branch weapons and class counters.
    attack *= 1 + unitTech(g, u, 'guns');
    // Armored Exploitation replaces the old kill-chain cap bonus. It applies
    // only to ordinary land Armor attacking a living, weakened enemy frame.
    // It does not boost counterfire, city defenses, elites or amphibious units.
    if (!counter && !t.elite && !t.naval && t.branch === 'Armor' &&
        target && target.hp > 0 && target.hp < maxHP(target) * 0.5)
      attack *= 1 + techValue(g, u.side, 'armor.assault');
    if (!t.naval && t.branch === 'Infantry' && victim && !victim.naval) {
      if (victim.branch === 'Armor') attack *= 1 + techValue(g, u.side, 'infantry.antiarmor');
      if (victim.branch === 'Artillery') attack *= 1 + techValue(g, u.side, 'infantry.antiartillery');
    }
    if (!t.naval && t.branch === 'Armor' && counter)
      attack *= 1 + techValue(g, u.side, 'armor.reactive');
    if (!t.naval && t.branch === 'Armor' && !counter && u.expAssaultMomentumTurn === g.turn)
      attack *= 1 + techValue(g, u.side, 'armor.momentum');
    if (!t.naval && t.branch === 'Artillery' && victim && !victim.naval && victim.branch === 'Artillery')
      attack *= 1 + techValue(g, u.side, 'artillery.counterammo');
    if (!t.naval && t.branch === 'Artillery' && target && dist(g, u, target) >= 2)
      attack *= 1 + techValue(g, u.side, 'artillery.longrange');
    if (t.naval === 'amphibious' && !counter && u.expLandedTurn === g.turn && u.landingAssaultUsedTurn !== g.turn)
      attack *= 1 + techValue(g, u.side, 'amphibious.landing');
    if (!t.naval && t.branch === 'Armor' && victim?.branch === 'Infantry') attack *= 1 + techValue(g, u.side, 'armor.secondary');
    if (!t.naval && t.branch === 'Artillery' && victim?.branch === 'Infantry') attack *= 1 + techValue(g, u.side, 'artillery.shells');
    const friends = (side, at, test) =>
      g.units.some(v => v.hp > 0 && v.side === side && v.id !== u.id && dist(g, v, at) === 1 && test(TYPES[v.type]));
    // Lance Formation and Factsphere Fire Control.
    if (!t.naval && t.branch === 'Armor' && techLevel(g, u.side, 'armor.formation') >= 1 && friends(u.side, u, v => v.branch === 'Armor' && !v.naval))
      attack *= 1 + techValue(g, u.side, 'armor.formation');
    if (
      !t.naval &&
      t.branch === 'Artillery' &&
      target &&
      techLevel(g, u.side, 'artillery.fire') >= 1 &&
      friends(u.side, target, v => v.branch !== 'Artillery')
    )
      attack *= 1.2;
    const pen = Math.min(.95,armorPenetration(g,u,target)
      + (!t.naval && t.branch === 'Armor' && victim && !victim.naval && victim.branch === 'Armor'
        ? techValue(g, u.side, 'armor.precision') : 0)
      + (!t.naval && t.branch === 'Artillery' && victim && !victim.naval && victim.branch === 'Armor'
        ? techValue(g, u.side, 'artillery.shellfuses') : 0)
      +(u.expCalibration?.15:0)
      +joePenetration(g,u,target)
      +(!counter&&target&&u.expPrecisionTarget?.turn===g.turn&&u.expPrecisionTarget.id===target.id?.25:0)
      +(u.elite&&!TYPES[u.type].naval&&g.units.some(v=>v.cmd&&v.hp>0&&v.side===u.side&&fx(v).elitePenSupport&&dist(g,v,u)<=2)?0.15:0));
    const formationArmor =
      target &&
      !victim.naval &&
      victim.branch === 'Armor' &&
      techLevel(g, target.side, 'armor.formation') >= 2 &&
      g.units.some(v => v.hp > 0 && v.side === target.side && v.id !== target.id && !TYPES[v.type].naval && TYPES[v.type].branch === 'Armor' && dist(g, v, target) === 1)
        ? 5
        : 0;
    const armor = target
      ? (victim.armor + eliteScale(target).armor + unitTech(g, target, 'armor') + formationArmor)
        * (target.expAyanoSwordBreak ? 0.75 : 1)
        * (f.swordmaster && dist(g, u, target) === 1 ? 0.60 : 1)
        * (fx(target).fortressArmor && stationAt(g, target)?.owner === target.side ? 1 + fx(target).fortressArmor : 1)
      : 35;
    const vritraBypass=!counter&&f.elitePenSupport&&victim?.branch==='Armor'&&u.expVritraUsedTurn!==g.turn;
    // First anti-armor hit bypasses 20% of the entire target-defense reduction,
    // rather than adding a second penetration or cover modifier.
    const beforeDefenses=attack;
    attack *= 100 / (100 + armor * (1 - pen) * (counter||!target||manfrediSuperiority(g,u,target)<1?2:1));
    if (target) {
      const tf = fx(target),
        nearCity = g.stations.some(s => s.owner === target.side && dist(g, s, target) <= 1);
      attack *= officerDefense(g, target);
      attack *= 1 - 0.03 * genericLevel(g, target, 'fortification');
      if (!victim.naval && t.branch === 'Artillery' && victim.branch === 'Armor') attack *= 1 - techValue(g, target.side, 'armor.blaze');
      if (!victim.naval && t.cls === 'siege' && victim.branch === 'Armor') attack *= 1 - techValue(g, target.side, 'armor.bulkheads');
      if (!victim.naval && victim.branch === 'Artillery' && !t.naval && t.branch === 'Artillery')
        attack *= 1 - techValue(g, target.side, 'artillery.counterbattery');
      // Exclusive class defenses: warships and amphibious units cannot inherit
      // Infantry/Armor/Artillery research through their underlying frame class.
      if (!victim.naval && victim.branch === 'Infantry' && !t.naval && t.branch === 'Armor')
        attack *= 1 - techValue(g, target.side, 'infantry.armorcounters');
      if (!victim.naval && victim.branch === 'Infantry' && !t.naval && t.branch === 'Artillery' && t.cls === 'siege')
        attack *= 1 - techValue(g, target.side, 'infantry.antisiege');
      if (!victim.naval && victim.branch === 'Armor' && !t.naval && t.branch === 'Armor')
        attack *= 1 - techValue(g, target.side, 'armor.turrets');
      if (victim.naval === 'ship' && t.naval === 'ship')
        attack *= 1 - techValue(g, target.side, 'naval.bulkheads');
      if (victim.naval === 'ship' && t.naval === 'amphibious')
        attack *= 1 - techValue(g, target.side, 'naval.closein');
      if (victim.naval === 'amphibious' && t.naval === 'ship' && isSea(tile(g, target.c, target.r)))
        attack *= 1 - techValue(g, target.side, 'amphibious.openwater');
      const ground = tile(g, target.c, target.r);
      if (!victim.naval && victim.branch === 'Infantry' && (ground.terrain === 'urban' || !!stationAt(g, target)))
        attack *= 1 - techValue(g, target.side, 'infantry.urban');
      if (!victim.naval && victim.branch === 'Infantry' && target.entrenched)
        attack *= 1 - techValue(g, target.side, 'infantry.entrench');
      if (nearCity) attack *= 1 - techValue(g, target.side, 'cities.bunkers');
      attack *= 1 - techValue(g, target.side, 'cities.shields');
      // Factsphere Screen: Infantry shields neighbouring Artillery.
      if (
        !victim.naval &&
        victim.branch === 'Artillery' &&
        techLevel(g, target.side, 'infantry.picket') >= 1 &&
        g.units.some(
          v => v.hp > 0 && v.side === target.side && TYPES[v.type].branch === 'Infantry' && dist(g, v, target) === 1,
        )

      // Commander signature abilities (defender side).
      // -30% defense from Electrical Overload is +30% incoming combat damage.
      if (target.expFernandoOverload && foe(g, target.side, u.side)) attack *= 1.30;
      if (tf.taken) attack *= tf.taken;
      if (ohgiSelflessLeadership(g, target)) attack *= 0.75;
      if (tf.swordmaster && dist(g, u, target) === 1) attack *= 0.70;

      // Mask Falls never stacks with either of those protections.
      if(target.expMaskFalls) attack *= 0.60;
      else {
        if(target.expImperialCommand) attack *= 0.75;
        if(target.expDeathEmperor) attack *= 0.75;
      }
      // The charge remains intact during previews, then is spent by the
      // actual direct hit, counterattack or splash resolution.
      if (target.expCCProtection) attack *= 0.70;
      if (target.expFarneseWithdrawal) attack *= 0.70;
      if (target.cmd === 'norland' && target.expNorlandTitan) attack *= 0.70;
      // Non-stacking Blue Geass protection, including Leila herself and
      // friendly formations at exactly two hexes. No offensive bonus.
      if (leilaCommand(g, target)) attack *= 0.9;
      const tef = eliteFx(target);
      if (tef.taken) attack *= tef.taken;
      if (tef.seaTaken && isSea(tile(g, target.c, target.r))) attack *= tef.seaTaken;
      if (tef.navalCounterTaken && counter && t.naval) attack *= tef.navalCounterTaken;
      const protector = g.units.find(v => {

        return v.hp > 0 && v.side === target.side && v.id !== target.id && p && dist(g, v, target) <= p.range;
      });

      if (tf.counterTaken && counter) attack *= tf.counterTaken;
      if (tf.cityGuard && nearCity) attack *= tf.cityGuard;
      if (nearCity && skillNear(g, target, 'garrisonCommand', 1, false)) attack *= 0.9;
      // Oscar passively protects other allies within two hexes of himself.
      // His marked ally instead receives 20% protection even after moving away.
      attack *= oscarDefenseMultiplier(g,target);
      if (target.expKlausLoyalty?.side === target.side) attack *= 0.8;
      attack *= skillDefense(g,target,counter,direct) *
        commanderDefense(g,target,u,counter,direct) *
        newcomerDefense(g,target,u,counter,direct);
      if (isSea(ground)) attack *= TYPES[target.type].naval ? 1 : 1 + seaPenalty(g, target.side);
      // Airships fly above the terrain: no cover, and no extra damage either.
      else if (!isAir(target)) {
        const bypass=(target.expInformationWarfare?.side===u.side?.25:0)+(!counter&&u.expPrecisionTarget?.turn===g.turn&&u.expPrecisionTarget.id===target.id?.4:0)+(u.forcedOffensive?.5:0)+(target.expRadar?.side===u.side&&TYPES[u.type].branch==='Artillery'?1:0)+(fx(u).highland&&tile(g,u.c,u.r)?.terrain==='mountain'?.5:0)+(fx(u).adaptiveBreakthrough&&u.movedDistance>=3?.2:0)+(fx(u).ziAmbush&&u.movedDistance>=2&&u.expZiAmbushTurn!==g.turn?.3:0)+(f.blackSeaFox&&u.expEduardoMountainTurn!==g.turn&&tile(g,u.c,u.r)?.terrain==='mountain'?.25:0);
        attack*=1-(TERRAIN[ground.terrain]?.cover||0)*(1-Math.min(1,bypass));
      }
    }
    if(vritraBypass&&attack<beforeDefenses)attack+=.2*(beforeDefenses-attack);
    // Standing next to Smilas is always dangerous for his allies, independent
    // of whether the current incoming hit is one he redirected.
    if (target && smilasAdjacentShield(g,target)) attack *= 1.15;
    // Xingke's Perfect Counter replaces the ordinary 65% retaliation multiplier.
    if (counter) attack *= f.unrestrictedCounter || ((f.perfectCounter || romeroStandFast(g,u)) ? 1
      : !t.naval && t.branch === 'Infantry' && techLevel(g, u.side, 'infantry.picket') >= 2 ? 0.90 : 0.65);
    return Math.max(1, Math.round(attack));
  }
  function preview(g, id, c, r) {
    const a = g.units.find(u => u.id === id),
      p = tile(g, c, r);
    if (!a || !p || atSea(g, a) || !hostileTarget(g, a, p) || !inRange(a, p, g)) return null;
    const t = TYPES[a.type],
      f = fx(a),
      d = unitAt(g, p),
      s = stationAt(g, p);
    const xianglinOpening = !!d && xianglinExploitWeakness(g, a, d),
      base = power(g, a, d, s),
      // Xianglin's target-specific 30% damage and Kallen's first strike are
      // unit-only bonuses; neither may amplify fortified city shields.
      cityBase = base /
        (d && f.firstStrike && !(a.expKallenFirstTargets || []).includes(d.id) ? 1 + f.firstStrike : 1) /
        (xianglinOpening ? 1.30 : 1),
      shield = s && foe(g, s.owner, a.side) && s.shield > 0;
    // Demolition Expert boosts direct hits on any enemy stationed in a city, even if its shield is down.
    const fullUnitDmg = d ? Math.round(base * (shield ? 0.55 : 1) * (f.yukiyaDemolition && s ? 1.50 : 1)) : 0;
    const smilasPreview = d ? smilasDamageShare(g,d,fullUnitDmg) : null;
    const unitDmg = smilasPreview ? smilasPreview.primary : 0;
    // Specialized infantry, artillery and naval research can increase city shield damage.
    const ef = eliteFx(a),
      raid = (1 + (!t.naval && t.branch === 'Infantry' ? techValue(g, a.side, 'infantry.mines') : 0)) *
        (1 + (!t.naval && t.branch === 'Artillery' ? techValue(g, a.side, 'artillery.siege') : 0)) *
        (1 + (t.naval === 'ship' && s && (s.portAt || adjacent(g, s).some(v => navigable(v)))
          ? techValue(g, a.side, 'naval.coastal') : 0)) *
        (1 + (t.naval === 'amphibious' && s &&
          (s.portAt || adjacent(g, s).some(v => navigable(v)))

        (1 + (f.vsCity || 0)) * (1 + (ef.vsCity || 0)) *

    const shieldDmg = shield
      ? Math.round(
          cityBase *

            (d ? 0.8 : 1.45) *
            raid,

      : 0;
    // Asahina's Rapid Assault: a target already hit this turn by his side cannot counter, and he crits more often.
    const followUp = !!f.followUp && !!d && d.struck?.turn === g.turn && d.struck.side === a.side;

      !a.expFernandoOverloadStrike &&

      inRange(d, a, g) &&

      t.crit +
        (f.crit || 0) +
         (f.liveOnCrit && a.hp / maxHP(a) < 0.5 ? f.liveOnCrit : 0) +
        (a.expCalibration ? .25 : 0) +

        techValue(g, d.side, `${branchOf(d.type)}.critpower`)
      : 1;
    const counterDamage = (senbaReaction ? Math.round(power(g,d,a,stationAt(g,a),true)*(.5/.65))
      : counter ? power(g,d,a,stationAt(g,a),true) : 0) *
      (fx(d).unrestrictedCounter ? 1 : suppressiveCounterFactor(g, a)) * counterCritMult;
    // Preview the approximate 5% reflection; the actual volley uses HP lost
    // after damage rolls and interception, and reflects only if Xingke survives.
    const counterReflection = counterAllowed && d.hp > unitDmg && fx(d).counterReflect
      ? Math.round(Math.min(d.hp, unitDmg) * fx(d).counterReflect) : 0;
    return {

      rawUnit: fullUnitDmg,
      shieldRedirect: smilasPreview?.redirected || 0,
      shieldRedirectId: smilasPreview?.ally?.id || null,
      shield: shieldDmg,
      counter: counterDamage + counterReflection,
      counterAllowed,
      // A critical hit cannot land on an immune defending formation.
      crit: d && fx(d).critImmune ? 0 : crit,
      baseCritMult: (t.critMult || 1.55) + (f.critBonus || 0),
      critMult: (t.critMult || 1.55) + (f.critBonus || 0) +
        techValue(g, a.side, `${branchOf(a.type)}.critpower`),
      // Marirrosa's sole passive replaces class splash with full-strength hits
      // on every adjacent enemy, at any legal range (including range 1).
      splash: f.estrellaBarrage ? 1 : (t.splash || 0) + (ef.splash || 0) + (!t.naval && t.branch === 'Artillery' && (t.splash || ef.splash) ? techValue(g, a.side, 'artillery.salvo') : 0),
      armorPen: Math.min(.95,armorPenetration(g,a,d)
        + (!t.naval && t.branch === 'Armor' && d && !TYPES[d.type].naval && TYPES[d.type].branch === 'Armor'
          ? techValue(g, a.side, 'armor.precision') : 0)
        + (!t.naval && t.branch === 'Artillery' && d && !TYPES[d.type].naval && TYPES[d.type].branch === 'Armor'
          ? techValue(g, a.side, 'artillery.shellfuses') : 0)
        +(a.expCalibration?.15:0)
        +joePenetration(g,a,d)
        +(a.elite&&!TYPES[a.type].naval&&g.units.some(v=>v.cmd&&v.hp>0&&v.side===a.side&&fx(v).elitePenSupport&&dist(g,v,a)<=2)?.15:0)),
    };

  function guilfordProtect(g, victim, attacker, amount) {
    const incoming = Math.max(0, Math.round(amount || 0));
    if (!incoming || !victim?.cmd || victim.hp <= 0 || !attacker ||
        !foe(g, victim.side, attacker.side) || atSea(g, victim))
      return incoming;
    const guards = g.units.filter(v => v.hp > 0 && v.id !== victim.id &&
      v.side === victim.side && v.cmd === 'guilford' && fx(v).guilfordGuard &&
      !atSea(g,v) && dist(g,v,victim) === 1).sort((a,b) => a.id-b.id);
    if (!guards.length) return incoming;
    // A lethal hit triggers the once-per-turn complete rescue BEFORE the usual
    // 50% sharing rule. Once spent, subsequent hits still share 50% normally.
    const sacrifice = incoming >= victim.hp
      ? guards.find(v => v.expGuilfordSacrificeUsedTurn !== g.turn) : null;
    const guard = sacrifice || guards[0];
    const redirected = sacrifice ? incoming : Math.round(incoming * .5);
    const before = guard.hp;
    guard.hp = Math.max(0, guard.hp - redirected);
    if (sacrifice) {
      guard.expGuilfordSacrificeUsedTurn = g.turn;
      victim.expGuilfordSavedAttack = { turn:g.turn, attackerId:attacker.id };
      guilfordCleanse(victim);
    }
    recordAbilitySignal(g,victim,sacrifice ? 'ULTIMATE SACRIFICE' : 'GUILFORD GUARD','guard');
    // Multiple threatened commanders can name different attackers; the +30%
    // does not stack against a single attacker.
    if (guard.hp > 0) {
      const mark = guard.expGuilfordRetribution || { targets:[], turnsLeft:0 };
      if (!mark.targets.includes(attacker.id)) mark.targets.push(attacker.id);
      mark.turnsLeft = Math.max(mark.turnsLeft, g.phase === guard.side ? 2 : 1);
      guard.expGuilfordRetribution = mark;
    }
    recordWarEvent(g,'damage',attacker.side,{targetSide:guard.side,
      value:before-guard.hp,unitId:attacker.id,group:warGroup(g,attacker)});
    log(g, sacrifice
      ? 'Ultimate Sacrifice: Guilford absorbs a lethal strike and cleanses the protected commander.'
      : 'My Princess, My Justice: Guilford intercepts half the strike.', guard.side);
    // A sacrificed Guilford is a genuine combat casualty, but the rescued
    // formation receives ZERO damage even if he is killed by the full blow.
    if (guard.hp <= 0) kill(g,guard,attacker,false,null,redirected);
    return sacrifice ? 0 : incoming - redirected;
  }
  // Combat damage is distributed immediately. Guilford's protection precedes

      .sort((a,b)=>dist(g,a,victim)-dist(g,b,victim)||a.id-b.id)[0];

  function smilasDealCombatDamage(g, victim, attacker, amount, allowJohanne=true) {
    // Scripted volleys, support strikes and splash cannot bypass the same
    // barrier that prevents ordinary ranged targeting.
    if(rangedImmuneTo(g,victim,attacker)||mindReadBlocksAttack(g,victim,attacker))return 0;
    const guard=allowJohanne?johanneBodyguard(g,victim,attacker):null;
    if(guard){johanneAbsorb(g,victim,attacker,guard,amount);return 0;}
    const incoming = darltonProtect(g,victim,attacker,guilfordProtect(g,victim,attacker,amount));
    const priorHP = victim.hp;
    const split = smilasDamageShare(g,victim,incoming);
    // Intercept before applying the hit to Smilas, so a bodyguard killed
    // shielding a mortally wounded Smilas still triggers Sacrifices.
    if (split.ally && split.redirected > 0) {
      const ally = split.ally, before = ally.hp;
      ally.hp = Math.max(0,ally.hp-split.redirected);
      klausLoyaltyAfterHit(g, ally, attacker, before);
      recordWarEvent(g,'damage',attacker.side,{
        targetSide:ally.side,value:before-ally.hp,unitId:attacker.id,group:warGroup(g,attacker)});
      // The full REDIRECTED killing blow (including its 15% exposure) is
      // what Necessary Sacrifices measures, not the ally's remaining HP.
      kill(g,ally,attacker,false,null,split.redirected);
    }
    victim.hp = Math.max(0,victim.hp-split.primary);
    klausLoyaltyAfterHit(g, victim, attacker, priorHP);
    return split.primary;
  }
  // force: nothing survives (F.L.E.I.J.A.); C.C.'s Code Bearer
  // can save one lethal hit per global turn, without granting invulnerability.
  // by: the side responsible when there is no attacking unit (a city battery).
  function kill(g, v, attacker, force = false, by = null, incomingDamage = 0) {
    if (v.hp > 0) return;
    // Marianne's Awakening: Anya survives an otherwise fatal hit at full HP.
    // The per-formation stamp persists in saves and follows the game's global turns.
    // Forced kills (F.L.E.I.J.A., lost carrier cargo) remain final.
    if (!force && fx(v).marianneAwakening &&
        (v.expMarianneAwakeningTurn == null ||
         g.turn - v.expMarianneAwakeningTurn >= fx(v).marianneAwakening)) {
      v.hp = maxHP(v);
      v.expMarianneAwakeningTurn = g.turn;
      log(g, "Marianne's Awakening restores Anya to full health.", v.side);
      recordAbilitySignal(g,v,"MARIANNE'S AWAKENING",'heal');
      return;
    }
    if ((fx(v).undying && v.undyingUsedTurn !== g.turn || v.cmd==='urabe'&&!v.expFinalStandUsed&&!!attacker&&!!v.expDirectIncoming) && !force) {
      // A commander who forces a lethal Code Bearer / Final Stand activation
      // has still earned a Valor citation, despite the scripted survival.
      if (v.cmd && attacker?.cmd)
        award(g, attacker.side, 'valor', `Overcame ${COMMANDERS[v.cmd].short}'s final defense`, attacker);
      if(v.cmd==='urabe'){
        v.expFinalStandUsed=true;v.hp=1;
        log(g,'Urabe survives a lethal direct hit: Final Stand.',v.side);
        if(attacker.hp>0&&!v.expRoloFrozen&&!atSea(g,v)&&inRange(v,attacker,g)&&hostileTarget(g,v,attacker)){
          const amount=power(g,v,attacker,stationAt(g,attacker),false);
          const oldHp=attacker.hp;
          smilasDealCombatDamage(g,attacker,v,amount);
          recordWarEvent(g,'damage',v.side,{targetSide:attacker.side,value:oldHp-attacker.hp,unitId:v.id,group:warGroup(g,v)});
          if(attacker.hp<=0){attacker.expDirectIncoming=true;kill(g,attacker,v,false,null,amount);delete attacker.expDirectIncoming;}
          recordAbilityAttack(g,v,attacker,oldHp-attacker.hp);
        }
        recordAbilitySignal(g,v,'FINAL STAND','guard');
        return;
      }
      v.hp = 1;
      v.undyingUsedTurn = g.turn;
      log(g, `${COMMANDERS[v.cmd].short} survives a lethal blow: Code Bearer.`, v.side);
      recordAbilitySignal(g,v,'CODE BEARER','guard');
      return;
    }
    // The intrinsic survivals above (Anya, C.C., Urabe) and earlier combat
    // diversions (Guilford, Darlton) take priority. Only an otherwise fatal
    // enemy combat hit on another allied commander can trigger Masaki.
    if(!force&&!v.expMasakiSacrificeUsed&&attacker&&v.cmd&&
       foe(g,v.side,attacker.side)){
      const rescuer=g.units.filter(m=>m.hp>0&&m.id!==v.id&&m.cmd==='masaki'&&
        m.side===v.side&&!atSea(g,m)&&fx(m).ultimateGuardian&&
        !m.expMasakiSacrificeUsed&&dist(g,m,v)<=2).sort((a,b)=>a.id-b.id)[0];
      if(rescuer){
        v.hp=maxHP(v);
        recordAbilitySignal(g,v,'ULTIMATE GUARDIAN','heal');
        rescuer.expMasakiSacrificeUsed=true;
        rescuer.hp=0;
        log(g,'Ultimate Guardian: Masaki sacrifices himself to restore '+
          COMMANDERS[v.cmd].short+' to full HP.',v.side);
        kill(g,rescuer,attacker,false,null,0);
        return;
      }
    }
    v.hp = 0;
    if(v.cmd==='lelouch'&&!v.expDeathEmperorTriggered){
      v.expDeathEmperorTriggered=true;
      for(const ally of g.units){
        if(ally.hp<=0||ally.id===v.id||ally.side!==v.side||atSea(g,ally)||dist(g,v,ally)>3||ally.expMaskFalls)continue;
        ally.expDeathEmperor={side:v.side,turns:2};
        ally.morale=1;ally.moved=false;ally.attacked=false;ally.chain=0;
        ally.moveActionsThisTurn=0;ally.movedDistance=0;
      }
      log(g,'The Death of the Emperor rallies nearby Britannian formations.',v.side);
    }
    // All Hail Lelouch: one heal pulse per confirmed enemy formation destroyed.
    if(attacker&&attacker.side!==v.side){
      const emperors=g.units.filter(e=>e.hp>0&&e.cmd==='lelouch'&&e.side===attacker.side&&dist(g,e,attacker)<=3);
      if(emperors.length)for(const ally of g.units){
        if(ally.hp>0&&ally.side===attacker.side&&!atSea(g,ally)&&emperors.some(e=>dist(g,e,ally)<=3))
          ally.hp=Math.min(maxHP(ally),ally.hp+Math.max(1,Math.round(maxHP(ally)*0.05)));
      }
    }
    // Every confirmed combat death can reflect half the uncapped killing
    // attack. No counterattack is created; reflection damage cannot itself
    // trigger Necessary Sacrifices or retaliation.
    const reflected = !force && !v.warCounted
      ? smilasSacrificeReflection(g,v,attacker,incomingDamage) : 0;
    if (reflected && attacker.hp > 0) {
      const before = attacker.hp;
      attacker.hp = Math.max(0,attacker.hp-reflected);
      recordWarEvent(g,'damage',v.side,{
        targetSide:attacker.side,value:before-attacker.hp,unitId:v.id,group:warGroup(g,v)});
      log(g,`Necessary Sacrifices reflects ${reflected} damage onto ${COMMANDERS[attacker.cmd]?.short||TYPES[attacker.type].short}.`,v.side);
      recordAbilitySignal(g,attacker,'REFLECT −'+Math.max(0,before-attacker.hp),'debuff');
      // No attack source or incomingDamage: an attacker killed by reflection
      // gets ordinary survival checks, but cannot reflect or counter again.
      if (attacker.hp <= 0) kill(g,attacker,null,false,v.side,0);
    }
    dropCarriedEliminator(g, v, 'carrier destroyed');
    if(!v.warCounted){v.warCounted=true;
      recordWarEvent(g,'unitDestroyed',attacker?.side||by||v.side,{unitId:v.id,unitType:v.type,
        unitSide:v.side,commander:v.cmd||null,personal:!!v.personal,byCombat:!!attacker||!!by,
        group:warGroup(g,attacker),invasion:!!v.landingRefreshTurn&&v.landingRefreshTurn===g.turn});
      farneseOnDeath(g,v,attacker);
    }
    // A sunk Carrier-Battleship takes the Knightmares aboard down with it.
    for (const c of v.cargo?.splice(0) || []) {
      c.hp = 0;
      kill(g, c, attacker, true);
    }
    // Urabe's Final Stand: his fall rallies friendly units within 2 hexes to High morale.
    if (fx(v).martyr) for (const w of g.units) if (w.hp > 0 && w.side === v.side && dist(g, w, v) <= 2) w.morale = 1;
    // Cargo lost with a destroyed carrier is a consequential loss, not a
    // separate direct kill by the commander who sank the carrier.
    if (attacker && !force) {
      if (attacker.cmd) {
        const k = attacker.cmd,
          tally = (g.missionKills ||= {}),
          instances = (g.missionCommanderKills ||= {}),
          instance = `${attacker.side}:${k}:${attacker.personal ? 'personal' : 'operation'}`;
        tally[k] = (tally[k] || 0) + 1;
        // Legacy aggregate tallies cannot distinguish officer instances; never inherit another one's kills.
        instances[instance] = (instances[instance] || 0) + 1;
        honorAction(g, attacker, 'kills');
        if (v.cmd) {
          honorAction(g, attacker, 'commanderDefeats');
          award(g, attacker.side, 'valor', `Defeated ${COMMANDERS[v.cmd].short}`, attacker);
        }
      }
      // Asura Strike Force: actual combat kills, including lethal counterfire,
      // rally Ashley and friendly on-map formations within two hexes. Forced
      // deaths (e.g. carrier cargo) and non-hostile deaths do not trigger it.
      if (fx(attacker).asuraStrikeForce && foe(g, v.side, attacker.side)) {
        for (const ally of g.units) {
          if (ally.hp > 0 && ally.side === attacker.side && !atSea(g, ally) &&
              dist(g, attacker, ally) <= 2)
            ally.morale = Math.min(1, ally.morale + 1);
        }
      }
      attacker.kills++;
      attacker.xp = Math.min(5, attacker.xp + 1);
      attacker.morale = clamp(attacker.morale + 1, -3, 1);
    }
    if (v.cmd) log(g, `${COMMANDERS[v.cmd].short}'s unit is lost.`, v.side);
    newcomerKilled(g,v,attacker);
    // Toru must rally on a genuine death, including enemy counterfire.
    if(!force&&attacker&&foe(g,v.side,attacker.side))toruLastPromise(g,v,attacker);
    hooks.kill?.(g, v, attacker, by);
  }
  // Centralized post-volley action restoration. Distinguish a frame kill from a
  // newly destroyed city shield: both qualify for movement refresh where the
  // unit/skill allows it, but only true kills grant kill-only rewards.
  function applyAttackConsequences(g, a, d, s, destroyed, defensesDestroyed, movementBeforeAttack, f, aef) {
    const eliteBreakthrough = !!aef.breakthrough,
      eliteRelentless = !!aef.relentless;
    // Light Armor has no kill-based Breakthrough. Medium, heavy, super-heavy
    // and Amphibious II frames (ordinary or elite) can chain killing attacks
    // indefinitely; existing other Breakthrough elites retain that benefit.
    const type = TYPES[a.type];
    const hasBreakthrough = !(type.cls === 'light' && !type.naval) &&
      !!(type.breakthrough || eliteBreakthrough);
    const unlimitedChain = hasBreakthrough &&
      (type.elite || type.naval === 'amphibious' ||
        ['medium', 'heavy', 'super'].includes(type.cls));
    const cap = unlimitedChain ? Infinity : (f.refire || aef.refire ? 2 : 1);
    let breakthrough = false;
    if (destroyed && a.hp > 0 && hasBreakthrough && a.chain < cap) {
      // Breakthrough refreshes the attack. Heavy and super-heavy movement
      // is governed separately by their two-move-per-turn limit.
      a.chain++;
      a.attacked = false;
      if ((TYPES[a.type].relentless || eliteRelentless)) a.moved = false;
      breakthrough = true;
    } else if (destroyed && a.hp > 0 && (TYPES[a.type].relentless || eliteRelentless)) {
      // Relentless frames also fire again after a kill when Breakthrough is unavailable.
      a.attacked = false;
      breakthrough = true;
    }

    // Only Siege Artillery and units with Breakthrough earn a follow-up attack
    // for destroying city defenses. A simultaneous kill cannot bank a second attack.
    if (defensesDestroyed && a.hp > 0 && a.attacked &&
        (hasBreakthrough || (type.cls === 'siege' && type.branch === 'Artillery'))) {
      a.attacked = false;
      breakthrough = true;
    }

    // Every amphibious Elite (all factions, any Elite level) can chain both
    // movement and attacks after a kill or city-defense break, even if it moved
    // before firing. Normal Amphibious II keeps attack-only Breakthrough.
    if ((destroyed || defensesDestroyed) && a.hp > 0 &&
        type.elite && type.naval === 'amphibious') {
      a.attacked = false;
      a.moved = false;
      a.skillReposition = 0;
      a.eliteMoveAfterKill = true;
      breakthrough = true;
    }
    if (a.hp > 0 && aef.moveAfterAttack) { a.moved = false; a.skillReposition = 0; }
    // Kill-triggered Elite movement also triggers when city defenses fall.
    // Vincent Ward's Lv.5 refresh remains limited to its first qualifying
    // event per turn; its unlimited Breakthrough attack chain is unchanged.
    if ((destroyed || defensesDestroyed) && a.hp > 0 && aef.moveAfterKill &&
        (a.elite !== 'vincent_ward' || a.expVincentMoveTurn !== g.turn)) {
      if (a.elite === 'vincent_ward') a.expVincentMoveTurn = g.turn;
      a.moved = false;
      a.skillReposition = 0;
      a.eliteMoveAfterKill = true;
    }
    // Heavy/super-heavy frames reposition after kills or city-defense breaks,
    // but never more than twice in a single turn.
    if ((destroyed || defensesDestroyed) && a.hp > 0 &&
        isHeavyChainMover(a) && !heavyMoveLimitReached(g, a)) {
      a.moved = false;
      a.skillReposition = 0;
    }
    afterCommanderAttack(g,a,d,s,destroyed,defensesDestroyed);
    // A kill only primes momentum if an existing breakthrough or signature
    // ability actually granted another attack. It never grants one itself.
    if (destroyed && a.hp > 0 && !a.attacked &&
        !TYPES[a.type].naval && TYPES[a.type].branch === 'Armor')
      a.expAssaultMomentumTurn = g.turn;
    // Preserve movement that was available before an attack refreshed another
    // shot. A new move is granted only by a class, Elite or commander ability.
    if ((destroyed || defensesDestroyed) && a.hp > 0 && !a.attacked && movementBeforeAttack) {
      a.moved = false;
      if (!(TYPES[a.type].relentless || eliteRelentless || aef.moveAfterAttack || aef.moveAfterKill)) {
        if (!movementBeforeAttack.skillReposition && !movementBeforeAttack.withdrawMove) {
          a.skillReposition = 0;
          a.withdrawMove = false;
        } else {
          a.skillReposition = Math.max(a.skillReposition, movementBeforeAttack.skillReposition);
          a.withdrawMove ||= movementBeforeAttack.withdrawMove;
        }
      }
    }

    // The kill's suppression overrides every chain and elite action refresh.
    if (a.expFarneseSlaughterTurn === g.turn) {
      a.moved = true; a.attacked = true;
      a.withdrawMove = false; a.skillReposition = 0;
      a.eliteMoveAfterKill = false;
    }
    return breakthrough;
  }

    if (f.firstAttackMorale && a.firstAttackMoraleTurn !== g.turn) {

      shieldMult = variance * (crit ? pr.baseCritMult : 1),
      hit = [];

      !(a.expKallenFirstTargets || []).includes(d.id);

      d.expKallenSlowNext = true;
