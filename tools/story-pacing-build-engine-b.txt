  // ======== Economy and cities ========
  const REGIONAL_RESOURCES = ['credits','industry','science','sakuradite'];
  const CITYLESS_WARCHEST_TURNS = 10;
  // Lightweight manufacturing Research costs shared across factions/modes.
  const FRAME_RESEARCH = Object.freeze({
    scout:0, assault:0, raider:2, light:0, medium:5, heavy:10,
    super:15, support:0, rocket:7, siege:12
  });
  function frameResearch(t) {
    if (!t) return 0;
    if (t.air) return t.research;
    if (t.naval === 'ship') return 12;
    if (t.naval === 'amphibious') return t.tier >= 2 ? 4 : 0;
    return FRAME_RESEARCH[t.cls] || 0;
  }
  // Scenarios complete initial treasury setup before assigning startUnits.
  // Double each side's opening stockpile once, not its recurring income.
  function doubleScenarioResearch(g) {
    if (g.mode !== 'scenario' || !g.scenario || !g.startUnits ||
        g.scenario.startingResearchDoubled) return;
    for (const [side,wallet] of Object.entries(g.economy || {}))
      if (side !== 'neutral') wallet.science = 2 * (wallet.science || 0);
    g.scenario.startingResearchDoubled = true;
  }


  // Replace scripted recurring subsidies with city-tier development. Scenario
  // starting cash remains authored; cityless expeditions retain finite reserves.
  // Several legacy scenario constructors clamp negative funding to zero. That
  // destroys the original target once standardized city output exceeds it.
  // Preserve their actual authored targets here rather than treating the
  // inflated new income as the goal.
  const REGIONAL_CREDIT_TARGETS = Object.freeze({
    'cf-eu-france':{cf:880,eu:820},
    'cf-eu-moscow':{cf:1080,eu:910},
    'eu-cf-beijing':{eu:900,cf:1050},
    'eu-cf-luoyang':{eu:1140,cf:1190},
    'eu-cf-sichuan':{eu:890,cf:930},
    'battle-luoyang':{britannia:1080,cf:1190},
    'invasion-japan':{britannia:400,bk:300,cf:320},
    'defense-japan':{bk:270,britannia:540},
  });
  // Reconcile every regional battle against its authored credit targets using
  // *only* genuine city tiers. A fortress is always city level 4 with a battery;
  // demoting one removes its battery as well. Factories and labs are untouched.
  function balanceScenarioTiers(g, targets) {
    if (!g?.stations?.length || !g.scenario) return;
    const stats = CITY_STANDARD_STATS;
    const sides = Object.entries(targets || {}).filter(([side, wanted]) =>
      Number.isFinite(wanted?.credits) && g.stations.some(s => s.owner === side));
    if (!sides.length) return;
    const bounds = Object.fromEntries(sides.map(([side, wanted]) => {
      const cities = g.stations.filter(s => s.owner === side);
      const min = cities.reduce((n,s) => n + (s.nationalCapital ? stats.capital.credits : stats[1].credits), 0);
      const max = cities.reduce((n,s) => n + (s.nationalCapital ? stats.capital.credits : stats[4].credits), 0);
      return [side, {min, max, requested:Math.max(1,Math.round(wanted.credits))}];
    }));
    // If a small invasion army cannot earn its old budget even with all its
    // cities at level 4, proportionally reduce the opposing targets too.
    // If that is still impossible (e.g. Canada has 68 defending cities and
    // the invasion holds only six), clamp to each side's legal tier limits.
    const lowestScale = Math.max(...Object.values(bounds).map(b => b.min / b.requested));
    const highestScale = Math.min(...Object.values(bounds).map(b => b.max / b.requested));
    const scale = lowestScale <= highestScale
      ? Math.max(lowestScale, Math.min(1,highestScale))
      : highestScale;
    g.scenario.tierBalanceShortfalls = {};
    g.scenario.tierBalanceSummary = {};
    for (const [side] of sides) {
      const {min,max,requested} = bounds[side];
      const goal = Math.max(min,Math.min(max,Math.round(requested * scale)));
      const cities = g.stations.filter(s => s.owner === side);
      const total = () => cities.reduce((sum,s) => sum + cityStats(s).credits, 0);
      const applyTier = (s,next) => {
        const previousMax = Math.max(1,Number(s.maxShield)||1);
        const fraction = Math.max(0,Math.min(1,(Number(s.shield)||0)/previousMax));
        s.cityTier = next;
        if (next === 4) {
          s.fort = true;
          s.gun ||= s.name + ' Defensive Battery';
          s.gunReady ??= g.turn || 1;
        } else if (s.fort) {
          s.fort = false;
          s.gun = null;
          s.gunReady = 0;
          delete s.gunUsedTurn;
        }
        syncBuildings(g,s);
        s.shield = Math.round(s.maxShield * fraction);
      };
      let before = total(), current = before;
      let raised = 0, lowered = 0, removedFortresses = 0, addedFortresses = 0;
      // Lower ordinary cities first, then less-important fortresses, and
      // demote headquarters last. National capitals are never downgraded.
      while (current > goal) {
        const candidates = cities.filter(s => !s.nationalCapital && cityTierOf(s)>1)
          .map(s => ({
            city:s, step:stats[cityTierOf(s)].credits-stats[cityTierOf(s)-1].credits,
          }))
          .filter(x => Math.abs(current-x.step-goal) < Math.abs(current-goal))
          .sort((a,b) =>
            Number(!!a.city.fort)-Number(!!b.city.fort) ||
            Number(!!a.city.regionalHQ)-Number(!!b.city.regionalHQ) ||
            Math.abs(current-a.step-goal)-Math.abs(current-b.step-goal) ||
            a.city.id-b.city.id);
        if (!candidates.length) break;
        const s=candidates[0].city, wasFort=!!s.fort;
        current-=candidates[0].step;
        applyTier(s,cityTierOf(s)-1);
        lowered++;
        if (wasFort && !s.fort) removedFortresses++;
      }
      // Improve small regional economies via city upgrades. The transition
      // from level 3 to level 4 creates the full fortress + battery, never a
      // level-4 city without a battery.
      while (current < goal) {
        const candidates = cities.filter(s => !s.nationalCapital && cityTierOf(s)<4)
          .map(s => ({
            city:s, step:stats[cityTierOf(s)+1].credits-stats[cityTierOf(s)].credits,
          }))
          .filter(x => Math.abs(current+x.step-goal) < Math.abs(current-goal))
          .sort((a,b) =>
            cityTierOf(a.city)-cityTierOf(b.city) ||
            Math.abs(current+a.step-goal)-Math.abs(current+b.step-goal) ||
            Number(!a.city.regionalHQ)-Number(!b.city.regionalHQ) ||
            a.city.id-b.city.id);
        if (!candidates.length) break;
        const s=candidates[0].city, wasFort=!!s.fort;
        current+=candidates[0].step;
        applyTier(s,cityTierOf(s)+1);
        raised++;
        if (!wasFort && s.fort) addedFortresses++;
      }
      g.scenario.tierBalanceShortfalls[side] = Math.max(0,requested-current);
      g.scenario.tierBalanceSummary[side] = {
        previousTarget:requested, scaledTarget:goal, before, after:total(),
        raised, lowered, removedFortresses, addedFortresses,
        minimum:min, maximum:max,
      };
    }
    g.scenario.tierBalanceVersion = 2;
  }
  // Regional starting economies are limited by removing actual minor cities,
  // never by changing city yields or applying an invisible recurring penalty.
  // This runs once while a NEW scenario's authored funding is being localized;
  // already-localized saves and World Conquest are left untouched.
  // Enforce authored regional city rosters (fixed per-scenario exclusions).
  // Existing saved games and Conquest never enter this migration.
  function pruneScenarioOpeningCities(g) {
    if (g.mode !== 'scenario' || !g.scenario || g.turn !== 1 || !g.startUnits ||
        g.cityStandardMigrated || g.scenario.cityPruningVersion ||
        g.scenario.pruningCities) return;
    g.scenario.pruningCities = true;
    try {
      const exclusions = REGIONAL_SCENARIO_CITY_ROSTERS[g.scenario.id] || [];
      const omitted = new Set(exclusions);
      const removed = g.stations.filter(s => omitted.has(s.name));
      const sides = [...new Set([...g.order, ...g.stations.map(s => s.owner)])]
        .filter(s => s && s !== 'neutral');
      const before = Object.fromEntries(sides.map(s =>
        [s, {income: income(g, s).credits,
          cities: g.stations.filter(c => c.owner === s).length}]));
      if (removed.length) {
        g.stations = g.stations.filter(s => !omitted.has(s.name));
        const targets = new Map();
        for (const lost of removed) {
          const friendly = g.stations.filter(s => s.owner === lost.owner);
          const eligible = friendly.length ? friendly : g.stations;
          if (!eligible.length) continue;
          targets.set(lost.id, eligible.slice().sort((a,b) =>
            dist(g, lost, a) - dist(g, lost, b) || a.id - b.id)[0]);
        }
        for (const t of g.tiles) {
          const replacement = targets.get(t.provinceCity);
          if (replacement) t.provinceCity = replacement.id;
        }
        for (const d of g.sites || []) {
          const replacement = targets.get(d.city);
          if (replacement) d.city = replacement.id;
        }
        // An army group must not refer to an abolished city as its origin
        // or scripted target. Redirect it to the nearest surviving city.
        for (const group of g.scenario.groups || []) {
          for (const field of ['origin','destinationName','targetCity']) {
            const old = removed.find(s => s.name === group[field]);
            if (old && targets.has(old.id)) group[field] = targets.get(old.id).name;
          }
        }
        g.mapRevision = (g.mapRevision || 0) + 1;
      }
      g.scenario.cityPruningVersion = 2;
      g.scenario.cityPruningSummary = Object.fromEntries(sides.map(side => {
        const count = g.stations.filter(s => s.owner === side).length;
        return [side, {before:before[side].income, after:income(g, side).credits,
          citiesBefore:before[side].cities, citiesAfter:count,
          removed:before[side].cities-count}];
      }));
    } finally {
      delete g.scenario.pruningCities;
    }
  }
  function localizeScenarioIncome(g) {
    if(g?.mode!=='scenario'||!g.scenario||
      g.scenario.economyLocalized||g.scenario.localizingIncome)return false;
    const funding=g.scenario.funding||{};
    const sides=Object.keys(funding);
    if(!sides.length)return false;
    g.scenario.localizingIncome=true;
    try {
      const targets={};
      for(const side of sides){
        const cities=g.stations.filter(s=>s.owner===side);
        if(!cities.length){
          const former=income(g,side);
          const wallet=funds(g,side);
          const reserveTurns=Number.isInteger(g.scenario.citylessWarChestTurns)
            &&g.scenario.citylessWarChestTurns>0
              ?g.scenario.citylessWarChestTurns:CITYLESS_WARCHEST_TURNS;
          for(const resource of REGIONAL_RESOURCES)
            wallet[resource]=Math.max(wallet[resource]||0,
              Math.max(0,Math.round(former[resource]*reserveTurns)));
          continue;
        }
        // The scenario constructors set funding=target-base. Recover the
        // intended credit target, but never retain a recurring hidden subsidy.
        // During localization income() still includes the authored funding,
        // so this value is already the original target (do not add twice).
        const totalWithFunding=income(g,side);
        const explicit = REGIONAL_CREDIT_TARGETS[g.scenario.id]?.[side];
        targets[side]={credits:explicit ?? Math.round(totalWithFunding.credits)};
      }
      if(!g.cityStandardMigrated)balanceScenarioTiers(g,targets);
      g.scenario.funding={};
      g.scenario.economyLocalized=true;
      pruneScenarioOpeningCities(g);
      return true;
    }finally {
      delete g.scenario.localizingIncome;
    }
  }

  function income(g, side) {
    if(g?.stations?.length&&!g.cityStandardVersion)standardizeCities(g);
    // Regional constructors set startUnits only after calculating and storing
    // their opening resource reserves. This safely localizes funding on the
    // first post-construction income query for any scenario or campaign battle.
    if (g.mode === 'scenario' && g.startUnits && !g.scenario?.localizingIncome &&
        !g.scenario?.pruningCities) {
      localizeScenarioIncome(g);
      pruneScenarioOpeningCities(g);
      doubleScenarioResearch(g);
    }
    if (g.scenario?.economyLocalized && g.mode === 'scenario' &&
        !g.stations.some(s => s.owner === side))
      return {credits:0,industry:0,science:0,sakuradite:0};
    const refining = 1 + techValue(g, side, 'cities.refining');
    const total = g.stations
      .filter(s => s.owner === side)
      .reduce((a, s) => {
        const y = cityOutput(g, s);
        if(s.expSupplyBlocked?.owner===side)return a;
        a.credits += y.credits*(s.expTaxUntil>=g.turn?.8:1);
        if(g.units.some(u=>u.hp>0&&u.side===side&&u.c===s.c&&u.r===s.r&&fx(u).nationalUnity))a.credits+=Math.round(y.credits*.15);
        if(g.units.some(u=>u.hp>0&&u.side===side&&u.c===s.c&&u.r===s.r&&fx(u).royalCompassion))a.credits+=Math.round(y.credits*.15);

        // The Mandate of Heaven aura does not stack with multiple copies.
        if(g.units.some(u=>u.hp>0&&!atSea(g,u)&&u.side===side&&fx(u).mandateOfHeaven&&dist(g,u,s)<=3)){
          a.credits+=y.credits*.25*(s.expTaxUntil>=g.turn?.8:1);
          a.industry+=y.industry*.25;
        }
        a.industry += y.industry;
        const lloydHere = g.units.some(u => u.hp > 0 && u.side === side && u.cmd === 'lloyd' && u.c === s.c && u.r === s.r);
        a.science += y.science * (lloydHere ? 2 : 1);
        // Scenario balancing income is owned by the city and transfers on
        // capture instead of surviving as a permanent off-map stipend.
        for(const resource of REGIONAL_RESOURCES)
          a[resource] += (s.divisionIncome?.[resource] || 0) +
            (s.regionalIncome?.[resource] || 0);
        return a;
      }, { credits: 0, industry: 0, science: 0, sakuradite: 0 });
    total.industry = Math.round(total.industry);
    total.science = Math.round(total.science);
    // Sakuradite deposits: extraction by refinery level, with Japan's output allocated among the powers; a level-3
    // refinery also exports for credits.
    for (const d of g.sites || []) {
      const host = depositHost(g, d);
      const lloydAtHost = host && host.owner === side && g.units.some(u => u.hp > 0 && u.side === side && u.cmd === 'lloyd' && u.c === host.c && u.r === host.r);
      total.sakuradite += (depositShares(g, d)[side] || 0) * refining * (g.units.some(u=>u.hp>0&&u.side===side&&fx(u).kyoto&&dist(g,u,d)<=2)?1.25:1) * (lloydAtHost ? 2 : 1);
      if (depositOwner(g, d) === side) total.credits += depositYield(g, d).credits;
    }
    // Regional scenarios localize the national Sakuradite allotment into
    // cities as well; Conquest retains its national grant.
    if (g.mode !== 'campaign' && (g.mode === 'scenario' ?
        (g.order.includes(side) && g.stations.some(s=>s.owner===side)) : isMajor(g,side)) && alive(g,side))
      total.sakuradite += SAKURADITE.national;
    total.sakuradite = Math.round(total.sakuradite);
    // Only during scenario construction and one-time migration do the authored
    // funding targets exist as virtual stipends. They are retired on launch.
    const support = g.mode === 'scenario' && !g.scenario?.economyLocalized &&
      alive(g, side) &&
      (g.stations.some(s => s.owner === side) ||
        g.scenario?.citylessFunding?.includes(side)) ? g.scenario?.funding?.[side] : null;
    if (support) for (const resource of ['credits','industry','science','sakuradite'])
      total[resource] += support[resource] || 0;
    return total;
  }
  // New units deploy on the city hex itself, naval units on the port's sea hex; nothing is built while a unit stands
  // there (move it off first).
  function recruitOptions(g, s, side, type = null) {
    const port = portBuilt(TYPES[type]);
    if ((port ? s.portOwner || s.owner : s.owner) !== side) return [];
    const at = port ? s.portAt && tile(g, s.portAt.c, s.portAt.r) : tile(g, s.c, s.r);
    return at && !unitAt(g, at) ? [at] : [];
  }
  // Unit prices depend on the frame and stack size, not faction.
  function price(type, stack = 1, g = null, side = null) {
    const t = TYPES[type];
    return {
      credits: Math.round(t.cost * (1 + 0.85 * (stack - 1))),
      industry: Math.round(t.industry * (1 + 0.85 * (stack - 1))),
      sakuradite: Math.round((t.sakuradite ?? SAKURADITE.cost[t.cls] ?? 0) * (1 + 0.85 * (stack - 1))),
      science: Math.round(frameResearch(t) * (1 + 0.85 * (stack - 1))),
    };
  }
  // Field Supply applies to the selected formation or the production hex.
  // It never subsidizes other factions or a commander embarked at sea.
  function klausSupplyNear(g, place, side) {
    return !!g && !!place && g.units.some(v => v.hp > 0 && v.side === side &&
      fx(v).fieldSupply && !atSea(g, v) && dist(g, v, place) <= 2);
  }
  // Anna's workshop works within two hexes, independently of city supply.
  // Disabled or embarked commanders never provide field maintenance.
  function annaMaintenanceNear(g, u) {
    return !!g && !!u && g.units.some(v => v.hp > 0 && v.side === u.side &&
      fx(v).annaMaintenance && !atSea(g, v) && dist(g, v, u) <= 2);
  }
  function halveResources(cost) {
    return Object.fromEntries(Object.entries(cost).map(([key, amount]) =>
      [key, Math.round(amount * 0.5)]));
  }
  function recruitCost(g, s, type, stack = 1) {
    const t = TYPES[type];
    if (!s || !t) return price(type, stack, g);
    const producer = portBuilt(t) ? (s.portOwner || s.owner) : s.owner;
    const place = portBuilt(t) && s.portAt ? s.portAt : s;
    const cost = price(type, stack, g, producer);
    return klausSupplyNear(g, place, producer) ? halveResources(cost) : cost;
  }
  function canBuy(g, s, type, stack = 1, vacating = null) {
    return !buyReason(g, s, type, stack, vacating);
  }
  function recruit(g, stationId, type, stack = 1, position) {
    const s = g.stations.find(s => s.id === stationId);
    const why = buyReason(g, s, type, stack);
    if (why) return { ok: false, reason: why };
    const producer = portBuilt(TYPES[type]) ? (s.portOwner || s.owner) : s.owner;
    const options = recruitOptions(g, s, producer, type);
    const p = position ? options.find(p => p.c === position.c && p.r === position.r) : options[0];
    if (!p) return { ok: false, reason: 'Deployment hex unavailable.' };
    // The decree belongs to the city owner, not an occupying third-party port operator.
    // Naval purchases consume it just like land purchases and building upgrades.
    const cost = decreePrice(recruitCost(g, s, type, stack), s.owner === producer ? s : null);
    if (s.expDecree && s.owner === producer) delete s.expDecree;
    warSpend(g,producer,cost,'production');
    spend(funds(g, producer), cost);
    if (portBuilt(TYPES[type])) s.portProducedTurn = g.turn;
    else s.producedTurn = g.turn;
    const u = newUnit(g, type, producer, p.c, p.r, stack, null, false);
    log(g, `${TYPES[type].short} ×${stack} rolls out at ${s.name}. Ready next turn.`, producer);
    return { ok: true, unit: u };
  }

  // Deployment cost is fixed by elite class, regardless of level or rarity.
  function elitePrice(id) {
    const e = ELITE_FORCES[id], t = e && TYPES[e.type];
    if (!t) return { credits: 0, industry: 0, sakuradite: 0 };
    return {
      credits: t.cost,
      industry: t.industry,
      sakuradite: SAKURADITE.elite[t.naval === 'amphibious' ? 'amphib' : t.cls] ?? 0,
      science: frameResearch(t),
    };
  }
  function eliteRecruitCost(g, s, id) {
    const t = TYPES[ELITE_FORCES[id]?.type];
    const producer = t && portBuilt(t) ? (s?.portOwner || s?.owner) : s?.owner;
    const place = t && portBuilt(t) && s?.portAt ? s.portAt : s;
    const cost = elitePrice(id);
    return klausSupplyNear(g, place, producer) ? halveResources(cost) : cost;
  }
  // Land elites use a city factory; amphibious elites use the independently owned
  // port and its separate one-unit-per-turn shipyard slot.
  // Mission-owned roster snapshots provide an explicit, save-safe borrowing map.
  // Outside Elite Campaign, ordinary faction unlock/recruit rules remain unchanged.
  // Faction ownership comes from ELITE_FORCES, including the Federation's
  // command, missile and siege variants. The 2018 UFN still uses Black Knights equipment.
  function eliteEligible(g, side, id) {
    const e = ELITE_FORCES[id];
    if (!e) return false;
    // Scenario-only factions (alternate history) follow the rules of the power whose equipment they use.
    const kit = g?.scenario?.equipment?.[side] || side;
    const coalition = g?.mode === 'scenario' && g.scenario?.rosterSources?.[side];
    if (coalition) return (g.scenario.eliteRosters?.[side] || []).includes(id) &&
      coalition.some(source => e.availableTo.includes(source));
    if (kit === 'cf') {
      if (ufnBlackKnights(g, side)) return e.availableTo.includes('bk');
      // Older saves may retain a three-land-unit roster snapshot. Native
      // Federation unlocks must not disappear because that snapshot is stale.
      return e.availableTo.includes('cf');
    }
    const roster = g.mode === 'campaign' ? g.campaign?.eliteRosters?.[side] :
      scenarioMeta(g).eliteRosterOnly ? g.scenario.eliteRosters?.[side] : null;
    if (roster) return roster.includes(id);
    // Euro Britannia shares Britannia's full Imperial Elite Force roster in every mode.
    return e.availableTo.includes(kit) || (kit === 'eb' && e.availableTo.includes('britannia'));
  }
  function eliteDeployReason(g, s, id, profile = {}) {
    const e = ELITE_FORCES[id],
      rec = e && eliteRecord(profile, id),
      t = e && TYPES[e.type];
    if (!e || !s || !t) return 'Unavailable';
    const naval = !!t.naval,
      producer = naval ? (s.portOwner || s.owner) : s.owner,
      availableLevel = e.scenarioOnly && eliteScenario(g) ? 1 : rec?.level;
    return (
      (g.over ? 'Operation over' : producer !== g.phase ? (naval ? 'Not your port' : 'Not your city') : null) ||
      (!naval ? cityBusyReason(g, s) : null) ||
      (s.expArkFleet?.owner === s.owner ? 'Ark Fleet terror prevents recruitment' : null) ||
      (s.expSupplyBlocked?.owner === s.owner ? 'City supply lines are disrupted' : null) ||
      (!eliteEligible(g, producer, id) ? 'This Elite Force is not available to this faction' : null) ||
      (e.scenarioOnly && !eliteScenario(g) ? 'Available only in the relevant Elite scenario' : null) ||
      (!availableLevel ? `Locked — collect ${ELITE_UNLOCK_FRAGMENTS} fragments and unlock it in HQ` : null) ||
      (naval && (!s.portAt || (s.portLevel || 0) < t.port) ? `Requires a level-${t.port} port` : null) ||
      (!naval && s.tier < t.tier ? `Requires factory level ${t.tier}` : null) ||
      (g.eliteDeployed?.[id] ? 'Already deployed in this operation' : null) ||
      (naval && s.portProducedTurn === g.turn ? 'This port already launched a unit this turn' : null) ||
      (!recruitOptions(g, s, producer, e.type).length ? (naval ? 'A unit is on the port' : 'A unit is on the city') : null) ||
      shortfall(funds(g, producer), decreePrice(eliteRecruitCost(g, s, id), s.owner === producer ? s : null))
    );
  }
  function deployElite(g, stationId, id, profile = {}, position) {
    const s = g.stations.find(v => v.id === stationId),
      why = eliteDeployReason(g, s, id, profile);
    if (why) return { ok: false, reason: why };
    const e = ELITE_FORCES[id],
      t = TYPES[e.type],
      producer = t.naval ? (s.portOwner || s.owner) : s.owner,
      rec = e.scenarioOnly && eliteScenario(g) ? { level: 1 } : eliteRecord(profile, id),
      options = recruitOptions(g, s, producer, e.type),
      p = position ? options.find(v => v.c === position.c && v.r === position.r) : options[0],
      cost = decreePrice(eliteRecruitCost(g, s, id), s.owner === producer ? s : null);
    if (!p) return { ok: false, reason: 'Deployment hex unavailable.' };
    if (s.expDecree && s.owner === producer) delete s.expDecree;
    warSpend(g,producer,cost,'production');
    spend(funds(g, producer), cost);
    if (t.naval) s.portProducedTurn = g.turn;
    else s.producedTurn = g.turn;
    g.eliteDeployed ||= {};
    g.eliteDeployed[id] = true;
    const u = newUnit(g, e.type, producer, p.c, p.r, 1, null, false);
    u.elite = id;
    if (g.mode === 'campaign') u.storyHQDeployed = true;
    // Player factory/port recruits use their unlocked HQ level in every mode.
    // Preplaced converted elites have their own Lv.1 starting rule; recruitment never changes them.
    u.eliteLevel = rec.level;
    u.hp = maxHP(u);
    log(g, `${t.name} · Elite Lv.${u.eliteLevel} deploys at ${t.naval ? 'the port of ' : ''}${s.name}. Ready next turn.`, producer);
    return { ok: true, unit: u, cost };
  }
  // Rival factories can repeat Elite Forces indefinitely, but each frame must
  // satisfy its factory tier just as it does for a player.
  function aiEliteRecruitReason(g, s, id, vacating = null) {
    const e = ELITE_FORCES[id], t = e && TYPES[e.type];
    if ((!eliteAIProduction(g) && !eliteCampaign(g)) || !s || !e || !t) return 'Not an Elite AI operation';
    const naval = !!t.naval,
      producer = naval ? (s.portOwner || s.owner) : s.owner;
    if (producer === g.player || !eliteEligible(g, producer, id) ||
        (eliteConquest(g) && (naval || !isMajor(g, producer))) ||
        (eliteScenario(g) && !g.order.includes(producer)) ||
        (eliteCampaign(g) && naval) ||
        (e.scenarioOnly && !eliteScenario(g)))
      return 'Not an eligible Elite AI factory or port';
    return (
      turnReason(g, producer) ||
      (!naval ? cityBusyReason(g, s) : null) ||
      (s.expSupplyBlocked?.owner === s.owner ? 'City supply lines are disrupted' : null) ||
      (naval && (!s.portAt || (s.portLevel || 0) < t.port)
        ? `Requires port level ${t.port}` : null) ||
      (!naval && s.tier < t.tier ? `Requires factory level ${t.tier}` : null) ||
      (naval && s.portProducedTurn === g.turn ? 'This port already launched a unit this turn' : null) ||
      (!recruitOptions(g, s, producer, e.type).length &&
        !(vacating && !naval && unitAt(g, s) === vacating)
        ? (naval ? 'A unit is on the port' : 'A unit is on the city') : null) ||
      shortfall(funds(g, producer), decreePrice(eliteRecruitCost(g, s, id), s.owner === producer ? s : null))
    );
  }
  function recruitAIElite(g, stationId, id) {
    const s = g.stations.find(v => v.id === stationId);
    const why = aiEliteRecruitReason(g, s, id);
    if (why) return { ok: false, reason: why };
    const e = ELITE_FORCES[id], t = TYPES[e.type],
      producer = t.naval ? (s.portOwner || s.owner) : s.owner,
      p = recruitOptions(g, s, producer, e.type)[0],
      level = eliteCampaign(g) ? 3 : 1,
      cost = decreePrice(eliteRecruitCost(g, s, id), s.owner === producer ? s : null);
    if (!p) return { ok: false, reason: 'Deployment hex unavailable' };
    if (s.expDecree && s.owner === producer) delete s.expDecree;
    warSpend(g,producer,cost,'production');
    spend(funds(g, producer), cost);
    if (t.naval) s.portProducedTurn = g.turn;
    else s.producedTurn = g.turn;
    const u = newUnit(g, e.type, producer, p.c, p.r, 1, null, false);
    u.eliteLevel = level;
    u.hp = maxHP(u);
    log(g, `${t.name} · Elite Lv.${level} deploys at ${t.naval ? 'the port of ' : ''}${s.name}. Ready next turn.`, producer);
    return { ok: true, unit: u, cost };
  }
  function unitStats(g, u) {
    const t = TYPES[u.type], s = eliteScale(u), r = rangeOf(g, u);
    return {
      hp: maxHP(u),
      attack: Math.round(intrinsicAttack(g, u)),
      armor: t.armor + s.armor + unitTech(g, u, 'armor'),
      move: isSea(tile(g, u.c, u.r))
        ? t.naval === 'amphibious' ? amphibiousSea(g, u) : atSea(g, u) ? seaMove(g, u) : movement(g, u)
        : movement(g, u),
      min: r.min,
      max: r.max,
    };
  }
  // Rakshata supports her own unit and adjacent allies. A single nearby
  // engineer grants the full benefit; overlapping copies never stack.
  function rakshataNear(g, u) {
    return !!g && g.units.some(v => v.hp > 0 && v.side === u.side &&
      fx(v).rakshataEngineering && dist(g, v, u) <= 1);
  }
  // u: the unit being reinforced, with Inoue and Rakshata discounts.
  function reinforceCost(type, g = null, side = null, u = null) {
    const p = price(type, 1, g, side),
      k = u && g && logisticsNear(g, u) ? 0.7 : 1,
      engineering = u && g && rakshataNear(g, u) ? 0.5 : 1,
      supply = u && annaMaintenanceNear(g, u) ? 0.5 : 1;
    return { credits: Math.round(p.credits * k * engineering * supply), industry: Math.round(p.industry * k * engineering * supply),
      sakuradite: Math.round(p.sakuradite * engineering * supply), science: Math.round(p.science * engineering * supply) };
  }
  // Manual repairs restore 20% of maximum HP for a fifth of the unit's build price.
  function repairCost(u, g = null) {
    const half = g && annaMaintenanceNear(g, u) ? 0.5 : 1;
    const staff = g && skillNear(g, u, 'staffRepair', 1) ? 0.8 : 1;
    const camelot = g && u.elite && !TYPES[u.type].naval && skillNear(g, u, 'combatEngineering', 2) ? 0.8 : 1;
    // Research discounts affect credits only: manual repairs still restore exactly 20% HP.
    const energyNetwork = g ? 1 - techValue(g, u.side, 'cities.energy') : 1;
    // A ship must occupy a port held by its own side, not merely be adjacent to any city.
    const friendlyDock = g && isShip(u) && atFriendlyPort(g, u);
    const dockyard = friendlyDock ? 1 - techValue(g, u.side, 'naval.damage') : 1;
    const normalCost = Math.max(10, Math.round(baseRepairCost(u) * half * staff * camelot *
      (g && logisticsNear(g, u) ? 0.7 : 1) * energyNetwork * dockyard *
      (g && rakshataNear(g, u) ? 0.5 : 1)));
    // A friendly city with zero defenses doubles land-unit repair costs.
    // If multiple friendly cities are in range, an intact one can supply the repair.
    if (g && !isShip(u)) {
      const nearby = g.stations.filter(s => s.owner === u.side && dist(g, s, u) <= 1);
      if (nearby.length && nearby.every(s => s.shield <= 0)) return normalCost * 2;
    }
    return normalCost;
  }
  function baseRepairCost(u) {
    return Math.max(20, Math.round(price(u.type, u.stack, null, u.side).credits * 0.2));
  }
  function reinforce(g, id) {
    const check = validateAction(g, { kind: 'reinforce', id });
    if (!check.ok) return check;
    const u = check.actor;
    const cost = reinforceCost(u.type, g, u.side, u),
      e = funds(g, u.side);
    warSpend(g,u.side,cost,'reinforcement');
    spend(e, cost);
    const old = maxHP(u);
    u.stack++;
    if (g.mode === 'conquest' && u.side === g.player && g.challengePassRun?.version === 1)
      g.challengePassRun.produced++;
    u.hp += maxHP(u) - old;
    // Rapid Maintenance never spends or restores a formation's normal orders.
    if (!annaMaintenanceNear(g, u)) u.moved = u.attacked = true;
    log(g, `${TYPES[u.type].short} reinforced to ${u.stack} frames.`, u.side);
    return { ok: true };
  }
  function repair(g, id) {
    const check = validateAction(g, { kind: 'repair', id });
    if (!check.ok) return check;
    const u = check.actor;
    const cost = repairCost(u, g);
    warSpend(g,u.side,{credits:cost},'repair');
    funds(g, u.side).credits -= cost;
    const amount = Math.min(maxHP(u) - u.hp, Math.round(maxHP(u) * 0.20));
    u.hp += amount;
    if (g.mode === 'conquest' && u.side === g.player && g.challengePassRun?.version === 1)
      g.challengePassRun.manualRepair = true;
    if (!annaMaintenanceNear(g, u)) u.moved = u.attacked = true;
    log(g, `${TYPES[u.type].short} repairs ${amount} frame.`, u.side);
    return { ok: true, amount };
  }
  const BUILDINGS = {
    factory: {
      name: 'Knightmare factory',
      field: 'tier',
      desc: 'Unlocks heavier frames (level 2: mainline, raider and rocket; level 3: heavy, super-heavy and siege). +10 industry per level. Only City Tier upgrades raise city defenses.',
    },
    lab: {
      name: 'Research lab',
      field: 'lab',
      desc: 'Produces research (+8 per level). Advanced units, reinforcement frames and Elite Forces cost Research; HQ upgrades use Command Tokens.',
    },
    port: {
      name: 'Port',
      field: 'portLevel',
      desc: 'Coastal cities only, on a sea hex beside the city. Level 1 builds amphibious Knightmares, level 2 Carrier-Battleships, and level 3 enables tier IV naval research. Docked naval units do not heal automatically.',
    },
    refinery: {
      name: 'Sakuradite refinery',
      field: 'refinery',
      desc: 'Only where there is a Sakuradite deposit. Extracts 25% of its output, then 50%, 75% and 100% (+15 credits) at levels 1–3.',
    },
  };
  function buildingLevel(s, kind) {
    return s[BUILDINGS[kind].field] || 0;
  }
  function buildCost(s, kind) {
    const l = buildingLevel(s, kind);
    // A factory wrecked to level 0 (F.L.E.I.J.A.) is rebuilt for the price of a level-1 lab.
    if (kind === 'factory') return l ? { credits: 160 * l, industry: 40 * l } : { credits: 110, industry: 25 };
    if (kind === 'lab') return { credits: 110 * (l + 1), industry: 25 * (l + 1) };
    if (kind === 'port') return { credits: 140 * (l + 1), industry: 35 * (l + 1) };
    return { credits: 120 * (l + 1), industry: 30 * (l + 1) };
  }
  // Allow repeated city-tier upgrades in one turn, including immediately after capture.
  // Newly captured cities stay at zero shield; higher maximum HP never restores damage.
  function upgradeCityTier(g, id) {
    const s = g.stations.find(v => v.id === id);
    const why = cityTierReason(g, s);
    if (why) return { ok: false, reason: why };
    const previous = cityTierOf(s), cost = decreePrice(cityTierCost(s), s);
    warSpend(g, s.owner, cost, 'construction');
    spend(funds(g, s.owner), cost);
    if (s.expDecree) delete s.expDecree;
    s.cityTier = previous + 1;
    if (s.cityTier === 4) {
      s.fort = true;
      s.gun ||= s.name + ' battery';
      s.gunReady = Math.max(s.gunReady || 0, g.turn + 1);
    }
    syncBuildings(g,s);
    s.shield = Math.min(s.shield, s.maxShield);
    if (g.mode === 'conquest' && s.owner === g.player && g.challengePassRun?.version === 1)
      g.challengePassRun.constructed = true;
    log(g, `${s.name}: upgraded to ${s.cityTier === 4 ? 'Fortress City' : 'City Level ' + s.cityTier}.`, s.owner);
    return { ok: true, tier: s.cityTier };
  }

  // Factory and lab upgrades are additions to the universal city-tier yield.
  function buildingFoundation(g,s) {
    const stats=cityStats(s);
    return s.buildingFoundation={
      industry:stats.industry,science:stats.science,maxShield:stats.hp,
    };
  }
  function syncBuildings(g,s) {
    const stats=cityStats(s);
    s.income=stats.credits;
    s.industry=stats.industry+10*(s.tier||0);
    s.science=stats.science+8*(s.lab||0);
    s.maxShield=stats.hp;
    s.buildingFoundation={industry:stats.industry,science:stats.science,maxShield:stats.hp};
  }
  function build(g, id, kind) {
    const s = g.stations.find(s => s.id === id),
      b = BUILDINGS[kind];
    const why = buildReason(g, s, kind);
    if (why) return { ok: false, reason: why };
    const cost = decreePrice(buildCost(s, kind),s),
      e = funds(g, s.owner);
    if(s.expDecree)delete s.expDecree;
    buildingFoundation(g, s);
    warSpend(g,s.owner,cost,'construction');
    spend(e, cost);
    if (kind === 'port' && !s.portAt) {
      const t = portSite(g, s);
      s.portAt = { c: t.c, r: t.r };
      s.portOwner = s.owner;
    }
    s.cityTier ??= cityTierOf(s); // Preserve city tier before any factory changes.
    s[b.field] = buildingLevel(s, kind) + 1;
    if (g.mode === 'conquest' && s.owner === g.player && g.challengePassRun?.version === 1)
      g.challengePassRun.constructed = true;
    if (kind === 'factory' || kind === 'lab') syncBuildings(g, s);
    log(g, `${s.name}: ${b.name} upgraded to level ${s[b.field]}.`, s.owner);
    return { ok: true };
  }

  // ======== Player city automation ========
  // Each automated city has one selected normal unit and makes one attempt
  // per run; automation can be rerun if a produced unit vacates the city.
  // Global Production Command settings handle building upgrades, formation size and protected reserves.
  const AUTOMATION_DEFAULT_RESERVE = { credits: 500, industry: 150, sakuradite: 25 };
  function automationState(g) {
    const a = (g.automation ||= {});
    a.enabled ??= false;
    a.autoUpgrade ??= true;
    a.stack = [1, 2, 3].includes(+a.stack) ? +a.stack : 1;
    a.reserve ||= {};
    for (const [k, v] of Object.entries(AUTOMATION_DEFAULT_RESERVE))
      if (!Number.isFinite(+a.reserve[k]) || +a.reserve[k] < 0) a.reserve[k] = v;
      else a.reserve[k] = Math.floor(+a.reserve[k]);
    a.cities ||= {};
    // Clean up the earlier policy prototype if a branch save already contains it.
    delete a.autoProduce;
    delete a.defaultPolicy;
    for (const o of Object.values(a.cities)) {
      delete o.policy;
      delete o.autoUpgrade;
      delete o.autoProduce;
      if (o.unit && !TYPES[o.unit]) delete o.unit;
    }
    return a;
  }
  function automationUnitOptions(g, s, side = g.player) {
    if (!s) return [];
    const landOwned = s.owner === side,
      portOwned = (s.portOwner || s.owner) === side,
      land = landOwned ? buildableTypes(g, side) : [],
      navy = portOwned && (s.portAt || (landOwned && portSite(g, s))) ? navalTypes(g, side) : [],
      air = landOwned && airshipType(g, side) ? [airshipType(g, side)] : [];
    return [...new Set([...land, ...navy, ...air])]
      .filter(id => {
        const t = TYPES[id];
        return t && !t.elite && (!t.campaign || Object.values(lineupOf(g, side)).includes(id)) &&
          (portBuilt(t) ? portOwned : landOwned) &&
          (t.side === side || t.side === 'neutral' || t.air || (g.buildable?.[side] || []).includes(id));
      })
      .sort((a, b) => {
        const ta = TYPES[a],
          tb = TYPES[b],
          navalA = ta.naval ? 1 : 0,
          navalB = tb.naval ? 1 : 0,
          branchOrder = { Infantry: 0, Armor: 1, Artillery: 2 };
        return (
          navalA - navalB ||
          (branchOrder[ta.branch] ?? 9) - (branchOrder[tb.branch] ?? 9) ||
          ta.tier - tb.tier ||
          ta.name.localeCompare(tb.name)
        );
      });
  }
  function cityAutomation(g, s, side = g.player) {
    const a = automationState(g),
      o = a.cities?.[s?.id] || {},
      options = automationUnitOptions(g, s, side);
    return { unit: options.includes(o.unit) ? o.unit : null };
  }
  function setCityAutomation(g, stationId, patch = {}) {
    const a = automationState(g),
      s = g.stations.find(v => v.id === +stationId);
    if (!s) return { ok: false, reason: 'Unknown city' };
    if (s.owner !== g.player && (!s.portAt || (s.portOwner || s.owner) !== g.player))
      return { ok: false, reason: 'Not your city or port' };
    const o = (a.cities[s.id] ||= {});
    if ('unit' in patch) {
      if (patch.unit == null || patch.unit === '') delete o.unit;
      else if (!automationUnitOptions(g, s).includes(patch.unit)) return { ok: false, reason: 'This city cannot queue that unit type' };
      else o.unit = patch.unit;
    }
    if (!Object.keys(o).length) delete a.cities[s.id];
    return { ok: true, settings: cityAutomation(g, s) };
  }
  function automationReserveAllows(g, side, cost = {}, reserve = null) {
    const e = funds(g, side),
      r = reserve || automationState(g).reserve;
    // A reserve only guards resources this order spends: low Sakuradite must not stop a frame that costs none.
    return ['credits', 'industry', 'science', 'sakuradite'].every(k =>
      !cost[k] || (e?.[k] || 0) - cost[k] >= (r?.[k] || 0));
  }
  function automationUpgradeOrder(g, s, unit) {
    const t = TYPES[unit];
    const order = [];
    // If a queued unit is blocked by infrastructure, upgrade that infrastructure first.
    if (portBuilt(t) && (s.portLevel || 0) < (t.port || 0)) order.push('port');
    if (t && !portBuilt(t) && s.tier < t.tier) order.push('factory');
    // Otherwise improve the economy in a predictable global order.
    order.push('refinery', 'lab', 'factory', 'port');
    return [...new Set(order)];
  }
  function emptyAutomationReport() {
    return { units: 0, upgrades: 0, spent: { credits: 0, industry: 0, science: 0, sakuradite: 0 }, entries: [] };
  }
  function addAutomationSpend(report, cost = {}) {
    for (const k of ['credits', 'industry', 'science', 'sakuradite']) report.spent[k] += cost[k] || 0;
  }
  function runCityAutomation(g, side = g.player, opts = {}) {
    const a = automationState(g),
      report = emptyAutomationReport();
    if (g.mode !== 'conquest' || g.over || g.phase !== side || (!a.enabled && !opts.force)) return report;
    const reserve = a.reserve;
    // An independent captured port may operate naval production without owning the city.
    for (const s of g.stations.filter(v => v.owner === side ||
      (v.portAt && (v.portOwner || v.owner) === side)).sort((x, y) => x.id - y.id)) {
      const unit = cityAutomation(g, s, side).unit;
      if (s.owner === side && a.autoUpgrade) {
        for (const kind of automationUpgradeOrder(g, s, unit)) {
          if (buildReason(g, s, kind)) continue;
          const cost = decreePrice(buildCost(s, kind), s);
          if (!automationReserveAllows(g, side, cost, reserve)) continue;
          const r = build(g, s.id, kind);
          if (r.ok) {
            report.upgrades++;
            addAutomationSpend(report, cost);
            report.entries.push({ kind: 'upgrade', city: s.name, building: kind, cost });
            break;
          }
        }
      }
      if (!unit || (portBuilt(TYPES[unit]) && s.portProducedTurn === g.turn)) continue;
      const t = TYPES[unit],
        producer = portBuilt(t) ? (s.portOwner || s.owner) : s.owner;
      if (producer !== side) continue;
      const stack = t.naval === 'ship' ? 1 : a.stack,
        cost = decreePrice(recruitCost(g, s, unit, stack), s.owner === side ? s : null);
      if (!automationReserveAllows(g, side, cost, reserve) || buyReason(g, s, unit, stack)) continue;
      const r = recruit(g, s.id, unit, stack);
      if (r.ok) {
        report.units++;
        addAutomationSpend(report, cost);
        report.entries.push({ kind: 'unit', city: s.name, type: unit, stack, cost });
      }
    }
    a.lastReport = { ...report, turn: g.turn };
    if (report.units || report.upgrades)
      log(
        g,
        `AUTOMATED LOGISTICS: ${report.units} unit${report.units === 1 ? '' : 's'} produced · ${report.upgrades} building upgrade${report.upgrades === 1 ? '' : 's'} · ${report.spent.credits} credits · ${report.spent.industry} industry${report.spent.science ? ` · ${report.spent.science} research` : ''}${report.spent.sakuradite ? ` · ${report.spent.sakuradite} Sakuradite` : ''} spent.`,
        side,
      );
    return report;
  }
  function bulkCityUpgrade(g, side = g.player, kind) {
    const report = emptyAutomationReport(),
      a = automationState(g);
    if (g.mode !== 'conquest') return { ...report, reason: 'Production Command is Conquest-only' };
    if (g.over || g.phase !== side) return { ...report, reason: 'Not your turn' };
    if (!BUILDINGS[kind]) return { ...report, reason: 'Unknown building' };
    for (const s of g.stations.filter(v => v.owner === side).sort((x, y) => x.id - y.id)) {
      if (buildReason(g, s, kind)) continue;
      const cost = decreePrice(buildCost(s, kind), s);
      if (!automationReserveAllows(g, side, cost, a.reserve)) continue;
      const r = build(g, s.id, kind);
      if (!r.ok) continue;
      report.upgrades++;
      addAutomationSpend(report, cost);
      report.entries.push({ kind: 'upgrade', city: s.name, building: kind, cost });
    }
    a.lastReport = { ...report, turn: g.turn, bulk: kind };
    return report;
  }

  function assign(g, id, k) {
    const u = g.units.find(u => u.id === id),
      a = COMMANDERS[k];
    const why = assignReason(g, u, k);
    if (why) return { ok: false, reason: why };
    funds(g, u.side).credits -= assignmentCost(g, k);
    const old = maxHP(u);
    u.cmd = k;
    u.personal = true;
    if (g.mode === 'conquest' && u.side === g.player && g.challengePassRun?.version === 1)
      g.challengePassRun.assigned[k] = true;
    u.cmdRank = g.roster[k].rank;
    u.hp += maxHP(u) - old;
    if (k === 'leila' || k === 'fernando') applyLeilaMoraleFloor(g);
    log(g, `${a.short} takes command of ${TYPES[u.type].short}.`, u.side);
    return { ok: true };
  }
  // A bonus shot still requires the shooter's real range and weapon class.
  function grandMastersParticipants(g, u, target) {
    if (!target || target.hp <= 0 || !foe(g, target.side, u.side)) return [];
    const eligible = w => w.hp > 0 && !atSea(g,w) &&
      w.expFarneseSlaughterTurn !== g.turn &&
      inRange(w,target,g) && hostileTarget(g,w,target);
    const leader = eligible(u) ? [u] : [];
    const allies = g.units.filter(w => w.id !== u.id && w.side === u.side &&
      dist(g,u,w) <= 2 && eligible(w));
    allies.sort((a,b) => power(g,b,target,stationAt(g,target)) -
      power(g,a,target,stationAt(g,target)) || a.id - b.id);
    return leader.concat(allies.slice(0,3));
  }
  // Command actions share cooldown/range validation; only targeted actions require a picker.
  function actionTargets(g, u) {
    if (!u || !COMMANDERS[u.cmd]?.action) return [];
    const kind = COMMANDERS[u.cmd].action.kind;
    if (kind === 'farneseWithdrawal') return g.units.some(v =>
      v.hp > 0 && v.side === u.side && dist(g,u,v) <= 3 &&
      v.hp < maxHP(v) * 0.5) ? [u] : [];
    if (kind === 'manfrediRequisition') return g.units.filter(v=>v.hp>0&&v.side===u.side&&!atSea(g,v)&&!isAir(v)&&!isShip(v)&&dist(g,u,v)<=2&&!TYPES[v.type].elite&&v.stack<3);
    if (kind === 'arkFleet') return g.stations.filter(s=>s.owner&&foe(g,s.owner,u.side));
    if (kind === 'command') return u.cmd === 'zero'
      ? commandTargets(g, u).concat(g.units.filter(v => v.hp > 0 &&
          foe(g, v.side, u.side) && !atSea(g, v) && !fx(v).geassProofMask && dist(g, u, v) <= 2 &&
          g.units.some(w=>w.hp>0&&w.id!==v.id&&w.side===v.side&&!atSea(g,w)&&inRange(v,w,g))))
      : commandTargets(g, u);
    if (kind === 'special') return newcomerTargets(g,u);
    if (kind === 'shinDeath') return g.units.filter(v=>v.hp>0&&foe(g,v.side,u.side)&&!atSea(g,v)&&!fx(v).geassProofMask&&dist(g,u,v)<=2);
    if (kind === 'maskFalls') return g.units.filter(v=>v.hp>0&&v.side===u.side&&!atSea(g,v)&&dist(g,u,v)<=3);
    if (kind === 'miguelSupply') return g.units.filter(v=>v.hp>0&&foe(g,v.side,u.side)&&!atSea(g,v)&&dist(g,u,v)<=3);
    if (kind === 'electricalOverload') return g.units.filter(v => v.hp > 0 &&
      foe(g, v.side, u.side) && !atSea(g, v) && dist(g, u, v) === 1 &&
      inRange(u, v, g) && hostileTarget(g, u, v));
    if (kind === 'freeze') return g.units.filter(v => v.hp > 0 &&
      foe(g, v.side, u.side) && !atSea(g, v) && !fx(v).geassProofMask && dist(g, u, v) <= 1);
    if (kind === 'checkmate') return g.units.filter(v => v.hp > 0 &&
      foe(g,v.side,u.side) && !atSea(g,v) && dist(g,u,v) <= 3);
    if (kind === 'grandMastersAssault') return g.units.filter(v => v.hp > 0 &&
      foe(g,v.side,u.side) && !atSea(g,v) && dist(g,u,v) <= 3 &&
      grandMastersParticipants(g,u,v).length > 0);
    if (kind === 'memoryRewrite') return g.units.filter(v => v.hp > 0 &&
      foe(g, v.side, u.side) && !atSea(g, v) && !fx(v).geassProofMask && dist(g, u, v) <= 3 &&
      v.morale > -3 && !v.expCharlesMemoryRewrite && !v.moraleWard && moraleFloor(g, v) <= -3);
    if (kind === 'geassCancel') return g.units.filter(v => v.hp > 0 &&
      foe(g,v.side,u.side) && !atSea(g,v) && !fx(v).geassProofMask && dist(g,u,v) <= 2);
    // C.C. protects one ally, not herself; all playable land formations qualify.
    if (kind === 'expendableVanguard') return g.units.filter(v => v.hp > 0 &&
      v.side === u.side && v.id !== u.id && !atSea(g, v) &&
      dist(g, u, v) <= 2 && !v.expSmilasVanguardCharge);
    if (kind === 'ohgiHeal') return g.units.filter(v => v.hp > 0 &&
      v.side === u.side && v.id !== u.id && !atSea(g, v) &&
      dist(g, u, v) <= 2 && v.hp < maxHP(v));
    if (kind === 'protection' || kind === 'oscarDefense') return g.units.filter(v => v.hp > 0 &&
      v.side === u.side && v.id !== u.id && !atSea(g, v) && dist(g, u, v) <= 2);
    // Imperial Command affects every on-map friendly formation in range,
    // including Lelouch himself. No target selection is required.
    if (kind === 'imperial') return g.units.filter(v => v.hp > 0 &&
      v.side === u.side && !atSea(g, v) && dist(g, u, v) <= 2);
    // Synchronized Maneuver refreshes Leila too, and any friendly on-map
    // formation (including naval units) that has spent its movement or attack.
    if (kind === 'withdraw') return g.units.filter(v => v.hp > 0 && v.side === u.side &&
      dist(g, u, v) <= 2 && (v.moved || v.attacked));
    return g.units.filter(v => v.hp > 0 && !atSea(g, v) && dist(g, u, v) <= (kind === 'cleanse' ? 1 : 2) &&
      (kind === 'cleanse' ? v.side === u.side : foe(g, v.side, u.side)));
  }
  // Cosmetic combat records describe strikes already resolved by the rules.
  // They never execute a second attack or spend a formation's normal actions.
  function commanderCombatRecord(attacker, target, damage, shieldDamage = 0, counter = 0) {
    return {
      attacker: { id:attacker.id, type:attacker.type, side:attacker.side, c:attacker.c, r:attacker.r },
      result: {
        ok:true, from:{c:attacker.c,r:attacker.r}, to:{c:target.c,r:target.r}, targetId:target.id,
        hit:[{id:target.id,c:target.c,r:target.r,damage:Math.max(0,damage)}],
        shieldDamage, counter:Math.max(0,counter), destroyed:target.hp<=0, crit:false,
      },
    };
  }
  // Transient presentation records stay outside saved game state. Nested
  // attacks (such as Electrical Overload) share their outer action's queue.
  const abilityVisualQueues = new WeakMap();
  function collectAbilityVisuals(g, resolve) {
    if (abilityVisualQueues.has(g)) return resolve();
    const events = [];
    abilityVisualQueues.set(g, events);
    try {
      const result = resolve();
      if (result?.ok && events.length) result.abilityEffects = events;
      return result;
    } finally { abilityVisualQueues.delete(g); }
  }
  function recordAbilityAttack(g, attacker, target, damage, crit = false) {
    const events = abilityVisualQueues.get(g);
    if (!events) return;
    const record = commanderCombatRecord(attacker,target,damage);
    record.result.crit = crit;
    events.push({kind:'attack',...record,target:{id:target.id,type:target.type,
      side:target.side,stack:target.stack,c:target.c,r:target.r}});
  }
  function recordAbilitySignal(g, target, label, kind = 'buff') {
    abilityVisualQueues.get(g)?.push({kind:'signal',to:{c:target.c,r:target.r},label,tone:kind});
  }
  function feint(g, id, targetId = null) {
    return collectAbilityVisuals(g, () => resolveFeint(g,id,targetId));
  }
  function resolveFeint(g, id, targetId = null) {
    const check = validateAction(g, { kind: 'commander', id });
    if (!check.ok) return check;
    const u = check.actor;
    const action = COMMANDERS[u.cmd].action, options = actionTargets(g, u);
    if (action.kind === 'special') return newcomerUse(g,u,targetId);
    if (action.kind === 'farneseWithdrawal') {
      let affected = 0;
      for (const ally of g.units) {
        if (ally.hp <= 0 || ally.side !== u.side || dist(g,u,ally) > 3 ||
            ally.hp >= maxHP(ally) * 0.5) continue;
        // Restore a spent move; bank a full extra move for units that have
        // not yet spent their movement. Attack availability is unchanged.
        if (ally.moved) {
          if (!heavyMoveLimitReached(g,ally)) {
            ally.moved = false; ally.withdrawMove = true;
            ally.skillReposition = 0;
          }
        } else if (!heavyMoveLimitReached(g,ally)) {
          ally.expFarneseExtraMove = Math.max(1,ally.expFarneseExtraMove||0);
        }
        ally.expFarneseWithdrawal = {side:ally.side,source:u.id};
        affected++;
      }
      if (!affected) return {ok:false,reason:'No friendly formations below 50% HP within 3 hexes'};
      u.feintCD = 0;u.feintUsedTurn = g.turn;
      log(g,'Orderly Withdrawal grants '+affected+' endangered friendly formations movement and protection.',u.side);
      return {ok:true,affected};
    }
    if(action.kind==='manfrediRequisition'){
      const v=targetId==null?options[0]:options.find(w=>w.id===targetId);
      if(!v)return {ok:false,reason:'Choose a normal friendly formation within 2 hexes with room for another frame'};
      const priorMax=maxHP(v);
      v.stack++;
      v.hp=Math.min(maxHP(v),v.hp+Math.max(1,Math.round(maxHP(v)-priorMax)));
      u.feintCD=action.cooldown;u.feintUsedTurn=g.turn;
      log(g,'Grand Master Manfredi adds a free frame to '+(COMMANDERS[v.cmd]?.short||TYPES[v.type].short)+'.',u.side);
      return {ok:true,target:v.id,affected:1,frames:1};
    }
    if (action.kind === 'arkFleet') {
      const city=targetId==null?options[0]:options.find(s=>s.id===targetId);
      if(!city)return {ok:false,reason:'Choose an enemy city'};
      city.expArkFleet={source:u.id,owner:city.owner,turns:1};
      for(const enemy of g.units)if(enemy.hp>0&&foe(g,enemy.side,u.side)&&dist(g,enemy,city)<=1){
        lowerMorale(g,enemy,1);
        enemy.expArkFleetSlow={side:enemy.side,turns:1,penalty:2};
      }
      // Independent deployment timer; ordinary Ark Fleet use never resets it.
      let airship=null;
      if(u.arkFleetAirshipCD==null)u.arkFleetAirshipCD=10;
      if(u.arkFleetAirshipCD<=0){
        const type='logres_airship';
        // Spawn next to Julius, not near the targeted city. Friendly territory
        // is not required: the airship is deployed by his fleet, not a city.
        const positions=within(g,u,2).filter(p=>{
          const d=dist(g,u,p);
          return d>=1&&d<=2&&!isSea(p)&&!TERRAIN[p.terrain]?.blocked&&
            !unitAt(g,p)&&!stationAt(g,p);
        }).sort((a,b)=>dist(g,u,a)-dist(g,u,b)||a.r-b.r||a.c-b.c);
        const p=positions[0];
        if(p){airship=newUnit(g,type,u.side,p.c,p.r,1,null,false);u.arkFleetAirshipCD=10;}
      }
      u.feintCD=0;u.feintUsedTurn=g.turn;
      log(g,'Ark Fleet spreads terror around '+city.name+(airship?' and deploys a free Logres airship':'' )+'.',u.side);
      return {ok:true,target:city.id,affected:1,airship:airship?.id};
    }
    if (action.kind === 'shinDeath') {
      const v=options.find(w=>w.id===targetId) || (targetId==null?options[0]:null);
      if(!v)return {ok:false,reason:'Choose an enemy within 2 hexes'};
      // The percentage defines base strike power, not unavoidable true damage.
      // Scale through the same attack/defense power ratio used by ordinary combat.
      const base=Math.max(1,Math.round(maxHP(v)*0.5));
      const ordinary=Math.max(0,power(g,u,v,stationAt(g,v),false,false));
      const unprotected=Math.max(1,power(g,u,null,null,false,false));
      const damage=Math.max(1,Math.round(base*ordinary/unprotected));
      const before=v.hp;
      smilasDealCombatDamage(g,v,u,damage);
      if(v.hp<=0){v.expDirectIncoming=true;kill(g,v,u,false,null,damage);delete v.expDirectIncoming;}
      u.feintCD=action.cooldown;u.feintUsedTurn=g.turn;
      log(g,'Command to Die strikes '+(COMMANDERS[v.cmd]?.short||TYPES[v.type].short)+'.',u.side);
      checkVictory(g);
      return {ok:true,target:v.id,damage:before-v.hp,
        combat:[commanderCombatRecord(u,v,before-v.hp)]};
    }
    if (action.kind === 'maskFalls') {
      if(u.side!==g.player)return {ok:false,reason:'The AI cannot use The Mask Falls'};
      const cost=Math.min(Math.max(0,u.hp-1),Math.round(maxHP(u)*0.8));
      u.hp-=cost;u.moved=true;u.attacked=true;
      for(const v of options){
        if(v.id!==u.id){
          v.moved=false;v.attacked=false;v.chain=0;v.moveActionsThisTurn=0;
          v.movedDistance=0;v.skillReposition=0;v.eliteMoveAfterKill=false;
          delete v.expKallenReposition;
        }
        // Death rally takes precedence only for units without an active Mask buff.
        if(!v.expDeathEmperor){
          v.expMaskFalls={side:u.side,source:u.id,attackReady:true};
        }
      }
      u.feintCD=0;u.feintUsedTurn=g.turn;
      log(g,'The Mask Falls: Lelouch sacrifices '+cost+' HP to rally '+options.length+' formations.',u.side);
      return {ok:true,affected:options.length,cost};
    }
    if(action.kind==='miguelSupply'){
      const v=targetId==null?[...options].sort((a,b)=>Number(!!b.cmd)-Number(!!a.cmd)||b.hp-a.hp||a.id-b.id)[0]
        :options.find(w=>w.id===targetId);
      if(!v)return {ok:false,reason:'Choose an enemy formation within 3 hexes'};
      v.expMiguelSupply={source:u.id,side:u.side};
      u.feintCD=0;u.feintUsedTurn=g.turn;
      log(g,'Miguel severs the supply lines of '+(COMMANDERS[v.cmd]?.short||TYPES[v.type].short)+'.',u.side);
      return {ok:true,target:v.id,affected:1};
    }
    if (action.kind === 'grandMastersAssault') {
      const v = targetId == null ? [...options].sort((a,b) =>
        Number(!!b.cmd)-Number(!!a.cmd) || b.hp-a.hp || a.id-b.id)[0]
        : options.find(w => w.id === targetId);
      if (!v) return { ok:false, reason:'Choose a reachable enemy formation within 3 hexes' };
      const participants = grandMastersParticipants(g,u,v);
      if (!participants.length) return { ok:false, reason:'No allied weapon can reach the target' };
      // Start cooldown before damage: allied kills may immediately shorten it.
      u.feintCD = action.cooldown;
      u.feintUsedTurn = g.turn;
      let totalDamage=0, shieldDamage=0, fired=0;
      const combat = [];
      for (const shooter of participants) {
        if (v.hp <= 0 || g.over) break;
        const shot = preview(g,shooter.id,v.c,v.r);
        if (!shot) continue;
        const city = stationAt(g,v);
        let shotShieldDamage = 0;
        if (city && foe(g,city.owner,shooter.side)) {
          city.attackedSinceRegen = true;
          const lost = Math.min(city.shield, Math.max(0,Math.round((shot.shield || 0)*.75)));
          city.shield -= lost;
          shieldDamage += lost;
          shotShieldDamage = lost;
        }
        let incoming = Math.max(1,Math.round((shot.rawUnit ?? shot.unit)*.75));
        if (v.expCCProtection) delete v.expCCProtection;
        consumeDefensiveSkills(g,v,shooter);
        incoming = newcomerIntercept(g,v,incoming);
        if (fx(v).futureSightDodge && random(g) < fx(v).futureSightDodge) {
          incoming = 0;
          recordAbilitySignal(g,v,'EVADED','guard');
          log(g,'FUTURE SIGHT — EVADED',v.side);
        }
        const prior=v.hp;
        smilasDealCombatDamage(g,v,shooter,incoming);
        const dealt=prior-v.hp;
        totalDamage += dealt;
        fired++;
        recordWarEvent(g,'damage',shooter.side,{targetSide:v.side,
          value:dealt,unitId:shooter.id,group:warGroup(g,shooter)});
        v.struck={turn:g.turn,side:shooter.side};
        if(v.skillMarks?.length)v.skillMarks=v.skillMarks.filter(m=>m.side!==shooter.side||!m.uses);
        if (v.hp <= 0) {
          v.expDirectIncoming = true;
          kill(g,v,shooter,false,null,incoming);
          delete v.expDirectIncoming;
        }
        combat.push(commanderCombatRecord(shooter,v,dealt,shotShieldDamage));
      }
      log(g,"Grand Master's Assault: "+fired+' participating shots, '+
        totalDamage+' damage and '+shieldDamage+' city-shield damage.',u.side);
      checkVictory(g);
      return {ok:true,target:v.id,affected:fired,damage:totalDamage,shieldDamage,combat};
    }
    if (action.kind === 'electricalOverload') {
      const v = targetId == null ? options[0] : options.find(w => w.id === targetId);
      if (!v) return { ok: false, reason: 'Choose an adjacent enemy in attack range' };
      // A regular damaging attack with its usual attack cost, but without
      // either normal retaliation or Senba's special defensive reaction.
      u.expFernandoOverloadStrike = true;
      let result;
      try { result = attack(g, u.id, v.c, v.r); }
      finally { delete u.expFernandoOverloadStrike; }
      if (!result.ok) return result;
      if (v.hp > 0) {
        v.expFernandoOverload = { source: u.id, side: u.side };
        v.moved = true;
      }
      u.feintCD = 0;
      u.feintUsedTurn = g.turn;
      log(g, `Electrical Overload strikes ${COMMANDERS[v.cmd]?.short || TYPES[v.type].short}.`, u.side);
      return { ok: true, target: v.id, affected: v.hp > 0 ? 1 : 0, attack: result };
    }
    if (action.kind === 'oscarDefense') {
      // Once per turn, select one allied formation, not Oscar himself.
      const v=targetId==null ? [...options].sort((a,b)=>
        Number(!!b.attacked)-Number(!!a.attacked) || a.hp/maxHP(a)-b.hp/maxHP(b) || a.id-b.id)[0]
        : options.find(w=>w.id===targetId);
      if(!v)return {ok:false,reason:'Choose an allied formation within 2 hexes'};
      // One designated defensive center per Oscar; no hidden stacking.
      for(const ally of g.units)
        if(ally.expOscarDefense?.source===u.id)delete ally.expOscarDefense;
      v.expOscarDefense={source:u.id,side:u.side};
      // Separate movement permission: 3 legal movement points, not an
      // action reset. A move consumes this free reposition first and leaves
      // the normal move and attack flags exactly as they were.
      v.expOscarReposition=true;
      u.feintCD=0;
      u.feintUsedTurn=g.turn;
      log(g,`Oscar places ${COMMANDERS[v.cmd]?.short||TYPES[v.type].short} under Emergency Defense.`,u.side);
      return {ok:true,target:v.id,affected:1,reposition:3};
    }
    if (action.kind === 'expendableVanguard') {
      const v = targetId == null ? [...options].sort((a,b) =>
        Number(!!b.cmd)-Number(!!a.cmd) ||
        Number(!!b.attacked)-Number(!!a.attacked) ||
        TYPES[b.type].attack*(b.stack||1)-TYPES[a.type].attack*(a.stack||1) ||
        a.id-b.id)[0] : options.find(v => v.id === targetId);
      if (!v) return { ok:false, reason:'Choose an eligible allied formation within 2 hexes' };
      // A full activation is granted even if both actions were still unused.
      // Keep once-per-turn special stamps and heavy-frame move caps intact.
      v.moved = false;
      v.attacked = false;
      v.chain = 0;
      v.skillReposition = 0;
      v.withdrawMove = false;
      v.eliteMoveAfterKill = false;
      delete v.expAkitoMoveReady;
      v.expSmilasVanguardCharge = true;
      u.feintCD = action.cooldown ?? 3;
      u.feintUsedTurn = g.turn;
      log(g, `Expendable Vanguard sends ${COMMANDERS[v.cmd]?.short||TYPES[v.type].short} forward at a deadly cost.`,u.side);
      return { ok:true, target:v.id, affected:1 };
    }
    if (action.kind === 'ohgiHeal') {
      const cost = Math.max(1, Math.round(maxHP(u) * 0.20));
      if (u.hp <= cost) return { ok: false, reason: 'Ohgi must survive the HP sacrifice' };
      const v = targetId == null ? [...options].sort((a, b) =>
        Math.min(maxHP(b) * 0.30, maxHP(b) - b.hp) -
        Math.min(maxHP(a) * 0.30, maxHP(a) - a.hp) ||
        Number(!!b.cmd) - Number(!!a.cmd) || a.id - b.id)[0]
        : options.find(w => w.id === targetId);
      if (!v) return { ok: false, reason: 'Choose a wounded ally within 2 hexes' };
      const healed = Math.min(maxHP(v) - v.hp, Math.max(1, Math.round(maxHP(v) * 0.30)));
      v.hp += healed;
      u.hp -= cost;
      u.feintCD = action.cooldown ?? 3;
      u.feintUsedTurn = g.turn;
      log(g, `Ohgi sacrifices ${cost} HP to heal ${COMMANDERS[v.cmd]?.short || TYPES[v.type].short} for ${healed} HP.`, u.side);
      return { ok: true, target: v.id, affected: 1, healed, cost };
    }
    if (action.kind === 'memoryRewrite') {
      const v = targetId == null ? [...options].sort((a, b) =>
        Number(!!b.cmd) - Number(!!a.cmd) ||
        TYPES[b.type].attack * (b.stack || 1) - TYPES[a.type].attack * (a.stack || 1) ||
        a.id - b.id)[0] : options.find(w => w.id === targetId);
      if (!v) return { ok: false, reason: 'Choose a vulnerable enemy formation within 3 hexes' };
      // Confused is the existing -3 morale state. Remember the previous
      // morale so this single-turn rewrite does not cause lasting penalties.
      v.expCharlesMemoryRewrite = { source: u.id, previousMorale: v.morale };
      v.morale = -3;
      u.feintCD = action.cooldown ?? 3;
      u.feintUsedTurn = g.turn;
      log(g, `Charles rewrites the memories of ${COMMANDERS[v.cmd]?.short || TYPES[v.type].short}: Confused until its next turn ends.`, u.side);
      return { ok: true, target: v.id, affected: 1 };
    }
    if (action.kind === 'checkmate') {
      const v = targetId == null ? [...options].sort((a,b) =>
        Number(!!b.cmd)-Number(!!a.cmd) || b.hp-a.hp || a.id-b.id)[0]
        : options.find(w => w.id === targetId);
      if (!v) return {ok:false,reason:'Choose an enemy formation within 3 hexes'};
      // A commander may designate only one formation at a time.
      for (const other of g.units)
        if (other.expCheckmate?.source === u.id) delete other.expCheckmate;
      v.expCheckmate = {source:u.id,side:u.side};
      u.feintCD = 0;
      u.feintUsedTurn = g.turn;
      log(g,`Checkmate: ${COMMANDERS[v.cmd]?.short||TYPES[v.type].short} is marked for concentrated fire.`,u.side);
      return {ok:true,target:v.id,affected:1};
    }
    if (action.kind === 'freeze') {
      // Freeze everybody before the health cost; even a lethal activation works.
      for (const v of options) {
        v.expRoloFrozen = { source: u.id, side: u.side };
        v.moved = true;
        v.attacked = true;
      }
      const cost = Math.max(1, Math.round(maxHP(u) * 0.10));
      u.hp = Math.max(0, u.hp - cost);
      u.feintCD = action.cooldown ?? 3;
      u.feintUsedTurn = g.turn;
      log(g, `Rolo's Absolute Suspension freezes ${options.length} enemy formations and costs ${cost} HP.`, u.side);
      if (u.hp <= 0) kill(g, u, null, true);
      checkVictory(g);
      return { ok: true, affected: options.length, cost, selfDestructed: u.hp <= 0 };
    }
    if (action.kind === 'geassCancel') {
      const v = targetId == null ? [...options].sort((a,b) =>
        Number(!!b.cmd)-Number(!!a.cmd) || b.hp-a.hp || a.id-b.id)[0]
        : options.find(w => w.id === targetId);
      if (!v) return {ok:false,reason:'Choose an enemy formation within 2 hexes'};
      // Dispel temporary commander-granted benefits, not intrinsic skills,
      // research or elite upgrades. Also lift Geass-induced negative statuses.
      for (const key of ['expImperialCommand','expCCProtection','expKallenReposition',
        'expKallenMoveBank','expAkitoMoveReady','expZeroGeass']) {
        if (key === 'expZeroGeass' && v.expZeroGeass)
          v.morale = Math.min(1,v.morale+(v.expZeroGeass.moraleLost||0));
        delete v[key];
      }
      // Suppress active use through the target's next faction activation.
      v.expGeassCancelled = true;
      u.feintCD = 0;
      u.feintUsedTurn = g.turn;
      log(g,`Geass Canceller disrupts ${COMMANDERS[v.cmd]?.short||TYPES[v.type].short}.`,u.side);
      return {ok:true,target:v.id,affected:1};
    }
    if (action.kind === 'protection') {
      // Player picks an ally explicitly; the AI chooses the ally most in need.
      const v = targetId == null ? [...options].sort((a, b) =>
        (maxHP(b) - b.hp) - (maxHP(a) - a.hp) ||
        a.morale - b.morale || a.id - b.id)[0] : options.find(w => w.id === targetId);
      if (!v) return { ok: false, reason: 'Choose an eligible ally within 2 hexes' };
      v.hp = Math.min(maxHP(v), v.hp + Math.round(maxHP(v) * 0.15));
      v.morale = Math.max(0, v.morale);
      // One-hit protection is not additive: recasting replaces its charge.
      v.expCCProtection = { source: u.id, side: u.side };
      u.feintCD = 0;
      u.feintUsedTurn = g.turn;
      log(g, `C.C.'s Protection shields ${COMMANDERS[v.cmd]?.short || TYPES[v.type].short}.`, u.side);
      return { ok: true, target: v.id, affected: 1 };
    }
    if (action.kind === 'imperial') {
      // One-turn support; repeated commanders refresh rather than stack.
      for (const v of options) {
        v.morale = 1;
        v.expImperialCommand = { side: u.side, source: u.id, turn: g.turn };
      }
      u.feintCD = 0;
      u.feintUsedTurn = g.turn;
      log(g, `Imperial Command strengthens ${options.length} friendly formations.`, u.side);
      return { ok: true, affected: options.length };
    }
    if (['command', 'designate', 'stratagem'].includes(action.kind)) {
      const combat = [];
      const v = targetId == null ? options.sort((a, b) => TYPES[b.type].attack * b.stack - TYPES[a.type].attack * a.stack || a.id - b.id)[0]
        : options.find(v => v.id === targetId);
      if (!v) return { ok: false, reason: 'Choose a valid target within 2 hexes' };
      if (action.kind === 'command' && u.cmd === 'zero' && foe(g, v.side, u.side)) {
        const victims=g.units.filter(w=>w.hp>0&&w.id!==v.id&&w.side===v.side&&!atSea(g,w)&&inRange(v,w,g))
          .sort((a,b)=>dist(g,v,a)-dist(g,v,b)||a.id-b.id);
        const w=victims[0];
        if(!w)return {ok:false,reason:'No same-faction formation within weapon range'};
        // A compelled strike bypasses hostility checks but not armor, terrain,
        // city cover or ordinary counterfire eligibility.
        const hit=Math.max(1,Math.round(power(g,v,w,stationAt(g,w),false,false)));
        const priorTargetHP = w.hp, priorAttackerHP = v.hp;
        smilasDealCombatDamage(g,w,v,hit);
        if(w.hp<=0){w.expDirectIncoming=true;kill(g,w,v,false,null,hit);delete w.expDirectIncoming;}
        if(v.hp>0&&w.hp>0&&inRange(w,v,g)&&w.morale>-3&&!w.expRoloFrozen&&!w.expCharlesMemoryRewrite){
          const retaliation=Math.max(1,Math.round(power(g,w,v,stationAt(g,v),true,false)));
          smilasDealCombatDamage(g,v,w,retaliation);
          if(v.hp<=0){v.expDirectIncoming=true;kill(g,v,w,false,null,retaliation);delete v.expDirectIncoming;}
        }
        combat.push(commanderCombatRecord(v,w,priorTargetHP-w.hp,0,priorAttackerHP-v.hp));
        checkVictory(g);
      } else if (action.kind === 'command') {
        v.moved = v.attacked = false;
        v.chain = 0;
        v.moveActionTurn = g.turn;
        v.moveActionsThisTurn = 0;
        v.movedDistance = 0;
        v.eliteMoveAfterKill = false;
        v.skillReposition = 0;
        v.withdrawMove = false;
        // Kallen's earned movement bank stays, but a pending limited
        // reposition must not override the full movement Zero restores.
        delete v.expKallenReposition;
        // A new normal activation, NOT a new turn: retain once-per-turn
        // stamps for commander skills, Elite perks and other special effects.
      } else addTargetMark(v, { side: u.side, source: u.id, range: 2,
        damage: action.kind === 'designate' ? 0.2 : 0, noCounter: action.kind === 'stratagem' });
      u.feintCD = COMMANDERS[u.cmd].action.cooldown ?? 3;
      u.feintUsedTurn = g.turn;
      log(g, `${action.verb}: ${COMMANDERS[v.cmd]?.short || TYPES[v.type].short} ${action.kind === 'command' ? (v.side === u.side ? 'receives a fresh activation' : 'is forced to attack its own faction') : 'designated'}.`, u.side);
      return { ok: true, target: v.id, affected: 1, combat };
    }
    for (const v of options) {
      if (action.kind === 'cleanse') { v.morale = Math.max(0, v.morale); v.moraleWard = { side: u.side }; }
      else if (action.kind === 'withdraw') {
        v.moved = false;
        v.withdrawMove = true; // One FULL legal movement action, including after attacking.
        v.skillReposition = 0;
        // Never restore v.attacked: the command does not grant additional attacks.
      }
      else lowerMorale(g, v, 2);
    }
    u.feintCD = COMMANDERS[u.cmd].action.cooldown ?? 3;
    u.feintUsedTurn = g.turn;
    log(g, `${action.verb} affects ${options.length} units.`, u.side);
    return { ok: true, affected: options.length };
  }


  // ======== Black Knights and JLF commanders: allegiance and signature skills ========
  // Only the 2018 U.F.N. coalition may field Black Knights and JLF
  // commanders; the 2017 Chinese Federation is independent.
  const ALLIES = { bk: ['jlf'] };
  function serves(k, side, g = null) {
    const a = COMMANDERS[k];
    // Alternate history governments have scenario-local officer allegiances.
    // Their commander pools must not leak into regular world/campaign modes.
    const scenarioRoster = g?.mode === 'scenario' ? g.scenario?.commanderSides?.[side] : null;
    if (Array.isArray(scenarioRoster)) return !!a && scenarioRoster.includes(k);
    // In 2018, former E.U. and Black Knights commanders serve the united federation.
    if (g?.mode === 'conquest' && g.era === 'ufn2018') {
      if (side === 'cf' && ['cf', 'eu', 'bk', 'jlf'].includes(a?.side)) return true;
      if (side === 'eb' && ['eb', 'britannia'].includes(a?.side)) return true;
    }
    // Euro Britannia's Story campaign uses the shared imperial HQ pool.
    // Britannia continues to use these officers as before.
    if (side === 'eb' && ['julius', 'shin', 'ashley', 'manfredi', 'farnese', 'augustus', 'kewell', 'norland', 'gaudefroy', 'raymond', 'suzaku', 'schneizel', 'oldrin', 'marrybell'].includes(k))
      return true;
    return !!a && (a.side === side || !!ALLIES[side]?.includes(a.side));
  }
  // Geass Absolute Command: a fresh activation for ONE acted ally (never
  // Zero himself), or disruption of ONE enemy. Both share the same use stamp.
  function commandTargets(g, u) {
    return g.units.filter(
      v => v.hp > 0 && v.side === u.side && v.id !== u.id &&
        (v.moved || v.attacked || v.chain > 0) && !atSea(g, v) &&
        v.morale > -3 && dist(g, u, v) <= 2,
    );
  }
  const skillNear = (g, u, flag, range, self = true) =>
    g.units.some(
      v => v.hp > 0 && !atSea(g, v) && v.side === u.side && (self || v.id !== u.id) && (!u.expInformationWarfare || v.id===u.id) && fx(v)[flag] &&
        (!v.auraDisrupted || !['orderCommander', 'screenedFire'].includes(flag)) && dist(g, v, u) <= range,
    );
  // Akito's Brain Raid requires complete separation from friendly AND
  // campaign-team allied formations, counting ships and the full two-hex ring.
  function akitoIsolated(g, u) {
    return !g.units.some(v => v.hp > 0 && v.id !== u.id &&
      !foe(g, v.side, u.side) && dist(g, u, v) <= 2);
  }
  function friendlyNeighbors(g, u, predicate = () => true) {
    return g.units.filter(v => v.hp > 0 && !atSea(g, v) && v.side === u.side && v.id !== u.id && dist(g, u, v) === 1 && predicate(v));
  }
  function lowerMorale(g, u, steps) {
    if (!u.moraleWard) u.morale = Math.max(romeroStandFast(g,u) ? 0 : moraleFloor(g,u), u.morale - steps);
  }
  function addTargetMark(u, mark) {
    u.skillMarks ||= [];
    u.skillMarks = u.skillMarks.filter(m => !(m.side === mark.side && m.source === mark.source));
    u.skillMarks.push(mark);
  }
  function targetMarked(g, attacker, victim) {
    let damage = 0, noCounter = false;
    for (const m of victim?.skillMarks || []) {
      if (m.side !== attacker.side || (m.ryo && m.source === attacker.id)) continue;
      if (m.range) {
        const source = g.units.find(v => v.id === m.source && v.hp > 0 && !atSea(g, v));
        if (!source || dist(g, source, attacker) > m.range) continue;
      }
      damage = Math.max(damage, m.damage || 0);
      noCounter ||= !!m.noCounter;
    }
    return { damage, noCounter };
  }
  function armorPenetration(g, u, target = null) {
    const engineering = skillNear(g, u, 'engineeringPen', 2) ? 0.1 : 0,
      harken = target && TYPES[u.type].branch === 'Infantry' && TYPES[target.type].branch === 'Armor'
        ? techValue(g, u.side, 'infantry.harken')
        : 0;
    return clamp(TYPES[u.type].pen + (fx(u).pen || 0) + (eliteFx(u).pen || 0) + engineering + harken, 0, 0.95);
  }
  // Exploit Weakness: at least two surviving, deployed friendly formations
  // must be adjacent to this specific enemy. The initiating attacker must be
  // Xianglin herself or a friendly formation within 2 hexes of a living
  // Xianglin. This is one non-stacking positional bonus, never a status mark.
  function xianglinExploitWeakness(g, attacker, target) {
    if (!attacker || !target || attacker.hp <= 0 || target.hp <= 0 ||
        !foe(g, attacker.side, target.side) || atSea(g, attacker)) return false;
    const nearby = g.units.filter(v =>
      v.hp > 0 && v.side === attacker.side && !atSea(g, v) &&
      dist(g, v, target) === 1);
    if (nearby.length < 2) return false;
    return g.units.some(v =>
      v.hp > 0 && v.side === attacker.side && v.cmd === 'xianglin' &&
      !atSea(g, v) && !v.auraDisrupted && fx(v).exploitWeakness &&
      dist(g, v, attacker) <= 2 &&
      (!attacker.expInformationWarfare || v.id === attacker.id));
  }
  // Selfless Leadership is active only while Ohgi is alive and STRICTLY
  // below half HP. Every qualifying ally, including Ohgi, receives one
  // non-stacking offensive/defensive bonus while within two hexes.
  function ohgiSelflessLeadership(g, recipient) {
    if (!recipient || recipient.hp <= 0) return false;
    return g.units.some(v => v.hp > 0 && v.side === recipient.side &&
      v.cmd === 'ohgi' && !atSea(g, v) && !v.auraDisrupted &&
      fx(v).selflessLeadership && v.hp < maxHP(v) * 0.5 &&
      dist(g, v, recipient) <= 2 &&
      (!recipient.expInformationWarfare || v.id === recipient.id));
  }
  // All three Smilas mechanics are combat-state rules. They have no buff
  // stamps, and never read the player's faction to decide whether to work.
  function smilasAdjacentShield(g, target) {
    return !!target && target.hp > 0 && !atSea(g,target) &&
      g.units.some(v => v.hp > 0 && v.id !== target.id &&
        v.side === target.side && v.cmd === 'smilas' &&
        !atSea(g,v) && !v.auraDisrupted && fx(v).humanShields &&
        dist(g,v,target) === 1);
  }
  function smilasDamageShare(g, victim, amount) {
    const raw = Math.max(0, Math.round(amount || 0));
    if (!victim || victim.hp <= 0 || victim.cmd !== 'smilas' ||
        !fx(victim).humanShields || atSea(g,victim) || victim.auraDisrupted)
      return { primary:raw, ally:null, redirected:0 };
    const ally = g.units.filter(v => v.hp > 0 && v.id !== victim.id &&
      v.side === victim.side && !atSea(g,v) &&
      dist(g,v,victim) === 1)
      .sort((a,b) => b.hp-a.hp || a.id-b.id)[0];
    if (!ally) return { primary:raw, ally:null, redirected:0 };
    const share = Math.round(raw * 0.50);
    // Transfer damage bypasses the recipient's own defenses, but suffers
    // the 15% Human Shields adjacency vulnerability exactly once.
    return { primary:raw-share, ally, redirected:Math.round(share*115/100) };
  }
  function smilasSacrificeReflection(g, victim, attacker, incomingDamage) {
    if (!victim || !attacker || !foe(g,victim.side,attacker.side) ||
        !Number.isFinite(incomingDamage) || incomingDamage <= 0) return 0;
    const supports = g.units.some(v => v.cmd === 'smilas' &&
      v.side === victim.side && (v.hp > 0 || v.id === victim.id) &&
      !atSea(g,v) && !v.auraDisrupted && fx(v).necessarySacrifices &&
      dist(g,v,victim) <= 2);
    return supports ? Math.round(incomingDamage*0.50) : 0;
  }
  function commanderAttack(g, u, target, counter) {
    const f = fx(u), friends = friendlyNeighbors(g, u);
    let mult = 1;
    // Brain Raid strengthens both direct attacks and counter-fire.
    if (f.loneRaider && akitoIsolated(g, u)) mult *= 1 + f.loneRaider;
    // Divine Tiger targets commander-led formations in both attacks and retaliation.
    if (f.duelist && target?.cmd) mult *= 1 + f.duelist;
    // Spearhead's Retribution applies to Guilford's initiating attacks and counterfire.
    if (f.guilfordRetaliation && target && u.expGuilfordRetribution?.targets?.includes(target.id))
      mult *= 1 + f.guilfordRetaliation;
    if (counter) {
      // Ryo's marked-target bonus also helps allied counterfire. His own
      // formation is excluded by targetMarked(), as on normal attacks.
      return mult * (target ? 1 + targetMarked(g, u, target).damage : 1);
    }
    if (u.assaultInspired) mult *= 1 + u.assaultInspired.value;
    if (!u.cmd && skillNear(g, u, 'commanderlessAttack', 2, false)) mult *= 1.15;
    if (TYPES[u.type].branch === 'Infantry' && skillNear(g, u, 'infantryCommand', 1, false)) mult *= 1.15;
    if (TYPES[u.type].branch === 'Armor' && skillNear(g, u, 'armorLeader', 2, false)) mult *= 1.1;
    if (f.runCharge && u.movedDistance >= 2) mult *= 1 + f.runCharge;
    if (f.combinedArms && friends.some(v => TYPES[v.type].branch !== TYPES[u.type].branch)) mult *= 1 + f.combinedArms;
    if (TYPES[u.type].branch === 'Armor' && friends.some(v => TYPES[v.type].branch === 'Armor') && skillNear(g, u, 'orderCommander', 1)) mult *= 1.15;
    if (!target) return mult;
    if (f.intrigueBonus && target.cmd) mult *= 1 + f.intrigueBonus;
    if (f.finisher && target.hp / maxHP(target) < 0.5) mult *= 1 + f.finisher;
    if (f.opener && target.hp >= maxHP(target)) mult *= 1 + f.opener;
    if (f.pursuit && target.lastTurnMoved) mult *= 1 + f.pursuit;
    mult *= 1 + targetMarked(g, u, target).damage;
    if (rangeOf(g, u).max > 1 && skillNear(g, u, 'screenedFire', 2) &&
      g.units.some(v => v.hp > 0 && !atSea(g, v) && v.side === u.side && TYPES[v.type].branch !== 'Artillery' && dist(g, v, target) === 1)) mult *= 1.2;
    return mult;
  }
  function bodyguarded(g, target) {
    return !!target.cmd && skillNear(g, target, 'bodyguard', 1, false);
  }
  function commanderDefense(g, target, attacker, counter, direct) {
    const f = fx(target), friends = friendlyNeighbors(g, target),
      guarded = bodyguarded(g, target);
    let mult = guarded ? 0.9 : 1;

    if (f.duelistTaken && attacker?.cmd) mult *= f.duelistTaken;
    if (counter && f.swordplay && dist(g, attacker, target) === 1) mult *= f.swordplay;

      if (f.foresight && target.foresightTurn !== g.turn) mult *= f.foresight;
      if (guarded && target.bodyguardTurn !== g.turn) mult *= 0.8;
    }
    return mult;

  function afterCommanderAttack(g, a, d, city, destroyed, defensesDestroyed = false) {
    const f = fx(a);
    if (d?.hp > 0) {
      // One persistent mark per Ryo/target, created by his first surviving hit.
      // Subsequent attacks cannot stack or strengthen it; ordinary target-mark
      // expiry at this faction's next turn handles both player and AI battles.
      if (f.ryoMark && !d.skillMarks?.some(m => m.ryo && m.side === a.side && m.source === a.id))
        addTargetMark(d, { side: a.side, source: a.id, damage: f.ryoMark, ryo: true });
      if (f.targetMark) addTargetMark(d, { side: a.side, source: a.id, damage: f.targetMark });
      if (f.intrigue && d.cmd) d.auraDisrupted = { side: a.side };
    }
    if (city && foe(g, city.owner, a.side)) {
      if (f.focusBombard) {
        const old = a.focusCity;
        const count = old?.city === city.id && old.lastTurn >= g.turn - 1 ? Math.min(3, old.count + 1) : 0;
        a.focusCity = { city: city.id, count, lastTurn: g.turn };
      }
      if (f.siegeMark && a.siegeMarkTurn !== g.turn) {
        city.bombardMark = { side: a.side, value: f.siegeMark };
        a.siegeMarkTurn = g.turn;
      }
    } else if (f.focusBombard) delete a.focusCity;
    if (a.hp <= 0) return;
    // Cornelia's recovery rewards either a frame kill or a fresh city-defense
    // break. A single volley that achieves both only heals once.
    if (f.destroyHeal && (destroyed || defensesDestroyed))
      a.hp = Math.min(maxHP(a), a.hp + Math.round(maxHP(a) * f.destroyHeal));
    if ((f.adaptiveBreakthrough&&a.movedDistance>=3||f.highland&&tile(g,a.c,a.r)?.terrain==='mountain'||f.blackSeaFox&&tile(g,a.c,a.r)?.terrain==='mountain'||f.swordplay&&d&&dist(g,a,d)===1)&&a.repositionTurn!==g.turn){a.repositionTurn=g.turn;grantReposition(a,2);}
    if (f.reposition && a.repositionTurn !== g.turn) { a.repositionTurn = g.turn; grantReposition(a, f.reposition); }
    // City defense breakthroughs share the kill-based movement reward, but
    // do not trigger unrelated kill-only healing or inspiration effects.
    if (destroyed || defensesDestroyed) {
      if (f.killReposition && a.killRepositionTurn !== g.turn) {
        a.killRepositionTurn = g.turn;
        grantReposition(a, f.killReposition);
      }
    }
    if (!destroyed) return;
    if (f.killHeal && a.killHealTurn !== g.turn) { a.killHealTurn = g.turn; a.hp = Math.min(maxHP(a), a.hp + Math.round(maxHP(a) * f.killHeal)); }

        v.assaultInspired = { side: a.side, value: Math.max(v.assaultInspired?.value || 0, f.assaultLeader) };
      }
    }
  }

  // Oscar fires an independent full-strength reaction after an enemy's
  // initiating attack against ANOTHER allied formation within 2 hexes.

  function oscarGuardCounterstrike(g,attacker,defender){
    if(!attacker||attacker.hp<=0||!defender||!foe(g,attacker.side,defender.side))return [];
    const guards=g.units.filter(v=>v.hp>0&&fx(v).guardCounterstrike&&

      dist(g,v,defender)<=2&&inRange(v,attacker,g)&&hostileTarget(g,v,attacker));
    const strikes=[];
    for(const oscar of guards){
      if(attacker.hp<=0)break;
      // A distinct enemy can trigger once per Oscar per global turn.
      if(oscar.expOscarCounterTargets?.turn!==g.turn)
        oscar.expOscarCounterTargets={turn:g.turn,ids:[]};
      if(oscar.expOscarCounterTargets.ids.includes(attacker.id))continue;
      oscar.expOscarCounterTargets.ids.push(attacker.id);
      const station=stationAt(g,attacker),pr=preview(g,oscar.id,attacker.c,attacker.r);
      if(!pr)continue;
      const crit=random(g)<pr.crit;
      const damage=Math.max(1,Math.round(pr.rawUnit*(.92+random(g)*.16)*
        (crit?pr.critMult:1)));
      if(attacker.expCCProtection)delete attacker.expCCProtection;
      let incoming=newcomerIntercept(g,attacker,damage);
      if(fx(attacker).futureSightDodge&&random(g)<fx(attacker).futureSightDodge){
        incoming=0;
        recordAbilitySignal(g,attacker,'EVADED','guard');
        log(g,'FUTURE SIGHT — EVADED',attacker.side);
      }
      const before=attacker.hp;
      const dealt=smilasDealCombatDamage(g,attacker,oscar,incoming);
      recordWarEvent(g,'damage',oscar.side,{targetSide:attacker.side,
        value:before-attacker.hp,unitId:oscar.id,group:warGroup(g,oscar)});
      attacker.struck={turn:g.turn,side:oscar.side};
      if(attacker.skillMarks?.length)
        attacker.skillMarks=attacker.skillMarks.filter(m=>m.side!==oscar.side||!m.uses);
      if(attacker.hp<=0){
        attacker.expDirectIncoming=true;
        kill(g,attacker,oscar,false,null,incoming);
        delete attacker.expDirectIncoming;
      }
      log(g,'Oscar counterstrikes '+(COMMANDERS[attacker.cmd]?.short||TYPES[attacker.type].short)+
        ' for '+dealt+' damage; no counterfire.',oscar.side);
      strikes.push({oscar:oscar.id,target:attacker.id,damage:dealt});

    }

  }

  // If Toru falls on the enemy turn, bank another legal action for allies'

  function toruLastPromise(g,toru,attacker){
    if(toru.cmd!=='toru'||!fx(toru).lastPromise||toru.expToruPromiseUsed||
       !attacker||!foe(g,toru.side,attacker.side))return;
    toru.expToruPromiseUsed=true;

         atSea(g,ally)||dist(g,toru,ally)>2)continue;

      recordAbilitySignal(g,ally,'THE LAST PROMISE');
      ally.expToruPromiseDamage=true;
      if(!immediate){
        ally.expToruPromisePending=true;
        continue;

      // Previously exhausted actions refresh immediately; actions not yet
      // spent receive a banked extra action after their first use.
      if(ally.attacked)ally.attacked=false;
      else ally.expToruExtraAttack=Math.max(ally.expToruExtraAttack||0,1);
      if(ally.moved)ally.moved=heavyMoveLimitReached(g,ally);
      else if(!heavyMoveLimitReached(g,ally))
        ally.expToruExtraMove=Math.max(ally.expToruExtraMove||0,1);
      ally.skillReposition=0;
      ally.withdrawMove=false;
    }
    log(g,"The Last Promise: Toru's sacrifice rallies nearby allies.",toru.side);

      const gained=Math.min(20,60-(u.expSalvageIndustry||0));funds(g,u.side).industry+=Math.max(0,gained);u.expSalvageIndustry=(u.expSalvageIndustry||0)+Math.max(0,gained);
      u.expSalvageCount=(u.expSalvageCount||0)+1;
      if(u.expSalvageCount%3===0)u.hp=Math.min(maxHP(u),u.hp+Math.round(maxHP(u)*.1));
      break;
    }
  }

  // ======== Sakuradite: the fourth resource, mined at a handful of deposits ========
  // Japan holds 70 of the world's 120 base output, and its output is allocated internationally (episode 8): the power
  // controlling a Japanese deposit keeps 60% and every other surviving major power receives 20%. The split, outputs,
  // extraction rates, the starting stockpile and prices are balance values, not canon quantities.
  const SAKURADITE = {
    start: 100,
    extraction: [0.25, 0.5, 0.75, 1], // share of a deposit's output by refinery level 0–3
    exportCredits: 15, // a level-3 refinery also exports for credits
    // Per frame, by class; basic classes need none (credits and industry cover them). Extra frames follow the 85% rule.
    cost: { raider: 2, medium: 3, rocket: 3, heavy: 5, siege: 8, super: 10 },
    // Elite Forces: fixed Sakuradite price by class, regardless of rarity or Elite level.
    elite: { raider: 10, medium: 12, heavy: 15, super: 20, rocket: 14, siege: 18, amphib: 12 },
    allocation: { sites: ['Mount Fuji', 'Hokkaido', 'Kyushu'], share: 0.2 }, // each other power's share of Japan
    national: 5, // every surviving major power's own supply a turn, tied to no deposit (Conquest only)
  };
  // Stockpiles and deposits for a new game.
  function setupSakuradite(g) {
    for (const [side, e] of Object.entries(g.economy)) e.sakuradite ??= isMajor(g, side) ? SAKURADITE.start : 0;
    if (g.sites) return g;
    g.sites = [];
    for (const [name, lon, lat, base, level, terrain] of RESOURCE_SITES) {
      const h = hexOf(lon, lat),
        t = tile(g, h.c, h.r),
        city = t && stationAt(g, t),
        id = g.sites.length;
      if (!t) continue;
      if (city) {
        city.refinery = Math.max(city.refinery || 0, level);
        g.sites.push({ id, name, c: city.c, r: city.r, base, city: city.id });
        continue;
      }
      const open = n => !navigable(n) && !TERRAIN[n.terrain]?.blocked && !stationAt(g, n),
        at = open(t) ? t : nearest(g, t, open);
      if (!at) continue;
      if (terrain) setTileTerrain(g, at, terrain);
      g.sites.push({ id, name, c: at.c, r: at.r, base, city: null, owner: at.owner || 'neutral', refinery: level });
    }
    return g;
  }
  // A deposit's refinery and owner live on its city, or on the mine itself.
  function depositHost(g, d) {
    return d.city == null ? d : g.stations.find(s => s.id === d.city) || null;
  }
  function depositOwner(g, d) {
    return depositHost(g, d)?.owner || null;
  }
  function depositOf(g, s) {
    return (s && g.sites?.find(d => d.city === s.id)) || null;
  }
  function siteAt(g, p) {
    return (p && g.sites?.find(d => d.city == null && d.c === p.c && d.r === p.r)) || null;
  }
  function depositYield(g, d) {
    const host = depositHost(g, d),
      level = clamp(host?.refinery || 0, 0, 3),
      rate = host ? SAKURADITE.extraction[level] : 0;
    return {
      level,
      rate,
      sakuradite: Math.round(d.base * rate),
      credits: rate && level >= 3 ? SAKURADITE.exportCredits : 0,
    };
  }
  // Who receives a deposit's extraction this turn: its owner, or for a Japanese deposit held by a major power, the
  // international allocation. Shares of powers that have surrendered stay with the controller. Unrounded.
  function depositShares(g, d) {
    const owner = depositOwner(g, d),
      out = depositYield(g, d).sakuradite,
      a = SAKURADITE.allocation;
    if (!owner || !out) return {};
    if (g.mode === 'campaign' || !a.sites.includes(d.name) || !alive(g, owner)) return { [owner]: out };
    const others = (g.mode === 'scenario' ? g.order : powersOf(g)).filter(s => s !== owner && alive(g, s)),
      shares = { [owner]: out * (1 - a.share * others.length) };
    for (const s of others) shares[s] = out * a.share;
    return shares;
  }
  function spend(e, cost) {
    e.credits -= cost.credits || 0;
    e.industry -= cost.industry || 0;
    if (cost.science) e.science -= cost.science;
    if (cost.sakuradite) e.sakuradite = (e.sakuradite || 0) - cost.sakuradite;
  }
  // A city's output per turn, with the deposit it works.
  function cityOutput(g, s) {
    const stationed = g.units.find(u => u.hp > 0 && u.side === s.owner && u.cmd && u.c === s.c && u.r === s.r),
      bonus = id => stationed ? 1 + 0.04 * genericLevel(g, stationed, id) : 1;
    return {
      credits: Math.round(s.income * treasuryBonus(g, s) * bonus('economic_expert')),
      industry: s.industry * (1 + techValue(g, s.owner, 'cities.industry')) * bonus('industrial_expert'),
      science: s.science * bonus('technology_expert'),
    };
  }
  function cityYield(g, s) {
    const d = depositOf(g, s),
      y = d ? depositYield(g, d) : { sakuradite: 0, credits: 0 },
      output = cityOutput(g, s),
      local = s.divisionIncome || {},
      regional = s.regionalIncome || {};
    return {
      credits: output.credits + y.credits + (local.credits || 0) + (regional.credits || 0),
      industry: Math.round(output.industry + (local.industry || 0) + (regional.industry || 0)),
      science: Math.round(output.science + (local.science || 0) + (regional.science || 0)),
      sakuradite: y.sakuradite + (local.sakuradite || 0) + (regional.sakuradite || 0),
    };
  }
  // Infantry or Armor moving onto a mine seizes it; it has no defenses.
  function seizeDeposit(g, u, p) {
    const d = siteAt(g, p);
    if (!d || !foe(g, d.owner, u.side) || !canCapture(u)) return null;
    const loser = d.owner;
    d.owner = u.side;
    log(
      g,
      `${COMMANDERS[u.cmd]?.short || TYPES[u.type].short} seizes the ${d.name} Sakuradite mine${FACTIONS[loser] && loser !== 'neutral' ? ' from the ' + FACTIONS[loser].short : ''}.`,
      u.side,
    );
    return d.name;
  }
  // Conquest capitulation transfers only mine ownership, not stored Sakuradite.
  function annexDeposits(g, loser, winner) {
    for (const d of g.sites || []) if (d.city == null && d.owner === loser) d.owner = winner;
    if (g.mode === 'conquest') return;
    // Preserve the pre-existing independent-scenario surrender reward.
    const e = funds(g, loser), w = funds(g, winner);
    if (!e || !w) return;
    w.sakuradite = (w.sakuradite || 0) + Math.round((e.sakuradite || 0) / 2);
    e.sakuradite = 0;
  }
  // Refineries at mines of their own (deposits under a city use the city's refinery building).
  function refineReason(g, d) {
    if (!d || d.city != null) return 'Unavailable';
    const foe = unitAt(g, d);
    return (
      (g.over ? 'Operation over' : d.owner !== g.phase ? 'Not your mine' : null) ||
      ((d.refinery || 0) >= 3 ? 'Maximum level' : null) ||
      (foe && foe.side !== d.owner ? 'Enemy unit on the mine' : null) ||
      shortfall(funds(g, d.owner), buildCost(d, 'refinery'))
    );
  }
  function refine(g, id) {
    const d = g.sites?.find(d => d.id === id),
      why = refineReason(g, d);
    if (why) return { ok: false, reason: why };
    const cost = buildCost(d, 'refinery');
    warSpend(g, d.owner, cost, 'construction');
    spend(funds(g, d.owner), cost);
    d.refinery = (d.refinery || 0) + 1;
    log(g, `${d.name}: Sakuradite refinery upgraded to level ${d.refinery}.`, d.owner);
    return { ok: true };
  }

  // ======== F.L.E.I.J.A.: the Sakuradite superweapon ========
  // The high-resolution world uses roughly 200 km hexes. A warhead reaches two rings: the first is catastrophic,
  // while the second is a weaker blast fringe. Campaign maps can still pass an explicit radius to blastArea().
  // Six World Campaign finales share their regional battles with Scenario Mode.
  // Outside these battles the long-standing scenario superweapon ban remains.
  const FINALE_SCENARIO_IDS = new Set([
    'eu-pendragon', 'cf-pendragon', 'battle-luoyang',
    'eu-cf-luoyang', 'cf-eu-france', 'operation-stonehenge'
  ]);
  function finaleScenario(g) {
    return g?.mode === 'scenario' && FINALE_SCENARIO_IDS.has(g.scenario?.id);
  }
  const FLEIJA = {
    radius: 2,
    cost: { credits: 1800, industry: 450, science: 300, sakuradite: 150 },
    turns: 4, // construction time
    max: 1, // combined stored and unfinished warheads; shared by Britannia and Euro Britannia in 2018
    lab: 3, // research lab level needed
    labTurn: 15, // 2017 World War: Research Lab III and F.L.E.I.J.A. unlock
    ufnLabTurn: 5, // 2018 Rise of the U.F.N.: earlier strategic-weapons unlock
    ringHP: 0.1, // first ring: units are left with 10% of their frame
    outerHP: 0.55, // second ring: units are left with at most 55%
    outerShield: 0.35, // second-ring cities retain at most 35% of their defenses
    aiThreshold: 1500, // the least target value a rival will spend a warhead on
  };
  const ELIMINATOR = {
    range: 3, // defense radius measured from the unit carrying the charge, or the city where it waits for one
    cost: { credits: 1200, industry: 300, science: 250, sakuradite: 60 },
    turns: 2,
    lab: 3,
    research: 2, // turns after the first detonation before Eliminators can be built
    max: 3, // charges a power may hold at once (building, waiting in a city or carried), one per city or unit
  };
  // The turn Eliminator research completes (g.fleijaDetonated is the turn of the first detonation), or null.
  function eliminatorTurn(g) {
    if (g.mode === 'scenario') return null; // All regional scenarios keep every city capturable.
    const first = g.fleijaDetonated;
    if (typeof first === 'number') return first + ELIMINATOR.research;
    if (first || (g.log || []).some(l => String(l.text || '').startsWith('F.L.E.I.J.A. detonation'))) return 0;
    return null;
  }
  function eliminatorUnlocked(g) {
    const at = eliminatorTurn(g);
    return at != null && g.turn >= at;
  }
  // Every Eliminator charge in play, positioned for dist(): one building or waiting in a city, or one carried by a
  // unit (aboard a Carrier-Battleship it covers from the ship). `ready` charges can intercept; a project completing
  // this turn already can, before its owner's phase begins.
  function eliminatorCharges(g, side = null) {
    const out = [];
    for (const s of g.stations) {
      if (side && s.owner !== side) continue;
      if ((s.eliminator || 0) > 0) out.push({ c: s.c, r: s.r, owner: s.owner, city: s, ready: true });
      else if (s.eliminatorProject?.side === s.owner)
        out.push({ c: s.c, r: s.r, owner: s.owner, city: s, ready: s.eliminatorProject.ready <= g.turn });
    }
    for (const v of g.units)
      if (v.hp > 0 && (!side || v.side === side))
        for (const u of [v, ...(v.cargo || [])])
          if (u.hp > 0 && (u.eliminator || 0) > 0) out.push({ c: v.c, r: v.r, owner: v.side, unit: u, ready: true });
    return out;
  }
  // A power's Eliminator charges: building, waiting in a city or carried.
  function sideEliminators(g, side) {
    return eliminatorCharges(g, side);
  }
  // How the owner's UI names whoever holds a charge.
  function eliminatorHolderName(holder) {
    if (!holder) return '';
    if (holder.city) return holder.city.name;
    const u = holder.unit;
    return u.cmd ? `${COMMANDERS[u.cmd]?.short || TYPES[u.type].short}'s ${TYPES[u.type].short}` : TYPES[u.type].short;
  }
  // A ready charge changes hands between a city and a unit in or next to it, or between two adjacent units. It is
  // free, but each charge moves once a turn, so it cannot relay down a line of units. `from`/`to` are
  // { city: id } or { unit: id }; units aboard a Carrier-Battleship cannot take part.
  function eliminatorHolder(g, ref) {
    if (ref?.city != null) return g.stations.find(s => s.id === ref.city) || null;
    if (ref?.unit != null) return g.units.find(u => u.id === ref.unit && u.hp > 0) || null;
    return null;
  }
  function transferEliminatorReason(g, from, to) {
    const a = eliminatorHolder(g, from),
      b = eliminatorHolder(g, to);
    if (g.mode === 'scenario') return 'F.L.E.I.J.A. Eliminators are unavailable in regional scenarios';
    if (g.over) return 'Operation over';
    if (!a || !b || a === b) return 'Unavailable';
    const side = from.city != null ? a.owner : a.side,
      other = to.city != null ? b.owner : b.side;
    if (from.city != null && to.city != null) return 'A charge moves between a city and a unit, or between units';
    if (side !== g.phase) return 'Not your turn';
    if (other !== side) return 'Only your own units and cities can hold your charges';
    if (!((a.eliminator || 0) > 0)) return 'No Eliminator charge to hand over';
    if ((b.eliminator || 0) > 0) return `${to.city != null ? 'That city' : 'That unit'} already holds an Eliminator charge`;
    if (to.city != null && b.eliminatorProject) return 'That city is building an Eliminator charge';
    if (a.eliminatorMoved === g.turn) return 'This charge has already changed hands this turn';
    if (dist(g, a, b) > 1) return 'Must be in the same or an adjacent hex';
    return null;
  }
  function transferEliminator(g, from, to) {
    const why = transferEliminatorReason(g, from, to);
    if (why) return { ok: false, reason: why };
    const a = eliminatorHolder(g, from),
      b = eliminatorHolder(g, to);
    a.eliminator = 0;
    delete a.eliminatorMoved;
    b.eliminator = 1;
    b.eliminatorMoved = g.turn;
    return { ok: true, holder: eliminatorHolderName(to.city != null ? { city: b } : { unit: b }) };
  }
  // A carrier that is destroyed takes its charge with it.
  function dropCarriedEliminator(g, v, why) {
    if (!((v?.eliminator || 0) > 0)) return;
    v.eliminator = 0;
    log(g, `A F.L.E.I.J.A. Eliminator charge is lost${why ? ' (' + why + ')' : ''}.`, v.side);
  }
  // F.L.E.I.J.A. is conquest-only. All major powers share their conquest's Lab III gate.
  function fleijaLabTurn(g) {
    return g.mode === 'conquest' && g.era === 'ufn2018' ? FLEIJA.ufnLabTurn : FLEIJA.labTurn;
  }
  function hasFleija(g, side) {
    return g.mode === 'conquest' && isMajor(g, side) && g.turn >= fleijaLabTurn(g);
  }
  // The 2018 Britannian allies share one strategic arsenal slot, but never
  // share funds, project ownership, a warhead's launch control, or unit orders.
  function fleijaPoolSides(g, side) {
    return g.mode === 'conquest' && g.era === 'ufn2018' &&
      (side === 'britannia' || side === 'eb') &&
      g.teams?.britannia && g.teams.britannia === g.teams.eb
      ? ['britannia', 'eb'] : [side];
  }
  function fleijaWarheadCount(g, side) {
    const members = fleijaPoolSides(g, side);
    return members.reduce((n, member) => n + (g.arsenal?.[member] || 0), 0) +
      g.stations.filter(s => members.includes(s.project?.side)).length;
  }
  // If a Britannian player has a funded, idle level-III lab, the AI ally must
  // not claim the newly free shared slot first (including after an AI launch).
  // The AI may use the slot if the player is unable to begin a project.
  function fleijaPlayerPriority(g, side) {
    if (side === g.player || !fleijaPoolSides(g, side).includes(g.player) ||
        !hasFleija(g, g.player) || g.over) return false;
    return g.stations.some(s => s.owner === g.player && !cityBusyReason(g, s) &&
      s.expSupplyBlocked?.owner !== g.player && (s.lab || 0) >= FLEIJA.lab && !shortfall(funds(g, g.player), FLEIJA.cost));
  }
  function cityBusyReason(g, s) {
    return s.project
      ? 'F.L.E.I.J.A. project under way'
      : s.eliminatorProject
        ? 'F.L.E.I.J.A. Eliminator project under way'
        : null;
  }
  function projectReason(g, s) {
    if (!s) return 'Unavailable';
    return (
      (g.over ? 'Operation over' : s.owner !== g.phase ? 'Not your city' : null) ||
      (!hasFleija(g, s.owner) ? (g.mode !== 'conquest'
        ? 'F.L.E.I.J.A. is unavailable outside conquests'
        : `Research lab level 3 unlocks on turn ${fleijaLabTurn(g)}`) : null) ||
      (s.expSupplyBlocked?.owner === s.owner ? 'City supply lines are disrupted' : null) ||
      cityBusyReason(g, s) ||
      (fleijaWarheadCount(g, s.owner) >= FLEIJA.max
        ? (fleijaPoolSides(g, s.owner).length > 1
          ? 'Britannia and Euro Britannia share one F.L.E.I.J.A. warhead slot (ready or under construction)'
          : 'Only one F.L.E.I.J.A. warhead may be held or under construction per faction') : null) ||
      ((s.lab || 0) < FLEIJA.lab ? `Requires research lab level ${FLEIJA.lab}` : null) ||
      shortfall(funds(g, s.owner), FLEIJA.cost)
    );
  }
  // Starting a warhead alerts every power; the city builds nothing else until it is done.
  function startProject(g, id) {
    const s = g.stations.find(s => s.id === id),
      why = projectReason(g, s);
    if (why) return { ok: false, reason: why };
    warSpend(g,s.owner,FLEIJA.cost,'F.L.E.I.J.A.');
    spend(funds(g, s.owner), FLEIJA.cost);
    s.project = { side: s.owner, started: g.turn, ready: g.turn + FLEIJA.turns };
    log(g, `INTELLIGENCE: Strategic weapons research detected in ${s.name}.`, s.owner);
    return { ok: true, ready: s.project.ready };
  }
  function eliminatorReason(g, s) {
    if (!s) return 'Unavailable';
    return (
      (g.over ? 'Operation over' : s.owner !== g.phase ? 'Not your city' : null) ||
      (eliminatorTurn(g) == null
        ? 'Available after the first F.L.E.I.J.A. detonation'
        : !eliminatorUnlocked(g)
          ? `Eliminator research completes on turn ${eliminatorTurn(g)}`
          : null) ||
      (s.expSupplyBlocked?.owner === s.owner ? 'City supply lines are disrupted' : null) ||
      cityBusyReason(g, s) ||
      ((s.lab || 0) < ELIMINATOR.lab ? `Requires research lab level ${ELIMINATOR.lab}` : null) ||
      ((s.eliminator || 0) > 0 ? 'This city already holds an Eliminator charge' : null) ||
      (sideEliminators(g, s.owner).length >= ELIMINATOR.max
        ? `At most ${ELIMINATOR.max} Eliminator charges at once (building, waiting or carried)`
        : null) ||
      shortfall(funds(g, s.owner), ELIMINATOR.cost)
    );
  }
  function startEliminator(g, id) {
    const s = g.stations.find(s => s.id === id),
      why = eliminatorReason(g, s);
    if (why) return { ok: false, reason: why };
    warSpend(g,s.owner,ELIMINATOR.cost,'Eliminator');
    spend(funds(g, s.owner), ELIMINATOR.cost);
    s.eliminatorProject = { side: s.owner, started: g.turn, ready: g.turn + ELIMINATOR.turns };
    // Defensive project locations are classified, unlike offensive warhead production.
    log(g, 'F.L.E.I.J.A. Eliminator development initiated at a classified site.', s.owner);
    return { ok: true, ready: s.eliminatorProject.ready };
  }
  function dropProject(g, s, why) {
    if (!s?.project) return;
    log(g, `${s.name}: the F.L.E.I.J.A. project is lost${why ? ' (' + why + ')' : ''}.`, s.owner);
    s.project = null;
  }
  function dropEliminator(g, s, why) {
    if (!s) return;
    if (s.eliminatorProject) {
      log(g, `A F.L.E.I.J.A. Eliminator project is lost${why ? ' (' + why + ')' : ''}.`, s.owner);
      s.eliminatorProject = null;
    }
    if (s.eliminator) {
      log(g, `A F.L.E.I.J.A. Eliminator charge is destroyed${why ? ' (' + why + ')' : ''}.`, s.owner);
      s.eliminator = 0;
      delete s.eliminatorMoved;
    }
  }
  // A surrendering power's projects and warheads are lost.
  function annexStrategic(g, loser) {
    for (const s of g.stations) {
      if (s.project?.side === loser) dropProject(g, s, 'surrender');
      if (s.eliminatorProject?.side === loser || (s.owner === loser && s.eliminator)) dropEliminator(g, s, 'surrender');
    }
    for (const v of g.units)
      if (v.side === loser) for (const u of [v, ...(v.cargo || [])]) dropCarriedEliminator(g, u, 'surrender');
    if (g.arsenal) g.arsenal[loser] = 0;
  }
  // Start of a power's turn: finished strategic projects come online.
  function strategicTurn(g, side) {
    if (g.mode === 'scenario') return; // Suppress even saved projects in every regional scenario.
    if (eliminatorTurn(g) === g.turn && !g.eliminatorAnnounced) {
      g.eliminatorAnnounced = true;
      log(g, 'INTELLIGENCE: F.L.E.I.J.A. Eliminator countermeasures are now available at level-3 research labs.', side);
    }
    for (const s of g.stations) {
      if (s.owner === side && s.expSupplyBlocked?.owner === side) {
        // Stop the production clock for this owner's turn; never complete a blocked project.
        for (const project of [s.project, s.eliminatorProject]) {
          if (project?.side === side && project.pausedTurn !== g.turn) {
            project.ready++;
            project.pausedTurn = g.turn;
          }
        }
        continue;
      }
      if (s.project?.side === side && s.owner === side && s.project.ready <= g.turn) {
        s.project = null;
        (g.arsenal ||= {})[side] = Math.min(FLEIJA.max, (g.arsenal[side] || 0) + 1);
        log(g, `${s.name} completes a F.L.E.I.J.A. warhead.`, side);
      }
      if (s.eliminatorProject?.side === side && s.owner === side && s.eliminatorProject.ready <= g.turn) {
        s.eliminatorProject = null;
        s.eliminator = 1;
        log(g, 'A F.L.E.I.J.A. Eliminator charge is ready at a classified site and waits for a carrier.', side);
      }
    }
  }
  function blastArea(g, p, radius = FLEIJA.radius) {
    return within(g, p, radius);
  }
  // The ready enemy charge nearest the target, carried or waiting in a city, or null. Carriers before cities at a tie.
  function eliminatorDefender(g, attacker, p) {
    if (!p) return null;
    return (
      [...eliminatorCharges(g), ...g.units
        .filter(u => u.hp > 0 && u.cmd === 'lloyd' && isMajor(g, u.side))
        .map(u => ({ c: u.c, r: u.r, owner: u.side, unit: u, ready: true, lloydPassive: true }))]
        .filter(h => h.ready && foe(g, h.owner, attacker) && isMajor(g, h.owner) && dist(g, h, p) <= ELIMINATOR.range)
        .sort(
          (a, b) =>
            dist(g, a, p) - dist(g, b, p) ||
            Number(!!a.city) - Number(!!b.city) ||
            (a.unit || a.city).id - (b.unit || b.city).id,
        )[0] || null
    );
  }
  // What a strike is called: the city or mine at ground zero, else the nearest city.
  function targetName(g, p) {
    const near = g.stations.slice().sort((a, b) => dist(g, a, p) - dist(g, b, p) || a.id - b.id)[0];
    return stationAt(g, p)?.name || siteAt(g, p)?.name || (near ? `near ${near.name}` : `hex ${p.c},${p.r}`);
  }
  function launchReason(g, side, p) {
    if (g.mode === 'scenario' && !finaleScenario(g)) return 'F.L.E.I.J.A. is unavailable in regional scenarios';
    if (finaleScenario(g) && g.turn < 5) return 'F.L.E.I.J.A. unlocks on turn 5';
    if (g.over) return 'Operation over';
    if (g.phase !== side) return 'Not your turn';
    if (!(g.arsenal?.[side] > 0)) return 'No F.L.E.I.J.A. warhead in the arsenal';
    // Stonehenge's expeditionary side starts offshore without a city.
    if (!finaleScenario(g) && !g.stations.some(s => s.owner === side)) return 'No city to launch from';
    if (!p || !tile(g, p.c, p.r)) return 'Choose a target hex';
    // No allied or own casualties, property damage, mine loss or city destruction anywhere in the blast radius.
    if (g.era === 'ufn2018' && blastArea(g, p).some(t => {
      const v = unitAt(g, t), city = stationAt(g, t), mine = siteAt(g, t);
      return (v && !foe(g, v.side, side)) ||
        (city && !foe(g, city.owner, side)) ||
        (mine && !foe(g, depositOwner(g, mine), side));
    })) return 'F.L.E.I.J.A. blast would harm friendly or allied forces, territory or facilities';
    if (stationAt(g, p)?.owner === side && g.stations.filter(s => s.owner === side).length === 1)
      return 'That is your last city';
    return null;
  }
  // Knock down every building by `levels` (Infinity: back to level 0) with the output and defenses they added.
  function ruin(g, s, levels) {
    buildingFoundation(g, s);
    const lostFactory = Math.min(levels, s.tier || 0),
      lostLab = Math.min(levels, s.lab || 0);
    s.tier = (s.tier || 0) - lostFactory;
    s.lab = (s.lab || 0) - lostLab;
    s.refinery = Math.max(0, (s.refinery || 0) - levels);
    s.portLevel = Math.max(0, (s.portLevel || 0) - levels);
    syncBuildings(g, s);
    s.shield = 0;
    dropProject(g, s, 'destroyed');
    dropEliminator(g, s, 'destroyed');
  }
  // Ground zero destroys a city for the rest of the battle: it stops being a city (no owner, output, port, project or
  // Eliminator charge, and it cannot be captured or rebuilt) and its deposit is lost. Its ruins stay on the map
  // (g.ruins) for the UI.
  function destroyCity(g, s) {
    dropProject(g, s, 'destroyed');
    dropEliminator(g, s, 'destroyed');
    const d = depositOf(g, s),
      lost = d ? destroyDeposit(g, d) : null;
    recordWarEvent(g,'cityDestroyed',s.owner,{cityId:s.id,city:s.name,owner:s.owner});
    (g.ruins ||= []).push({ name: s.name, c: s.c, r: s.r, owner: s.owner, capital: !!s.capital, turn: g.turn });
    g.stations.splice(g.stations.indexOf(s), 1);
    if (g.mode !== 'campaign') {
      // Its province does not become an unclaimable ghost region. Remaining
      // cities of the same faction inherit the land without changing its color.
      // If this was the last city, surrender will attach it to the victor.
      for (const t of g.tiles) if (t.provinceCity === s.id) delete t.provinceCity;
      assignCityProvinces(g);
    }
    if (g.automation?.cities) delete g.automation.cities[s.id];
    log(g, `${s.name} is destroyed by F.L.E.I.J.A.: only ruins remain for the rest of the war.`, s.owner);
    return lost;
  }
  // A deposit at ground zero never produces again.
  function destroyDeposit(g, d) {
    g.sites.splice(g.sites.indexOf(d), 1);
    log(g, `The ${d.name} Sakuradite deposit is destroyed and will never produce again.`, depositOwner(g, d) || g.phase);
    return d.name;
  }
  // Detonation: everything at ground zero is erased, the ring is left at 10% with collapsed morale.
  function launch(g, side, c, r) {
    const center = tile(g, c, r),
      why = launchReason(g, side, center);
    if (why) return { ok: false, reason: why };
    const origin = g.stations
        .filter(s => s.owner === side)
        .sort((a, b) => dist(g, a, center) - dist(g, b, center) || a.id - b.id)[0],
      name = targetName(g, center);
    g.arsenal[side]--;
    (g.launched ||= {})[side] = g.turn;
    const defense = eliminatorDefender(g, side, center);
    recordWarEvent(g,'fleijaLaunch',side,{city:name,c,r,intercepted:!!defense,defender:defense?.owner||null});
    if (defense) {
      // A charge completing this turn can defend before its owner's phase begins.
      // Two turns of research + two of production therefore beat the second warhead.
      const holder = defense.unit || defense.city;
      if (defense.city?.eliminatorProject?.ready <= g.turn) defense.city.eliminatorProject = null;
      if (!defense.lloydPassive) {
        holder.eliminator = 0;
        delete holder.eliminatorMoved;
      }
      log(g, `F.L.E.I.J.A. Eliminator neutralizes the incoming warhead aimed at ${name}.`, defense.owner);
      return {
        ok: true,
        side,
        from: origin ? { c: origin.c, r: origin.r } : { c, r },
        to: { c, r },
        name,
        intercepted: true,
        defender: defense.owner,
        eliminatorCity: eliminatorHolderName(defense),
        eliminatorCarrier: defense.unit ? defense.unit.id : null,
        destroyed: [],
        crippled: [],
        damaged: [],
        cities: [],
        hit: [],
      };
    }
    const unlocksEliminator = g.mode !== 'scenario' && eliminatorTurn(g) == null;
    const destroyed = [],
      crippled = [],
      damaged = [],
      cities = [],
      depleted = [],
      hit = [];
    for (const t of blastArea(g, center)) {
      const blastDistance = dist(g, t, center),
        ring = blastDistance === 1,
        outer = blastDistance > 1,
        v = unitAt(g, t),
        s = stationAt(g, t),
        d = siteAt(g, t);
      if (v && blastDistance === 0) {
        hit.push({ id: v.id, c: t.c, r: t.r, damage: v.hp });
        recordWarEvent(g,'damage',side,{targetSide:v.side,value:v.hp,weapon:'F.L.E.I.J.A.'});
        v.hp = 0;
        kill(g, v, null, true, side);
        destroyed.push(v.id);
      } else if (v) {
        const ratio = outer ? FLEIJA.outerHP : FLEIJA.ringHP,
          left = Math.min(v.hp, Math.max(1, Math.round(maxHP(v) * ratio)));
        hit.push({ id: v.id, c: t.c, r: t.r, damage: v.hp - left });
        recordWarEvent(g,'damage',side,{targetSide:v.side,value:v.hp-left,weapon:'F.L.E.I.J.A.'});
        v.hp = left;
        if (ring) {
          v.morale = moraleFloor(g, v);
          crippled.push(v.id);
        } else {
          v.morale = Math.min(v.morale ?? 0, -1);
          // Blue Geass also protects against the outer-ring morale shock.
          if (leilaCommand(g, v)) v.morale = Math.max(0, v.morale);
          damaged.push(v.id);
        }
      }
      if (s) {
        if (blastDistance === 0) {
          // A direct hit permanently removes the city in every mode, including
          // regional finales. Scenario victories count city ruins as eliminated.
          const lost = destroyCity(g, s);
          if (lost) depleted.push(lost);
          cities.push({ name: s.name, severity: 'ground', destroyed: true, owner: s.owner });
        } else if (ring) {
          ruin(g, s, 1);
          s.attackedSinceRegen = true;
          cities.push({ name: s.name, severity: 'inner' });
        } else {
          s.shield = Math.min(s.shield, Math.round(s.maxShield * FLEIJA.outerShield));
          s.attackedSinceRegen = true;
          cities.push({ name: s.name, severity: 'outer' });
        }
      }
      if (d) {
        if (blastDistance === 0) depleted.push(destroyDeposit(g, d));
        else if (ring) d.refinery = Math.max(0, (d.refinery || 0) - 1);
      }
      if (blastDistance === 0 && !navigable(t) && !TERRAIN[t.terrain]?.blocked) setTileTerrain(g, t, 'crater');
    }
    // Only an unintercepted enemy-city destruction at ground zero counts toward
    // this attacker's Tier 1 targeting milestone. Splash and blocked launches do not.
    if (cities.some(c => c.destroyed && foe(g, c.owner, side))) {
      const hits = (g.fleijaSuccessfulCityHits ||= {});
      hits[side] = (hits[side] || 0) + 1;
    }
    if (unlocksEliminator) g.fleijaDetonated = g.turn;
    // A power whose last city was destroyed surrenders to the launcher.
    const ground = cities.find(c => c.destroyed);
    if (ground && ground.owner !== side && alive(g, ground.owner) && !g.stations.some(c => c.owner === ground.owner))
      surrender(g, ground.owner, side, ground);
    log(
      g,
      `F.L.E.I.J.A. detonation at ${name}: ${destroyed.length} units erased, ${crippled.length} crippled, ${damaged.length} damaged${cities.length ? ', ' + cities.length + ' cities affected' : ''}.`,
      side,
    );
    if (unlocksEliminator)
      log(
        g,
        `INTELLIGENCE: every power begins F.L.E.I.J.A. Eliminator research; countermeasures are available from turn ${eliminatorTurn(g)}.`,
        side,
      );
    checkVictory(g);
    return {
      ok: true,
      side,
      from: origin ? { c: origin.c, r: origin.r } : { c, r },
      to: { c, r },
      name,
      destroyed,
      crippled,
      damaged,
      cities,
      depleted,
      hit,
      eliminatorUnlocked: unlocksEliminator,
      eliminatorTurn: eliminatorTurn(g),
    };
  }

  // ---- Rival high command and F.L.E.I.J.A. ----
  // The project city: the best lab, then the city farthest from the enemy.
  function fleijaCity(g, side, front) {
    return (
      g.stations
        .filter(s => s.owner === side && !cityBusyReason(g, s))
        .sort((a, b) => (b.lab || 0) - (a.lab || 0) || front(b) - front(a) || a.id - b.id)[0] || null
    );
  }
  // Where a rival builds its next Eliminator: a free level-3-lab city outside the cover of its other charges, the
  // capital first, then the richest. Null when there is none. `anyLab`: the best such city to raise a lab in instead.
  function eliminatorCity(g, side, anyLab = false) {
    const cover = sideEliminators(g, side);
    return (
      g.stations
        .filter(
          s =>
            s.owner === side &&
            !cityBusyReason(g, s) &&
            !(s.eliminator > 0) &&
            (anyLab || (s.lab || 0) >= ELIMINATOR.lab) &&
            !cover.some(c => dist(g, c, s) <= ELIMINATOR.range),
        )
        .sort(
          (a, b) =>
            (anyLab ? (b.lab || 0) - (a.lab || 0) : 0) ||
            Number(b.capitalOf === side) - Number(a.capitalOf === side) ||
            b.income + b.industry - (a.income + a.industry) ||
            a.id - b.id,
        )[0] || null
    );
  }
  // The most valuable target that spares the launcher's own units and cities, or null below the threshold.
  function aiLaunchTarget(g, side) {
    const rival = s => !!s && s !== 'neutral' && foe(g, s, side),
      seen = new Set();
    const tier1Phase = (g.fleijaSuccessfulCityHits?.[side] || 0) >= 5;
    let best = null;
    const candidates = [
      ...g.units.filter(u => u.hp > 0 && rival(u.side)),
      ...g.stations.filter(s => rival(s.owner)),
    ].map(p => tile(g, p.c, p.r));
    for (const p of candidates) {
      if (!p || seen.has(key(p))) continue;
      seen.add(key(p));
      let score = 0,
        safe = true;
      for (const t of blastArea(g, p)) {
        const blastDistance = dist(g, t, p),
          ring = blastDistance > 0,
          outer = blastDistance > 1,
          v = unitAt(g, t),
          s = stationAt(g, t);
        if ((v && !foe(g, v.side, side)) || (s && !foe(g, s.owner, side)) ||
            (g.era === 'ufn2018' && siteAt(g, t) && !foe(g, depositOwner(g, siteAt(g, t)), side))) {
          safe = false;
          break;
        }
        if (v && rival(v.side)) {
          score +=
            price(v.type, v.stack, g, v.side).credits * (v.hp / maxHP(v)) * (outer ? 0.35 : ring ? 0.75 : 1) + (v.cmd ? 200 : 0);
          if (tier1Phase && COMMANDERS[v.cmd]?.storyTier === 1)
            score += !ring ? 4500 : outer ? 300 : 1100;
        }
        // Cities are worth what the blast destroys: one at ground zero is erased with its output (and its owner
        // surrenders if it was the last); a capital in the inner ring is worth more only when the launcher has troops
        // close enough to take it afterwards.
        const d = !ring && (siteAt(g, t) || (s && depositOf(g, s)));
        if (d && rival(depositOwner(g, d))) score += 25 * d.base;
        if (s && rival(s.owner)) {
          const levels = (s.tier || 0) + (s.lab || 0) + (s.refinery || 0);
          score += outer
            ? 15 * levels + s.shield * 0.08
            : ring
              ? 40 * levels + s.shield * 0.2
              : 300 + 100 * levels + s.shield * 0.5 + 5 * (s.income + s.industry + s.science);
          if (!ring && isMajor(g, s.owner) && g.stations.filter(c => c.owner === s.owner).length === 1) score += 3000;
          if (s.project) score += 2000;
          if (
            ring &&
            !outer &&
            s.capitalOf === s.owner &&
            alive(g, s.owner) &&
            g.units.some(u => u.hp > 0 && u.side === side && canCapture(u) && dist(g, u, s) <= 4)
          )
            score += 1500;
        }
      }
      if (!safe) continue;
      const pick = { p, score };
      // Enemy Eliminator positions are secret: do not probe their interception coverage.
      if (!best || score > best.score) best = pick;
    }
    return best && best.score >= FLEIJA.aiThreshold ? best.p : null;
  }
  function beginTurn(g, side, collect = true) {
    // Entrenchment is earned when a power finishes its turn without moving the Infantry unit.
    // The protection persists through rival turns, then ends when that power's next turn begins.
    const outgoing = g.phase;
    // Suppression lasts through only this current faction turn, including
    // when a formation kills by off-turn counterfire.
    for (const v of g.units) delete v.expFarneseSlaughterTurn;
    if(outgoing){for(const v of g.units)if(v.side===outgoing){
        // Retaliation lasts through Guilford's NEXT full friendly action phase.
        if (v.expGuilfordRetribution &&
            --v.expGuilfordRetribution.turnsLeft <= 0)
          delete v.expGuilfordRetribution;
        if(v.expInformationWarfare)delete v.expInformationWarfare;
        if(v.expMargaretRedeploy){v.moved=!!v.expMargaretRedeploy.wasMoved;v.skillReposition=0;delete v.expMargaretRedeploy;}
        if(v.expShadowMove){v.moved=true;v.skillReposition=0;delete v.expShadowMove;}
        // Unused Emergency Defense movement expires when the owner's
        // faction ends its turn; the defensive aura remains through rivals.
        delete v.expOscarReposition;
        // Suppression remains for the target's full action phase.
        delete v.expMiguelSupply;
        // Mao's movement debuff expires after this faction's completed turn.
        delete v.expMaoMoveActive;
        delete v.expToruExtraAttack;delete v.expToruExtraMove;
        delete v.expFarneseExtraMove;
        delete v.expMargaretBonusTurn;delete v.expShadowAttackTurn;delete v.expPrecisionTarget;
        delete v.expSilenced;
        // Yukiya's electronic disruption expires only after the affected
        // faction completes its next turn.
        delete v.expYukiyaElectronicActive;
        // Sword Breaker persists throughout the affected side's next turn.
        delete v.expAyanoSwordBreak;
        // Expire Rolo's freeze only AFTER this faction has lost its entire
        // next turn, never at the start of its disabled activation.
        delete v.expRoloFrozen;
        // Charles's rewrite persists through the target's next full turn.
        // Restore morale only as that faction's action phase finishes.
        if (v.expCharlesMemoryRewrite) {
          v.morale = v.expCharlesMemoryRewrite.previousMorale;
          delete v.expCharlesMemoryRewrite;
        }
        // Ark Fleet's movement disruption ends only after the target's full action phase.
        if(v.expArkFleetSlow?.side===outgoing){
          v.expArkFleetSlow.turns--;
          if(v.expArkFleetSlow.turns<=0)delete v.expArkFleetSlow;
        }
        // Slow lasts through the target faction's next action phase.
        delete v.expKallenSlowActive;
        if(v.expChengWeaken?.turns>0&&!--v.expChengWeaken.turns)delete v.expChengWeaken;
        if(v.expPuristCounter&&v.expPuristCounter.startTurn<g.turn)delete v.expPuristCounter;
        if(v.expSmilasSuppression&&v.expSmilasSuppressionTurn<g.turn)delete v.expSmilasSuppression;
        if(v.despairWeakened?.source==='zi')delete v.despairWeakened;
      }
      for(const station of g.stations)if(station.expSupplyBlocked?.owner===outgoing){station.expSupplyBlocked.turns=(station.expSupplyBlocked.turns||1)-1;if(station.expSupplyBlocked.turns<=0)delete station.expSupplyBlocked;}
      // End Ark Fleet's recruitment and shield shutdown after one full defender turn.
      for(const city of g.stations)if(city.expArkFleet?.owner===outgoing){
        city.expArkFleet.turns--;
        if(city.expArkFleet.turns<=0)delete city.expArkFleet;
      }
    }
    if (outgoing && outgoing !== side)
      for (const u of g.units)
        if (u.hp > 0 && u.side === outgoing && !TYPES[u.type].naval && TYPES[u.type].branch === 'Infantry')
          u.entrenched = !u.lastTurnMoved;
    g.phase = side;
    if (collect) {
      const inc = income(g, side),
        e = funds(g, side),
        modifier = side !== g.player ? DIFFICULTIES[g.difficulty]?.income || 1 : 1;
      e.credits += Math.round(inc.credits * modifier);
      e.industry += Math.round(inc.industry * modifier);
      e.science += Math.round(inc.science * modifier);
      e.sakuradite = (e.sakuradite || 0) + Math.round(inc.sakuradite * modifier);
      recordWarEvent(g,'income',side,{credits:Math.round(inc.credits*modifier),
        industry:Math.round(inc.industry*modifier),science:Math.round(inc.science*modifier),
        sakuradite:Math.round(inc.sakuradite*modifier)});
    }
    for (const v of allUnits(g)) {
      // Emergency Defense lasts through opponents' turns, then expires at
      // the start of the designated ally's next friendly turn.
      if(v.expOscarDefense?.side===side)delete v.expOscarDefense;
      if(v.expFarneseWithdrawal?.side===side)delete v.expFarneseWithdrawal;
      // The stun persists through the opponent's turn, then expires when
      // Fernando's faction receives its next turn.
      if(v.expFernandoOverload?.side===side)delete v.expFernandoOverload;
      if(v.expKlausLoyalty?.side===side)delete v.expKlausLoyalty;
      if(v.expJoeDiagnosis?.until<g.turn)delete v.expJoeDiagnosis;
      v.skillMarks = (v.skillMarks || []).filter(m => m.side !== side);
      if (v.expCheckmate?.side === side) delete v.expCheckmate;
      for (const field of ['auraDisrupted', 'moraleWard', 'assaultInspired', 'despairWeakened']) if (v[field]?.side === side) delete v[field];
    }
    for (const city of g.stations){if(city.bombardMark?.side===side)delete city.bombardMark;if(city.expYoshidaBreak?.turn<g.turn)delete city.expYoshidaBreak;}
    // Off-map passengers have no terrain, morale or healing effects, but their
    // commander action cooldown advances on their faction's turn.
    for (const ship of g.units) if (ship.side === side)
      for (const passenger of ship.cargo || []) if (passenger.hp > 0 && passenger.side === side)
        passenger.feintCD = Math.max(0, (passenger.feintCD || 0) - 1);
    const mine = g.units.filter(u => u.hp > 0 && u.side === side);
    // Mao reads the intentions of every enemy commander within two hexes.
    // Each target takes one morale hit and one non-stacking movement penalty
    // per Mao faction turn. Moving out of range does not remove that penalty.
    const maos=mine.filter(u=>u.cmd==='mao'&&!atSea(g,u));
    if(maos.length){
      let affected=0;
      for(const enemy of g.units){
        if(enemy.hp<=0||!enemy.cmd||atSea(g,enemy)||!foe(g,enemy.side,side)||
           !maos.some(mao=>dist(g,mao,enemy)<=2))continue;
        lowerMorale(g,enemy,1);
        enemy.expMaoMovePending=true;
        affected++;
      }
      if(affected)log(g,'Psychological Pressure: '+affected+
        ' enemy commanders lose morale and 2 movement next turn.',side);
    }
    // Information Warfare is a friendly-turn pulse, not a continuous aura.
    for(const julius of mine.filter(v=>v.cmd==='julius'&&!atSea(g,v))){
      if(julius.arkFleetAirshipCD==null)julius.arkFleetAirshipCD=10;
      else julius.arkFleetAirshipCD=Math.max(0,julius.arkFleetAirshipCD-1);
      for(const enemy of g.units)if(enemy.hp>0&&foe(g,enemy.side,side)&&!atSea(g,enemy)&&dist(g,julius,enemy)<=3)
        lowerMorale(g,enemy,1);
    }
    // Existing negative morale is brought up to Steady immediately while
    // protected; this never promotes a unit from Steady to High.
    applyLeilaMoraleFloor(g);
    // Unbreakable Imperial Army is automatic, not an action or an aura that stacks.
    const emperors=mine.filter(v=>v.cmd==='lelouch'&&!atSea(g,v));
    for(const emperor of emperors)for(const ally of mine){
      if(!atSea(g,ally)&&dist(g,emperor,ally)<=2){
        ally.morale=1;
        ally.expImperialCommand={side,source:emperor.id,turn:g.turn};
      }
    }
    for (const u of mine) {
      // Existing saved Norland formations may carry a Geass debuff from
      // before his mask became passive. Undo it without touching other saves.
      if (u.cmd === 'norland' && fx(u).geassProofMask) {
        if (u.expCharlesMemoryRewrite?.previousMorale != null)
          u.morale = u.expCharlesMemoryRewrite.previousMorale;
        if (u.expZeroGeass)
          u.morale = Math.min(1, u.morale + (u.expZeroGeass.moraleLost || 0));
        delete u.expCharlesMemoryRewrite;
        delete u.expZeroGeass;
        delete u.expRoloFrozen;
        delete u.expGeassCancelled;
      }
      // Titan Mode stays active across opponents' turns, ending when Norland's
      // own faction next begins a turn. Cooldown continues independently.
      if (u.cmd === 'norland') delete u.expNorlandTitan;
      // Geass Absolute Command's enemy-mode effects expire at the START
      // of the affected faction's next turn, before ordinary morale recovery.
      if (u.expZeroGeass) {
        u.morale = Math.min(1, u.morale + (u.expZeroGeass.moraleLost || 0));
        delete u.expZeroGeass;
      }
      // Lasts through opponents' turns; expires on this formation's next turn.
      if(!u.expImperialCommand||u.expImperialCommand.turn!==g.turn)delete u.expImperialCommand;
      delete u.expMaskFalls;
      if(u.expDeathEmperor&&!--u.expDeathEmperor.turns)delete u.expDeathEmperor;
      delete u.expGeassCancelled;
      u.expKallenSlowActive = !!u.expKallenSlowNext;
      delete u.expKallenSlowNext;
      if (u.cmd === 'kallen') u.expKallenFirstTargets = [];
      delete u.expKallenReposition;
      // Do not clear expKallenMoveBank: unspent movement carries forward.
      u.entrenched = false;
      u.lastTurnMoved = false;
      u.guardReady = !!u.held;
      u.expRomeroMoved=false;
      u.movedDistance = 0;
      u.skillReposition = 0;
      u.withdrawMove = false;
      delete u.expAkitoMoveReady;
      u.moved = false;
      u.attacked = false;

      delete u.expForcedVulnerable;

      const nearby = g.units.filter(v => v.hp > 0 && foe(g, v.side, side) && dist(g, u, v) === 1).length;

      else if (u.morale > desired) u.morale--;

      if (mine.some(m => m.hp > 0 && fx(m).royalCompassion && dist(g, m, u) <= 2))
        u.morale = Math.min(1, u.morale + 1);

      if (!u.expMiguelSupply && TYPES[u.type].naval !== 'ship' &&

      if (!u.expMiguelSupply && repairLevel) u.hp = Math.min(maxHP(u), u.hp + Math.round(maxHP(u) * 0.01 * repairLevel));

      if(u.expToruExtraAttack||u.expToruExtraMove)u.morale=1;

      s.shield = Math.min(s.maxShield, s.shield + Math.round(recovery));
      delete s.attackedSinceRegen;
    }
    strategicTurn(g, side);
    hooks.turn?.(g, side);
    checkVictory(g);
  }
  // Fortress batteries: range 2 (3 with Overcharge), then always two turns to recharge.
  const FORTRESS_GUN = { range: 2, recharge: 2, fixed: 60, share: 0.1 };
  function fortressName(s) {
    return s.gun || `${s.name} battery`;
  }
  function fortressRecharge(g, s) {
    return FORTRESS_GUN.recharge;
  }
  function fortressRange(g, s) {
    return FORTRESS_GUN.range + (techLevel(g, s.owner, 'cities.overcharge') >= 1 ? 1 : 0);
  }
  function fortressReady(g, s) {
    return !!s?.fort && s.owner === g.phase && !g.over && s.shield > 0 && (s.gunReady || 0) <= g.turn;
  }
  // Battery Capacitors raise the owner's battery damage; Electromagnetic Armor shrugs part of it off.
  function fortressDamage(g, foe, owner) {
    return Math.max(
      1,
      Math.round(
        (FORTRESS_GUN.fixed + maxHP(foe) * FORTRESS_GUN.share) *
          (1 + techValue(g, owner, 'cities.battery')) *
          (!TYPES[foe.type].naval && TYPES[foe.type].branch === 'Armor' ? 1 - techValue(g, foe.side, 'armor.bulkheads') : 1),
      ),
    );
  }
  function fortressTargets(g, s) {
    if (!fortressReady(g, s)) return [];
    return within(g, s, fortressRange(g, s))
      .map(p => unitAt(g, p))
      .filter(u => u && foe(g, u.side, s.owner) &&
        !rangedImmuneTo(g,u,s))
      .map(u => tile(g, u.c, u.r));
  }
  function fireFortress(g, id, c, r) {
    const check = validateAction(g, { kind: 'fortress', id, c, r });
    if (!check.ok) return check;
    const s = check.station, foe = check.defender;
    const damage = fortressDamage(g, foe, s.owner),
      name = fortressName(s),
      hit = [];
    const priorHP=foe.hp;
    foe.hp = Math.max(0, foe.hp - damage);
    recordWarEvent(g,'damage',s.owner,{targetSide:foe.side,value:priorHP-foe.hp,city:s.name});
    lowerMorale(g, foe, 1);
    s.gunReady = g.turn + fortressRecharge(g, s);
    log(g, `${name} strikes ${TYPES[foe.type].short} for ${damage}.`, s.owner);
    // Battery Overcharge II: the blast also catches enemy units next to the target.
    if (techLevel(g, s.owner, 'cities.overcharge') >= 2)
      for (const v of g.units) {
        if (v.hp <= 0 || !isFoe(g, v.side, s.owner) || v.id === foe.id ||
            dist(g, v, foe) !== 1 || rangedImmuneTo(g,v,s)) continue;
        const amount = Math.round(fortressDamage(g, v, s.owner) * 0.4);
        const priorHP=v.hp;
        v.hp = Math.max(0, v.hp - amount);
        recordWarEvent(g,'damage',s.owner,{targetSide:v.side,value:priorHP-v.hp,city:s.name});
        hit.push({ id: v.id, c: v.c, r: v.r, damage: amount });
        kill(g, v, null, false, s.owner);
      }
    kill(g, foe, null, false, s.owner);
    const destroyed = foe.hp <= 0;
    checkVictory(g);
    return { ok: true, name, from: { c: s.c, r: s.r }, to: { c: foe.c, r: foe.r }, id: foe.id, damage, destroyed, hit };
  }
  function checkVictory(g) {

    decideVictory(g);
    if (g.over) finalizeHonors(g);
    return g.over;
  }

  // World conquest ends only when a faction or coalition eliminates every rival power.
  function decideVictory(g) {
    if (g.mode === 'campaign') return hooks.decide?.(g);
    if (g.mode === 'scenario') {
      const conquest = scenarioOf(g)?.winner(g),
        contenders = g.order.filter(s => g.stations.some(c => c.owner === s));
      // Amphibious regional invasions may start with no cities; surviving expeditions remain in play.
      if (scenarioMeta(g).expeditionSide) {
        if (conquest) g.over = conquest;
        return g.over;
      }
      if (!contenders.includes(g.player)) {
        // Player defeat can end their run, but award no faction a victory
        // until that faction controls every Japanese city.
        const soleVictor = conquest?.winner || null;
        g.over = {
          winner: soleVictor,
          eliminated: true,
          eliminatedBy: g.fallen?.[g.player]?.by || null,
          reason: soleVictor
            ? 'Your last stronghold in this theatre has fallen. The ' + FACTIONS[soleVictor].name +
              ' has conquered every city in the theatre.'
            : contenders.length === 1
              ? 'Your last stronghold in this theatre has fallen. The surviving power has not yet conquered every city in this theatre.'
              : 'Your last stronghold in this theatre has fallen. ' +
                'The war continues between ' +
                contenders.map(side => FACTIONS[side].name).join(' and ') + '.',
        };
        return g.over;
      }
      if (conquest) g.over = conquest;
      return g.over;
    }
    if (g.era === 'ufn2018') {
      const britannian = g.stations.filter(s => s.owner === 'britannia' || s.owner === 'eb').length;
      const ufn = g.stations.filter(s => s.owner === 'cf').length;
      const britPlayer = g.player === 'britannia' || g.player === 'eb';
      const coalitionWinner = britPlayer ? g.player : 'britannia';
      if (!ufn || !britannian) {
        const winningTeam = !ufn ? 'britannian' : 'ufn';
        g.over = {
          winner: winningTeam === 'ufn' ? 'cf' : coalitionWinner,
          team: winningTeam,
          reason: !ufn ? 'The U.F.N. has surrendered. The Britannian alliance is victorious.'
            : 'Both Britannian powers have surrendered. The U.F.N. is victorious.',
        };
      }
      return g.over;
    }
    const P = g.player,
      fall = g.fallen?.[P],
      rivals = MAJORS.filter(s => s !== P && alive(g, s));
    if (fall)
      g.over = { winner: fall.by, reason: `${fall.city}, your last city, has fallen. The ${FACTIONS[P].name} has surrendered.` };
    else if (!rivals.length)
      g.over = {
        winner: P,
        reason: `Every rival power has surrendered to the ${FACTIONS[P].name}. The world is yours.`,
      };
    return g.over;
  }
  function objectiveText(g) {
    if (g.mode === 'campaign' && hooks.objective) return hooks.objective(g);
    if (g.mode === 'scenario')
      return scenarioOf(g)?.objective(g) || 'Conquer every enemy city';
    if (g.era === 'ufn2018') return g.player === 'cf'
      ? 'Defeat both Britannian powers. Australia is not a victory target.'
      : 'Fight alongside the other Britannian power to eliminate the U.F.N. Australia is not a victory target.';
    const rivals = MAJORS.filter(s => s !== g.player && alive(g, s)).map(s => `the ${FACTIONS[s].short}`);
    return rivals.length
      ? `Defeat ${rivals.join(' and ')}. A power surrenders only when it has lost every city.`
      : 'Every rival power has surrendered.';
  }

  // Lower less-important ordinary city tiers to prevent Conquest economies
  // from ballooning after removing the hand-authored income overrides.
  function rebalanceConquestCityTiers(g) {
    const targets=g.era==='ufn2018'
      ? {britannia:1450,eb:600,cf:1550}
      : {britannia:1135,eu:1150,cf:1150};
    for(const [side,target] of Object.entries(targets)){
      const cities=g.stations.filter(s=>s.owner===side);
      let total=cities.reduce((sum,s)=>sum+cityStats(s).credits,0);
      const choices=cities.filter(s=>!s.capital&&!s.fort)
        .sort((a,b)=>cityTierOf(a)-cityTierOf(b)||a.id-b.id);
      while(total>target){
        const s=choices.find(city=>cityTierOf(city)>1);
        if(!s)break;
        const level=cityTierOf(s);
        const saved=CITY_STANDARD_STATS[level].credits-CITY_STANDARD_STATS[level-1].credits;
        if(total-saved<target&&total-target<saved/2)break;
        s.cityTier=level-1;
        total-=saved;
      }
    }
  }
  // Versioned conversion used both for new games and existing saved wars.
  function standardizeCities(g,migrating=false) {
    if(!g?.stations?.length||g.cityStandardVersion===CITY_STANDARD_VERSION)return false;
    for(const s of g.stations){
      const priorMax=Number(s.maxShield)||0;
      const fraction=priorMax>0?Math.max(0,Math.min(1,(Number(s.shield)||0)/priorMax)):1;
      s.cityTier=cityTierOf(s);
      if(s.fort&&s.cityTier<4)s.cityTier=4;
      if(g.mode==='scenario'){
        if(s.capitalOf||s.capital){
          s.regionalHQ=true;
          s.regionalHQOf=s.capitalOf||s.owner;
        }
        s.nationalCapital=isNationalCapital(g,s);
        s.capital=s.nationalCapital;
        // Regional HQs retain their strategic role but cannot give capital
        // conquest honors or national-capital economic/defensive bonuses.
        if(!s.nationalCapital)s.capitalOf=null;
      }else s.nationalCapital=!!s.capital;
      delete s.regionalIncome;
      delete s.divisionIncome;
      delete s.fortBonus;
      s._cityHPFraction=fraction;
    }
    if(g.mode==='conquest'&&!migrating)rebalanceConquestCityTiers(g);
    for(const s of g.stations){
      syncBuildings(g,s);
      s.shield=migrating?Math.round(s.maxShield*s._cityHPFraction):s.maxShield;
      delete s._cityHPFraction;
    }
    g.cityStandardVersion=CITY_STANDARD_VERSION;
    return true;
  }

  function repairCapeYorkOwnership(g) {

  function repairSakhalinTerritory(g, initial = false) {

  // Correct pre-fix 2018 saves that still have Gibraltar's Euro-Britannian
  // province running through western Morocco and the Sahara. A tile is repaired
  // only if it remains tied to Gibraltar (or has no old province ID), while
  // both neighbouring African cities still belong to Britannia. Captured
  // African cities and their genuine Euro-Britannian provinces are preserved.
  function repairGibraltarAfricanCorridor(g) {
    if (g.mode !== 'conquest' || g.era !== 'ufn2018' ||
        g.cols !== WORLD.cols || g.rows !== WORLD.rows) return;
    const city = name => g.stations.find(s => s.name === name);
    const gibraltar = city('Gibraltar');
    if (gibraltar?.owner !== 'eb' ||
        city('Casablanca')?.owner !== 'britannia' ||

  function addDifficultyReinforcements(g, d, sources, count) {
    for (let n = 0, tries = 0; n < count && sources.length && tries < count * 6; tries++) {
      const src = sources[Math.floor(random(g) * sources.length)],

        g.officers[u.cmd].rank = RANKS.length - 1;
    const used = {};
    for (const u of g.units) {
      const old = TYPES[u.type];
      if (!old) throw new Error('Unknown European War starting unit '+u.type);
      // Carrier-battleships remain ordinary hulls even on Elite difficulty.
      if (old.naval === 'ship') continue;
      const roster = g.scenario.eliteRosters[u.side] || [];
      let id = old.naval ? (naval[u.side] || naval[g.scenario.equipment?.[u.side]])?.[old.naval] : null;
      if (!old.naval) {
        const pool = roster.filter(id => eliteEligible(g, u.side, id) && !TYPES[ELITE_FORCES[id]?.type]?.naval);
        const sameClass = pool.filter(id => TYPES[ELITE_FORCES[id].type].cls === old.cls);
        const sameBranch = pool.filter(id => TYPES[ELITE_FORCES[id].type].branch === old.branch);
        const choices = sameClass.length ? sameClass : sameBranch.length ? sameBranch : pool;
        if (!choices.length) throw new Error('Missing European Elite Force for '+u.side);
        const signature = g.scenario.eliteSignatures?.[u.cmd] || signatures[u.cmd];
        const n = u.side+':'+old.cls;
        id = signature && roster.includes(signature)
          ? signature : choices[(used[n] || 0) % choices.length];
        used[n] = (used[n] || 0) + 1;
      }
      if (!id || !roster.includes(id) || !eliteEligible(g, u.side, id))
        throw new Error('Missing European naval/land Elite Force for '+u.side+': '+u.type);
      u.type = ELITE_FORCES[id].type;
      u.elite = id;
      // Rival commander-led starts are Lv.3; other conversions are Lv.1. Player-built elites use HQ levels.
      u.eliteLevel = u.cmd && foe(g, u.side, g.player) ? 3 : 1;
      u.eliteStart = true;
      u.stack = 1;
      if (u.cmd) u.cmdRank = g.officers[u.cmd]?.rank ?? u.cmdRank;
      u.hpTech = hullTech(g,u);
      u.hp = maxHP(u);
    }
    for (const s of g.stations) fortify(g,s);
    return g;
  }
  // Nearest tile to p (breadth-first, wrapping) that passes test.
  function nearest(g, p, test) {
    const start = tile(g, p.c, p.r),
      seen = new Set([key(start)]),
      queue = [start];
    while (queue.length) {
      const t = queue.shift();
      if (test(t)) return t;
      for (const n of adjacent(g, t))
        if (!seen.has(key(n))) {
          seen.add(key(n));
          queue.push(n);
        }
    }
    return null;
  }

  // ======== Pathfinding: goal fields and landmasses, for standing orders and the AI ========
  // The side-wide targets as [position, value] pairs: rival cities (a F.L.E.I.J.A. project outranks even a capital) and
  // Sakuradite mines held by others (Mount Fuji pulls almost like a capital). Rival Eliminator projects are classified,
  // so they never attract attention.
  function goalSeeds(g, side) {
    const seeds = [];
    for (const s of g.stations)
      if (foe(g, s.owner, side))
        seeds.push([s, s.project ? -8 : s.capitalOf && alive(g, s.owner) ? -6 : s.owner === 'neutral' ? 1 : 0]);
    for (const d of g.sites || []) if (d.city == null && foe(g, d.owner, side)) seeds.push([d, d.base >= 30 ? -5 : -1]);
    return seeds;
  }
  // Path cost from every hex to the nearest city this side wants (rival capitals count extra), over land and sea.
  // `seeds` ([position, value] pairs) replaces the side-wide targets, e.g. with one front's objectives; `only`
  // ('land' or 'sea') keeps the paths on one surface; 'air' crosses everything at one point a hex.
  // Optional unit routing excludes hexes that reachable()/move() cannot traverse.
  // An inaccessible goal can still seed an approach: standing orders finish beside it.
  function routePassable(g, u, p) {
    const t = TYPES[u.type], st = stationAt(g, p), occ = unitAt(g, p);
    if (t.naval === 'ship' && ((!t.air && !navigable(p)) || st)) return false;
    if (st && foe(g, st.owner, u.side) && !(canCapture(u) && st.shield <= 0)) return false;
    return !occ || occ === u || (occ.side === u.side && (t.naval || !canBoard(g, occ)));
  }
  function goalField(g, side, seeds = null, only = null, unit = null, maxCost = Infinity, ignoreTerrain = false) {
    const field = new Float32Array(g.tiles.length).fill(Infinity),
      hd = [],
      hi = [];
    // Binary heap over parallel arrays (distance, tile index).
    const push = (i, d) => {
      let n = hd.length;
      hd.push(d);
      hi.push(i);
      while (n > 0) {
        const p = (n - 1) >> 1;
        if (hd[p] <= d) break;
        hd[n] = hd[p];
        hi[n] = hi[p];
        n = p;
      }
      hd[n] = d;
      hi[n] = i;
    };
    const pop = () => {
      const top = hi[0],
        d = hd.pop(),
        i = hi.pop(),
        size = hd.length;
      if (size) {
        let n = 0;
        for (;;) {
          const l = 2 * n + 1,
            r = l + 1;
          let m = l < size && hd[l] < d ? l : -1;
          if (r < size && hd[r] < (m < 0 ? d : hd[l])) m = r;
          if (m < 0) break;
          hd[n] = hd[m];
          hi[n] = hi[m];
          n = m;
        }
        hd[n] = d;
        hi[n] = i;
      }
      return top;
    };
    for (const [p, d] of seeds || goalSeeds(g, side)) {
      const i = p.r * g.cols + p.c;
      if (d < field[i]) {
        field[i] = d;
        push(i, d);
      }
    }
    if (g.mode === 'campaign' && !seeds) {
      for (const u of g.units)
        if (u.hp > 0 && foe(g, u.side, side)) {
          const i = u.r * g.cols + u.c;
          if (field[i] > 2) {
            field[i] = 2;
            push(i, 2);
          }
        }
      for (const [c, r, d = -4] of g.campaign?.goals?.[side] || []) {
        const i = r * g.cols + c;
        field[i] = d;
        push(i, d);
      }
    }
    const nb = neighborTable(g),
      tiles = g.tiles,
      rivers = g.mode === 'scenario' && !!g.scenario?.riverBarriers; // decided once, not per step
    while (hd.length) {
      const d = hd[0],
        i = pop();
      if (d > field[i] || d >= maxCost) continue;
      const sea = isSea(tiles[i]);
      for (let k = i * 6; k < i * 6 + 6; k++) {
        const j = nb[k];
        if (j < 0) continue;
        const n = tiles[j];
        if (only !== 'air' && (TERRAIN[n.terrain]?.blocked || (only === 'sea' ? !navigable(n) : only === 'land' && isSea(n)))) continue;
        if (unit && !routePassable(g, unit, n)) continue;
        // Reverse search: the step from neighbour j toward goal i pays for i.
        const step = isSea(tiles[i]) || ignoreTerrain ? 1 : unit ? terrainCost(g, unit, tiles[i]) : terrainStepCost(g, tiles[i]);
        const nd = d + (only === 'sea' || only === 'air' ? 1 :
          (isSea(n) !== sea ? 4 : 0) + step + (rivers && !ignoreTerrain ? riverCrossingCost(g, tiles[i], n, unit) : 0));
        if (nd <= maxCost && nd < field[j]) {
          field[j] = nd;
          push(j, nd);
        }
      }
    }
    return field;
  }
  // Six neighbour indices per tile (-1 off the map), cached per map; terrain is read live since craters can appear.
  const neighborCache = new WeakMap();
  function neighborTable(g) {
    let nb = neighborCache.get(g.tiles);
    if (nb) return nb;
    nb = new Int32Array(g.tiles.length * 6).fill(-1);
    for (const t of g.tiles) adjacent(g, t).forEach((n, k) => (nb[(t.r * g.cols + t.c) * 6 + k] = n.r * g.cols + n.c));
    neighborCache.set(g.tiles, nb);
    return nb;
  }
  // Islands and continents, for deciding which troops need a ship (cached per map).
  const landCache = new WeakMap();
  function landmass(g) {
    const saved = landCache.get(g.tiles), revision = g.mapTerrainRevision || 0;
    if (saved?.revision === revision) return saved.labels;
    const m = new Int32Array(g.tiles.length).fill(-1);
    let id = 0;
    for (const t of g.tiles) {
      if (isSea(t) || TERRAIN[t.terrain]?.blocked || m[t.r * g.cols + t.c] >= 0) continue;
      const q = [t];
      m[t.r * g.cols + t.c] = id;
      while (q.length) {
        const x = q.pop();
        for (const n of adjacent(g, x))
          if (!isSea(n) && !TERRAIN[n.terrain]?.blocked && m[n.r * g.cols + n.c] < 0) {
            m[n.r * g.cols + n.c] = id;
            q.push(n);
          }
      }
      id++;
    }
    landCache.set(g.tiles, {revision,labels:m});
    return m;
  }
  const massOf = (g, p) => landmass(g)[p.r * g.cols + p.c];

  // ======== Standing orders ========
  // A player's unit can be given a destination (u.goto). At the start of each of its side's turns it moves as far
  // along the way as it can, until it arrives or the order is cancelled. Warships keep to the sea; other units take to
  // the sea only when the destination lies across it (amphibious frames go either way). It never attacks on its own.
  function gotoSurface(g, u, p) {
    const t = TYPES[u.type];
    if (t.air) return 'air';
    if (t.naval === 'ship') return 'sea';
    return !t.naval && !atSea(g, u) && massOf(g, u) >= 0 && massOf(g, u) === massOf(g, p) ? 'land' : null;
  }
  function routeField(g, u, p, avoidBlockers = true) {
    const surface = gotoSurface(g, u, p),
      field = goalField(g, u.side, [[p, 0]], surface, avoidBlockers ? u : null);
    // A stale map component or an impassable land route must not prevent a viable coastal journey.
    return surface === 'land' && !Number.isFinite(field[u.r * g.cols + u.c]) &&
      (!avoidBlockers || !Number.isFinite(goalField(g, u.side, [[p, 0]], surface)[u.r * g.cols + u.c]))
      ? goalField(g, u.side, [[p, 0]], null, avoidBlockers ? u : null) : field;
  }
  function gotoReason(g, u, p) {
    if (!u || u.hp <= 0) return 'Unavailable';
    if (g.over) return 'Operation over';
    if (u.side !== g.phase) return 'Not your unit';
    const t = p && tile(g, p.c, p.r),
      naval = TYPES[u.type].naval;
    if (!t) return 'Choose a hex on the map';
    const air = isAir(u);
    if (TERRAIN[t.terrain]?.blocked && !air) return 'Impassable terrain';
    if (air && stationAt(g, t)) return 'Airships cannot enter cities';
    if (naval === 'ship' && !air && (!navigable(t) || stationAt(g, t))) return 'Warships stay at sea';
    if (!naval && isSea(t)) return 'Choose a land hex';
    if (t.c === u.c && t.r === u.r) return 'Already there';
    // Temporary occupancy or defenses may clear before the unit arrives.
    return Number.isFinite(routeField(g, u, t, false)[u.r * g.cols + u.c]) ? null : 'No route there';
  }
  function setGoto(g, id, c, r) {
    const u = g.units.find(v => v.id === id),
      p = tile(g, c, r),
      why = gotoReason(g, u, p);
    if (why) return { ok: false, reason: why };
    u.goto = { c: p.c, r: p.r };
    return { ok: true, goto: u.goto };
  }
  // A regional scenario's Army Group order: route every living member to `target`.
  // Scenarios pass their own wording for a missing group and an unreachable target.
  function orderArmyGroup(g, id, target, missing, unreachable) {
    if (g.mode !== 'scenario' || g.phase !== g.player) return { ok: false, reason: 'Not your turn' };
    const group = g.scenario.groups.find(v => v.id === id && v.side === g.player);
    if (!group) return { ok: false, reason: missing };
    let assigned = 0, failed = 0;
    for (const unitId of group.units) {
      const u = g.units.find(v => v.id === unitId && v.hp > 0 && v.side === g.player);
      if (!u) continue;
      const res = setGoto(g, u.id, target.c, target.r);
      if (res.ok) assigned++; else failed++;
    }
    if (assigned) group.destination = { c: target.c, r: target.r };
    return { ok: assigned > 0, assigned, failed, reason: assigned ? '' : unreachable };
  }
  // The closest tile within `radius` rows and columns of `p` that passes `valid`.
  function nearestTile(g, p, valid, radius) {
    let best = null, d = Infinity;
    for (let r = Math.max(0, p.r - radius); r <= Math.min(g.rows - 1, p.r + radius); r++)
      for (let c = Math.max(0, p.c - radius); c <= Math.min(g.cols - 1, p.c + radius); c++) {
        const t = tile(g, c, r);
        if (!t || !valid(t)) continue;
        const n = distance(p, t, g);
        if (n < d) { d = n; best = t; }
      }
    return best;
  }
  function clearGoto(g, id) {
    const u = g.units.find(v => v.id === id);
    if (!u?.goto) return { ok: false, reason: 'No destination set' };
    delete u.goto;
    return { ok: true };
  }
  // Arrived: on the destination, or beside it when it cannot be entered (occupied, or an enemy city it cannot take).
  function gotoDone(g, u, p) {
    if (u.c === p.c && u.r === p.r) return true;
    if (dist(g, u, p) > 1) return false;
    const occ = unitAt(g, p),
      st = stationAt(g, p);
    return (!!occ && occ !== u) || (!!st && foe(g, st.owner, u.side) && !(canCapture(u) && st.shield <= 0));
  }
  // Moves every unit of `side` with a destination one turn along its route, nearest first. Returns what happened:
  // moved [{ id, from, to, captured, seized, annexed }], arrived [id] (order complete), blocked [id] (no free hex
  // farther along its legal route this turn) and lost [id] (no terrain route remains; order cancelled).
  function runGotos(g, side, { assaultOnly = false } = {}) {
    const report = { moved: [], arrived: [], blocked: [], lost: [] },
      fields = new Map();
    const orders = g.units
      // Never rerun a standing order after a unit has moved manually or by automation this turn.
      // Some elite/mobility skills refresh u.moved; autoOrderTurn closes that loophole.
      .filter(u => u.side === side && u.hp > 0 && u.goto && !u.moved && !u.lastTurnMoved && u.autoOrderTurn !== g.turn)
      // The post-fire capture pass must not rerun unrelated personal orders.
      .filter(u => !assaultOnly || (g.armyGroups || []).some(gr =>
        gr.side === side && gr.stance === 'assault' && gr.units.includes(u.id) &&
        gr.operation?.autoRouted?.[u.id]?.c === u.goto?.c &&
        gr.operation?.autoRouted?.[u.id]?.r === u.goto?.r))
      .sort((a, b) => dist(g, a, a.goto) - dist(g, b, b.goto) || a.id - b.id);
    for (const u of orders) {
      if (g.over) break;
      const dest = tile(g, u.goto.c, u.goto.r);
      const assault = (g.armyGroups || []).some(gr => gr.stance === 'assault' && gr.side === side && gr.units.includes(u.id));
      const defendedObjective = assault && dest && dist(g, u, dest) <= 1 &&
        ((unitAt(g, dest) && unitAt(g, dest) !== u && foe(g, unitAt(g, dest).side, side)) ||
         (stationAt(g, dest) && foe(g, stationAt(g, dest).owner, side) && stationAt(g, dest).shield > 0));
      if (dest && gotoDone(g, u, dest) && !defendedObjective) {
        report.arrived.push(u.id);
        delete u.goto;
        continue;
      }
      // A hostile, occupied or shielded objective may be unenterable this turn.
      // Hold the Assault order for bombardment rather than canceling an otherwise valid objective.
      // Amphibious units that can legally enter the tile still proceed with their capture.
      if (defendedObjective && !reachable(g, u).has(key(dest))) {
        report.blocked.push(u.id);
        continue;
      }
      // Replan separately for this unit against the live board, after earlier orders move.
      const field = dest && routeField(g, u, dest);
      const cost = p => field[p.r * g.cols + p.c],
        here = field ? cost(u) : Infinity;
      if (!Number.isFinite(here)) {
        const k = dest && `${dest.c},${dest.r},${gotoSurface(g, u, dest)}`;
        if (dest && !fields.has(k)) fields.set(k, routeField(g, u, dest, false));
        if (dest && Number.isFinite(fields.get(k)[u.r * g.cols + u.c])) report.blocked.push(u.id);
        else {
          report.lost.push(u.id);
          delete u.goto;
        }
        continue;
      }
      let best = null;
      for (const key of reachable(g, u).keys()) {
        const [c, r] = key.split(',').map(Number),
          p = tile(g, c, r),
          v = cost(p);
        if (unitAt(g, p) || !(v < here)) continue;
        if (!best || v < best.v || (v === best.v && dist(g, p, dest) < dist(g, best.p, dest))) best = { p, v };
      }
      const m = best && move(g, u.id, best.p.c, best.p.r);
      if (!m?.ok) {
        // Assault orders persist when adjacent to an enemy-held objective.
        if (defendedObjective) continue;
        report.blocked.push(u.id);
        continue;
      }
      u.autoOrderTurn = g.turn;
      report.moved.push({ id: u.id, from: m.from, to: m.to, captured: m.captured, seized: m.seized, annexed: m.annexed,
        ...(m.abilityEffects ? {abilityEffects:m.abilityEffects} : {}) });
      if (gotoDone(g, u, dest) && !(assault && dest && dist(g, u, dest) <= 1 &&
        ((unitAt(g, dest) && unitAt(g, dest) !== u && foe(g, unitAt(g, dest).side, side)) ||
         (stationAt(g, dest) && foe(g, stationAt(g, dest).owner, side) && stationAt(g, dest).shield > 0)))) {
        report.arrived.push(u.id);
        delete u.goto;
      }
    }
    return report;
  }
